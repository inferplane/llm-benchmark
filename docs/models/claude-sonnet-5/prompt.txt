        tx: 0, ty: 0,
        vx: 0, vy: 0,
        size: rand(1.2, 3.4),
        speed: rand(0.002, 0.01),
        wobble: rand(0, Math.PI * 2),
        wobbleSpeed: rand(0.01, 0.04),
        hueShift: Math.random(),
        scatterTarget: { x: 0, y: 0 }

    let targetX = pointer.active ? pointer.x : 0;

    const ease = reduceMotion ? 0.03 : 0.06;
    core.x += (targetX - core.x) * ease;
    core.y += (targetY - core.y) * ease;

      const gatheredY = core.y + orbitY + wob * 0.6;

      const scatteredX = core.x + p.scatterTarget.x;
      const scatteredY = core.y + p.scatterTarget.y;

      p.ty = gatheredY + (scatteredY - gatheredY) * scatterEnergy;

      const followEase = reduceMotion ? 0.04 : 0.12;
      p.x += (p.tx - p.x) * followEase;
      p.y += (p.ty - p.y) * followEase;
