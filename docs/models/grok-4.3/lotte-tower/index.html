  ctx.strokeStyle = '#ffcc66';
  ctx.lineWidth = 1.8;
  
  // Draw tapered form with ceramic-inspired lines
  for (let i = 0; i < segments; i++) {
    const t = i / segments;
    const t2 = (i+1) / segments;
    const w1 = baseW * (1-t) + topW * t;
    const w2 = baseW * (1-t2) + topW * t2;
    const y1 = height/2 - height * t;
    const y2 = height/2 - height * t2;
    
    // Front face
    const p1 = project(-w1, y1, 0);
    const p2 = project(w1, y1, 0);
    const p3 = project(w2, y2, 0);
    const p4 = project(-w2, y2, 0);
    
    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.lineTo(p3.x, p3.y);
    ctx.lineTo(p4.x, p4.y);
    ctx.closePath();
    ctx.stroke();
  }
  
  // Vertical ceramic lines
  ctx.strokeStyle = '#ffaa33';
  ctx.lineWidth = 1.2;
  for (let i = -3; i <= 3; i++) {
    const x = i * 18;
    const pTop = project(x, -height/2, 0);
    const pBot = project(x, height/2, 0);
    ctx.beginPath();
    ctx.moveTo(pTop.x, pTop.y);
    ctx.lineTo(pBot.x, pBot.y);
    ctx.stroke();
  }
}
