                    controls.enabled = true;
                } else {
                    if (viewType === 'site') createSitePlan();
                    else if (viewType === 'floor') createFloorPlan();
                    else if (viewType === 'elevation') createElevation();
                    else if (viewType === 'section') createSection();

                    controls.enabled = false;
                }
            });

            // Handle window resize
            window.addEventListener('resize', function() {
                const container = document.getElementById('model-canvas');
                const width = container.clientWidth;
                const height = container.clientHeight;

                camera.aspect = width / height;
                camera.updateProjectionMatrix();

                renderer.setSize(width, height);
            });
        });
    </script>
