// Initialize Three.js scene
let scene, camera, renderer, controls, towerMesh;
const modelContainer = document.getElementById('modelContainer');
const resetViewBtn = document.getElementById('resetView');

function initThreeJS() {
  scene = new THREE.Scene();
  
  const aspect = modelContainer.clientWidth / modelContainer.clientHeight;
  camera = new THREE.PerspectiveCamera(45, aspect, 0.1, 1000);
  camera.position.set(150, 150, 200);
  
  renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(modelContainer.clientWidth, modelContainer.clientHeight);
  modelContainer.appendChild(renderer.domElement);
  
  // Lights
  const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
  scene.add(ambientLight);
  
  const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
  dirLight.position.set(50, 100, 75);
  scene.add(dirLight);
  
  // Controls
  controls = new THREE.OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true;
  
  // Create initial tower
  createTower(0); // Concept stage
  
  animate();
}

function createTower(stage) {
  // Remove existing tower
  if (towerMesh) {
    scene.remove(towerMesh);
    towerMesh.geometry.dispose();
    towerMesh.material.dispose();
  }
  
  // Tower parameters based on stage
  let height, bottomRadius, topRadius, segments;
  
  switch(stage) {
    case 0: // concept
      height = 100;
      bottomRadius = 20;
      topRadius = 15;
      segments = 8;
      break;
    case 1: // massing
      height = 120;
      bottomRadius = 25;
      topRadius = 18;
      segments = 12;
      break;
    case 2: // taper
      height = 140;
      bottomRadius = 28;
      topRadius = 12;
      segments = 16;
      break;
    case 3: // facade
      height = 150;
      bottomRadius = 30;
      topRadius = 10;
      segments = 24;
      break;
    case 4: // final
      height = 155; // scaled for view
      bottomRadius = 32;
      topRadius = 8;
      segments = 32;
      break;
    default:
      height = 100;
      bottomRadius = 20;
      topRadius = 15;
      segments = 8;
  }
  
  const geometry = new THREE.CylinderGeometry(topRadius, bottomRadius, height, segments);
  const material = new THREE.MeshLambertMaterial({ 
    color: 0x88ccff,
    wireframe: stage < 3 ? true : false
  });
  
  towerMesh = new THREE.Mesh(geometry, material);
  towerMesh.position.y = height / 2;
  scene.add(towerMesh);
}

function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}

function onWindowResize() {
  const aspect = modelContainer.clientWidth / modelContainer.clientHeight;
  camera.aspect = aspect;
  camera.updateProjectionMatrix();
  renderer.setSize(modelContainer.clientWidth, modelContainer.clientHeight);
}
