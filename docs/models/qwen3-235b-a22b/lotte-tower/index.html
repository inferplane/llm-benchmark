              te[15] = te[8]*me[3] + te[9]*me[7] + te[10]*me[11] + te[11]*me[15];
              te[8] = t1; te[9] = t2; te[10] = t3;
              t1 = te[12]*me[0] + te[13]*me[4] + te[14]*me[8] + te[15]*me[12];
              t2 = te[12]*me[1] + te[13]*me[5] + te[14]*me[9] + te[15]*me[13];
              t3 = te[12]*me[2] + te[13]*me[6] + te[14]*me[10] + te[15]*me[14];
              te[15] = te[12]*me[3] + te[13]*me[7] + te[14]*me[11] + te[15]*me[15];
              te[12] = t1; te[13] = t2; te[14] = t3;
              return this;
            }
            getElement(row, col) { return this.elements[col * 4 + row]; }
            setElement(row, col, value) { this.elements[col * 4 + row] = value; }
            setPerspective(fov, aspect, near, far) {
              const f = 1 / Math.tan(fov * Math.PI / 360);
              const nf = 1 / (near - far);
              this.elements[0] = f / aspect; this.elements[1] = 0; this.elements[2] = 0; this.elements[3] = 0;
              this.elements[4] = 0; this.elements[5] = f; this.elements[6] = 0; this.elements[7] = 0;
              this.elements[8] = 0; this.elements[9] = 0; this.elements[10] = (far + near) * nf; this.elements[11] = -1;
              this.elements[12] = 0; this.elements[13] = 0; this.elements[14] = 2 * far * near * nf; this.elements[15] = 0;
              return this;
            }
            setPosition(v) { this.elements[12] = v.x; this.elements[13] = v.y; this.elements[14] = v.z; }
            getInverse(m) {
              const me = m.elements;
              const te = this.elements;
              const n11 = me[0], n12 = me[1], n13 = me[2], n14 = me[3];
              const n21 = me[4], n22 = me[5], n23 = me[6], n24 = me[7];
              const n31 = me[8], n32 = me[9], n33 = me[10], n34 = me[11];
              const n41 = me[12], n42 = me[13], n43 = me[14], n44 = me[15];
              const t11 = n23*n34*n42 - n24*n33*n42 + n24*n32*n43 - n22*n34*n43 - n23*n32*n44 + n22*n33*n44;
              const t12 = n14*n33*n42 - n13*n34*n42 - n14*n32*n43 + n12*n34*n43 + n13*n32*n44 - n12*n33*n44;
              const t13 = n13*n24*n42 - n14*n23*n42 + n14*n22*n43 - n12*n24*n43 - n13*n22*n44 + n12*n23*n44;
              const t14 = n14*n23*n32 - n13*n24*n32 - n14*n22*n33 + n12*n24*n33 + n13*n22*n34 - n12*n23*n34;
              const det = n11*t11 + n21*t12 + n31*t13 + n41*t14;
              if (det === 0) return this.identity();
              const detInv = 1 / det;
              te[0] = t11 * detInv; te[4] = (n24*n33*n41 - n23*n34*n41 - n24*n31*n43 + n21*n34*n43 + n23*n31*n44 - n21*n33*n44) * detInv;
              te[8] = (n22*n34*n41 - n24*n32*n41 + n24*n31*n42 - n21*n34*n42 - n22*n31*n44 + n21*n32*n44) * detInv;
              te[12] = (n23*n32*n41 - n22*n33*n41 - n23*n31*n42 + n21*n33*n42 + n22*n31*n43 - n21*n32*n43) * detInv;
              te[1] = t12 * detInv; te[5] = (n13*n34*n41 - n14*n33*n41 + n14*n31*n43 - n11*n34*n43 - n13*n31*n44 + n11*n33*n44) * detInv;
              te[9] = (n14*n32*n41 - n12*n34*n41 - n14*n31*n42 + n11*n34*n42 + n12*n31*n44 - n11*n32*n44) * detInv;
              te[13] = (n12*n33*n41 - n13*n32*n41 + n13*n31*n42 - n11*n33*n42 - n12*n31*n43 + n11*n32*n43) * detInv;
              te[2] = t13 * detInv; te[6] = (n14*n23*n41 - n13*n24*n41 - n14*n21*n43 + n11*n24*n43 + n13*n21*n44 - n11*n23*n44) * detInv;
              te[10] = (n12*n24*n41 - n14*n22*n41 + n14*n21*n42 - n11*n24*n42 - n12*n21*n44 + n11*n22*n44) * detInv;
              te[14] = (n13*n22*n41 - n12*n23*n41 - n13*n21*n42 + n11*n23*n42 + n12*n21*n43 - n11*n22*n43) * detInv;
              te[3] = t14 * detInv; te[7] = (n13*n24*n31 - n14*n23*n31 + n14*n21*n33 - n11*n24*n33 - n13*n21*n34 + n11*n23*n34) * detInv;
              te[11] = (n14*n22*n31 - n12*n24*n31 - n14*n21*n32 + n11*n24*n32 + n12*n21*n34 - n11*n22*n34) * detInv;
              te[15] = (n12*n23*n31 - n13*n22*n31 + n13*n21*n32 - n11*n23*n32 - n12*n21*n33 + n11*n22*n33) * detInv;
              return this;
            }
          };

          const PerspectiveCamera = class {
            constructor(fov = 50, aspect = 1, near = 0.1, far = 2000) {
              this.fov = fov; this.aspect = aspect; this.near = near; this.far = far;
              this.position = new Vector3();

          const WebGLRenderer = class {
            constructor({ canvas }) {
              this.canvas = canvas;
              this.gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
              if (!this.gl) throw new Error('WebGL not supported');
              const gl = this.gl;
              this.domElement = canvas;
              this.setSize = function(width, height, updateStyle = false) {
                canvas.width = width * window.devicePixelRatio;
                canvas.height = height * window.devicePixelRatio;
                if (updateStyle) {
                  canvas.style.width = width + 'px';
                  canvas.style.height = height + 'px';
                }
                gl.viewport(0, 0, canvas.width, canvas.height);
              };
              this.setClearColor = function(color) {
                const c = new Color(color);
                gl.clearColor(c.r, c.g, c.b, 1.0);
                return this;
              };
              this.clear = function() { gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT); };
              this.render = function(scene, camera) {
                gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
                gl.enable(gl.DEPTH_TEST);
                gl.enable(gl.BLEND);
                gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);
                const projMat = camera.projectionMatrix;
                const viewMat = camera.matrixWorldInverse.getInverse(camera.matrixWorld);
                const viewProjMat = new Matrix4().multiply(projMat).multiply(viewMat);
                scene.children.forEach(child => {
                  if (child instanceof Mesh) {
                    renderMesh(child, viewProjMat, gl);
                  } else if (child instanceof Group) {
                    child.children.forEach(mesh => {
                      if (mesh instanceof Mesh) {
                        renderMesh(mesh, viewProjMat, gl);
                      }
                    });
                  }
                });
              };
            }

                const dir = e.deltaY > 0 ? 1.1 : 0.9;
                camera.position.multiplyScalar(dir);

              });

          function renderMesh(mesh, viewProjMat, gl) {
            const vertices = mesh.geometry.vertices;
            const indices = mesh.geometry.indices;
            const color = mesh.material.color;
            const pos = mesh.position;
            const projVertices = [];
            for (let i = 0; i < vertices.length; i += 3) {
              const x = vertices[i] + pos.x;
              const y = vertices[i+1] + pos.y;
              const z = vertices[i+2] + pos.z;
              const w = viewProjMat.getElement(3,0)*x + viewProjMat.getElement(3,1)*y + viewProjMat.getElement(3,2)*z + viewProjMat.getElement(3,3);
              if (w <= 0) continue;
              const px = (viewProjMat.getElement(0,0)*x + viewProjMat.getElement(0,1)*y + viewProjMat.getElement(0,2)*z + viewProjMat.getElement(0,3)) / w;
              const py = (viewProjMat.getElement(1,0)*x + viewProjMat.getElement(1,1)*y + viewProjMat.getElement(1,2)*z + viewProjMat.getElement(1,3)) / w;
              const pz = (viewProjMat.getElement(2,0)*x + viewProjMat.getElement(2,1)*y + viewProjMat.getElement(2,2)*z + viewProjMat.getElement(2,3)) / w;
              projVertices.push({ x: px, y: py, z: pz, r: color.r, g: color.g, b: color.b });
            }
            gl.beginPath();
            for (let i = 0; i < indices.length; i += 3) {
              const i1 = indices[i], i2 = indices[i+1], i3 = indices[i+2];
              if (i1 >= projVertices.length || i2 >= projVertices.length || i3 >= projVertices.length) continue;
              const v1 = projVertices[i1], v2 = projVertices[i2], v3 = projVertices[i3];
              if (!v1 || !v2 || !v3) continue;
              const cx = gl.canvas.width / 2;
              const cy = gl.canvas.height / 2;
              const scale = Math.min(cx, cy);
              const x1 = cx + v1.x * scale;
              const y1 = cy - v1.y * scale;
              const x2 = cx + v2.x * scale;
              const y2 = cy - v2.y * scale;
              const x3 = cx + v3.x * scale;
              const y3 = cy - v3.y * scale;
              if (v1.z < 0 && v2.z < 0 && v3.z < 0) continue;
              gl.moveTo(x1, y1);
              gl.lineTo(x2, y2);
              gl.lineTo(x3, y3);
              gl.closePath();
            }
            gl.strokeStyle = 'rgba(0,51,102,0.8)';
            gl.lineWidth = 1;
            gl.stroke();
            gl.fillStyle = 'rgba(0,51,102,0.1)';
            gl.fill();
          }

          return {
            Scene, PerspectiveCamera, WebGLRenderer, BoxGeometry,
            MeshBasicMaterial, Mesh, Group, Vector3, Euler, Quaternion,
            Matrix4, Color, Object3D, OrbitControls
          };

    let scene, modelCamera, modelRenderer, controls;
    function init3D() {
      scene = new THREE.Scene();
      modelCamera = new THREE.PerspectiveCamera(50, 1, 0.1, 2000);
      modelRenderer = new THREE.WebGLRenderer({ canvas: modelCanvas, antialias: true });
      modelRenderer.setClearColor('white');

    function updateModel(stageKey) {
      while(scene.children.length) {
        const obj = scene.children[0];
        if (obj.geometry) obj.geometry.dispose();
        if (obj.material) obj.material.dispose();
        scene.remove(obj);
      }
      const geo = stageGeometries[stageKey] || stageGeometries.final;
      const group = new THREE.Group();
      const segments = 20;
      for (let i = 0; i < segments; i++) {
        const segmentHeight = geo.height / segments;
        const t = i / segments;
        const baseSize = geo.base * (1 - t * 0.8);
        const h = segmentHeight * 0.9;
        const y = (geo.height / 2) - (i + 0.5) * segmentHeight;
        const mesh = new THREE.Mesh(
          new THREE.BoxGeometry(baseSize, h, baseSize),
          new THREE.MeshBasicMaterial({ color: '#003366', wireframe: true })
        );
        mesh.position.y = y;
        group.add(mesh);
      }
      scene.add(group);
    }
