        function drawSitePlan(canvas) {
            const ctx = canvas.getContext('2d');
            const w = canvas.width;
            const h = canvas.height;

            ctx.fillStyle = '#0a0e27';
            ctx.fillRect(0, 0, w, h);

            // 격자
            ctx.strokeStyle = 'rgba(255, 200, 100, 0.1)';
            ctx.lineWidth = 0.5;
            for (let i = 0; i < w; i += 40) {
                ctx.beginPath();
                ctx.moveTo(i, 0);
                ctx.lineTo(i, h);
                ctx.stroke();
            }
            for (let i = 0; i < h; i += 40) {
                ctx.beginPath();
                ctx.moveTo(0, i);
                ctx.lineTo(w, i);
                ctx.stroke();
            }

            // 부지 경계
            ctx.strokeStyle = '#ffc864';
            ctx.lineWidth = 2;
            ctx.strokeRect(w * 0.15, h * 0.15, w * 0.7, h * 0.7);

            // 건물 풋프린트 (원형)
            ctx.fillStyle = 'rgba(255, 200, 100, 0.2)';
            ctx.beginPath();
            ctx.arc(w * 0.5, h * 0.5, w * 0.25, 0, Math.PI * 2);
            ctx.fill();

            ctx.strokeStyle = '#ffc864';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(w * 0.5, h * 0.5, w * 0.25, 0, Math.PI * 2);
            ctx.stroke();

            // 라벨
            ctx.fillStyle = '#ffc864';
            ctx.font = 'bold 16px sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText('롯데월드타워', w * 0.5, h * 0.5);
            ctx.font = '12px sans-serif';
            ctx.fillText('부지 배치도', w * 0.5, h * 0.85);
        }

        function drawFloorPlan(canvas) {
            const ctx = canvas.getContext('2d');
            const w = canvas.width;
            const h = canvas.height;

            ctx.fillStyle = '#0a0e27';
            ctx.fillRect(0, 0, w, h);

            // 격자
            ctx.strokeStyle = 'rgba(255, 200, 100, 0.1)';
            ctx.lineWidth = 0.5;
            for (let i = 0; i < w; i += 40) {
                ctx.beginPath();
                ctx.moveTo(i, 0);
                ctx.lineTo(i, h);
                ctx.stroke();
            }
            for (let i = 0; i < h; i += 40) {
                ctx.beginPath();
                ctx.moveTo(0, i);
                ctx.lineTo(w, i);
                ctx.stroke();
            }

            // 기준층 평면 (원형, 중층부 기준)
            ctx.fillStyle = 'rgba(255, 200, 100, 0.15)';
            ctx.beginPath();
            ctx.arc(w * 0.5, h * 0.5, w * 0.28, 0, Math.PI * 2);
            ctx.fill();

            ctx.strokeStyle = '#ffc864';
            ctx.lineWidth = 2.5;
            ctx.beginPath();
            ctx.arc(w * 0.5, h * 0.5, w * 0.28, 0, Math.PI * 2);
            ctx.stroke();

            // 내부 구조 표시
            ctx.strokeStyle = 'rgba(255, 200, 100, 0.4)';
            ctx.lineWidth = 1;
            for (let angle = 0; angle < Math.PI * 2; angle += Math.PI / 6) {
                const x1 = w * 0.5 + Math.cos(angle) * w * 0.1;
                const y1 = h * 0.5 + Math.sin(angle) * w * 0.1;
                const x2 = w * 0.5 + Math.cos(angle) * w * 0.28;
                const y2 = h * 0.5 + Math.sin(angle) * w * 0.28;
                ctx.beginPath();
                ctx.moveTo(x1, y1);
                ctx.lineTo(x2, y2);
                ctx.stroke();
            }

            // 라벨
            ctx.fillStyle = '#ffc864';
            ctx.font = 'bold 16px sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText('기준층 평면도', w * 0.5, h * 0.85);
            ctx.font = '12px sans-serif';
            ctx.fillStyle = '#ffb366';
            ctx.fillText('약 6,000 m² (개념 추정)', w * 0.5, h * 0.92);
        }

        }

        function initThreeJS() {
            const container = document.getElementById('modelViewer');
            const width = container.clientWidth;
            const height = container.clientHeight;

            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x050a14);

            camera = new THREE.PerspectiveCamera(75, width / height, 0.1, 10000);
            camera.position.set(0, 150, 200);

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(width, height);
            renderer.setPixelRatio(window.devicePixelRatio);
            container.appendChild(renderer.domElement);

            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.autoRotate = false;
            controls.target.set(0, 150, 0);
            controls.update();

            // 조명
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
            scene.add(ambientLight);

            const directionalLight = new THREE.DirectionalLight(0xffffff, 0.8);
            directionalLight.position.set(100, 200, 100);
            scene.add(directionalLight);

            // 애니메이션 루프
            function animate() {
                requestAnimationFrame(animate);
                controls.update();
                renderer.render(scene, camera);
            }
            animate();

            // 윈도우 리사이즈
            window.addEventListener('resize', () => {
                const newWidth = container.clientWidth;
                const newHeight = container.clientHeight;
                camera.aspect = newWidth / newHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(newWidth, newHeight);
            });
        }

        function createTowerGeometry(stage) {
            const group = new THREE.Group();

            // 기본 파라미터
            const baseRadius = 80;
            const topRadius = 20;
            const height = 555;
            const segments = 32;

            let radiusFunc, colorFunc;

            if (stage === 0) {
                // 개념 단계: 단순 원뿔
                radiusFunc = (t) => baseRadius * (1 - t * 0.75);
                colorFunc = () => new THREE.Color(0xff8844);
            } else if (stage === 1) {
                // 기본 설계: 더 부드러운 테이퍼
                radiusFunc = (t) => baseRadius * Math.pow(1 - t, 1.2);
                colorFunc = () => new THREE.Color(0xffc864);
            } else if (stage === 2) {
                // 상세 설계: 곡선 형태
                radiusFunc = (t) => baseRadius * (1 - t * 0.8 + Math.sin(t * Math.PI) * 0.1);
                colorFunc = () => new THREE.Color(0xffb366);
            } else {
                // 완성 형상: 한국 도자기 곡선 참고
                radiusFunc = (t) => baseRadius * Math.pow(1 - t * 0.85, 1.1) * (1 + Math.sin(t * Math.PI * 2) * 0.05);
                colorFunc = () => new THREE.Color(0xffc864);
            }

            // 메인 타워 지오메트리
            const geometry = new THREE.BufferGeometry();
            const vertices = [];
            const indices = [];
            const colors = [];

            for (let i = 0; i <= segments; i++) {
                const angle = (i / segments) * Math.PI * 2;
                for (let j = 0; j <= 50; j++) {
                    const t = j / 50;
                    const radius = radiusFunc(t);
                    const y = t * height;
                    const x = Math.cos(angle) * radius;
                    const z = Math.sin(angle) * radius;

                    vertices.push(x, y, z);
                    colors.push(...colorFunc().toArray());
                }
            }

            // 인덱스 생성
            for (let i = 0; i < segments; i++) {
                for (let j = 0; j < 50; j++) {
                    const a = i * 51 + j;
                    const b = i * 51 + j + 1;
                    const c = ((i + 1) % (segments + 1)) * 51 + j;
                    const d = ((i + 1) % (segments + 1)) * 51 + j + 1;

                    indices.push(a, c, b);
                    indices.push(b, c, d);
                }
            }

            geometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array(vertices), 3));
            geometry.setAttribute('color', new THREE.BufferAttribute(new Float32Array(colors), 3));
            geometry.setIndex(new THREE.BufferAttribute(new Uint32Array(indices), 1));
            geometry.computeVertexNormals();

            const material = new THREE.MeshPhongMaterial({
                vertexColors: true,
                emissive: 0x1a1a1a,
                shininess: 100,
                wireframe: false
            });

            const mesh = new THREE.Mesh(geometry, material);
            group.add(mesh);

            // 기초 표시
            const baseGeometry = new THREE.CylinderGeometry(baseRadius * 1.2, baseRadius * 1.5, 20, segments);
            const baseMaterial = new THREE.MeshPhongMaterial({ color: 0x666666 });
            const baseMesh = new THREE.Mesh(baseGeometry, baseMaterial);
            baseMesh.position.y = -10;
            group.add(baseMesh);

            return group;
        }

        function updateModel(stage) {
            if (currentModel) {
                scene.remove(currentModel);
            }
            currentModel = createTowerGeometry(stage);
            scene.add(currentModel);
            currentStage = stage;
        }
