(function(){
  "use strict";

  /* ---------- 공통 데이터 ---------- */
  const H_M = 555;          // 공개 값: 높이 (m)
  const FLOORS = 123;       // 공개 값: 지상 층수
  const BASE_M = 120;       // 추정: 기준 외곽 한 변 (m)
  const SITE_W = 400, SITE_H = 300; // 추정: 대지 개념 크기 (m)
  const H_U = 111;          // 3D 단위 높이 (1 unit = 5 m, 555/5)
  const BASE_U = 24;        // 3D 단위 기준 폭 (= 120 m / 5, 추정)
  const SLABS = 90;

  // 단계별 외곽 비율 p(t): t=0 지상, t=1 최상단. 모두 추정 개념 함수.
  const STAGES = [
    { name:"1단계 · 개념 매스", p:t=>1-0.45*t, tw:0, col:0x9db4c4,
      text:"단순한 사각 테이퍼 매스로 높이와 기준 외곽만 잡은 가상 단계입니다. 곡선과 회전은 아직 없습니다." },
    { name:"2단계 · 구간별 후퇴", p:t=> t<0.33?1 : t<0.66?0.8 : 0.8-0.35*(t-0.66)/0.34, tw:0.12, col:0xa9b8a0,
      text:"층이 올라갈수록 외곽이 구간별로 줄어드는 단계입니다. 후퇴 위치와 비율은 설명을 위해 추정한 값입니다." },
    { name:"3단계 · 곡선 외곽 검토", p:t=>0.4+0.6*Math.pow(1-t,1.6), tw:0.25, col:0xc7b48a,
      text:"위로 갈수록 좁아지는 곡선 외곽을 검토한 단계입니다. 곡선 형태는 도자기의 선을 참고했다는 공개 설명을 개념적으로 옮긴 것입니다." },
    { name:"완성 모델 (개념 재구성)", p:t=>0.38+0.62*Math.pow(1-t,1.6)+0.03*Math.sin(Math.PI*t), tw:0.32, col:0xe8e2d4,
      text:"555 m 높이와 123층을 기준으로 한 완성 형상의 개념 재구성입니다. 실제 평면 형태·비틀림 양·부재 위치는 공개 자료에 없어 추정입니다." }
  ];

  const VIEWS = {
    site: { title:"배치도 (개념)", info:"대지 경계와 건물 위치를 보여 줍니다. 대지 크기(약 400 m × 300 m)와 방위는 추정입니다. 건물 발자국은 기준 외곽(약 120 m, 추정)입니다." },
    plan: { title:"기준층 평면도", info:"선택한 층 높이에서의 외곽 평면입니다. 점선은 1층 외곽, 실선은 선택한 층의 외곽입니다. 회전각은 3D 모델의 비틀림 값과 같은 규칙을 씁니다(추정)." },
    elev: { title:"입면도", info:"정면에서 본 외곽 윤곽입니다. 높이 555 m와 123층은 공개 값이고, 폭 변화 곡선은 추정입니다. 빨간 선은 선택한 층입니다." },
    sect: { title:"단면도", info:"중심을 자른 개념 단면입니다. 가운데 어두운 띠는 코어를 가정한 위치이며 실제 코어 치수가 아닙니다. 바닥판 선은 10개 층 간격으로 그렸습니다." }
  };

  const state = { view:"elev", stage:3, floor:62 };
  const reduceMotion = !!(window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches);

  /* ---------- 유틸 ---------- */
  const $ = s=>document.querySelector(s);
  const cv = $("#view2d");
  const ctx = cv.getContext("2d");
  const floorInput = $("#floor");
  const floorTag = (f)=>{
    const hm = Math.round(f/FLOORS*H_M);
    return `${f}층 · 약 ${hm} m (층 높이 균등 가정, 추정)`;
  };
  const stageP = (t)=>STAGES[state.stage].p(t);
  const curT = ()=> state.floor/FLOORS;

  /* ---------- 2D 도면 ---------- */
  function resize2D(){
    const dpr = Math.min(window.devicePixelRatio||1, 2);
    const w = cv.clientWidth, h = cv.clientHeight;
    cv.width = Math.max(1, Math.round(w*dpr));
    cv.height = Math.max(1, Math.round(h*dpr));
    ctx.setTransform(dpr,0,0,dpr,0,0);
    draw2D();
  }

  function fit(wm,hm,pad=46){
    const W=cv.clientWidth, Hh=cv.clientHeight;
    const s=Math.min((W-2*pad)/wm,(Hh-2*pad)/hm);
    return {s, cx:W/2, base:Hh-pad, W, Hh, pad};
  }

  function scaleBar(m, px, py, s){
    ctx.strokeStyle="#eef2f6"; ctx.fillStyle="#eef2f6"; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(px,py); ctx.lineTo(px+m*s,py); ctx.stroke();
    ctx.font="13px system-ui"; ctx.fillText(`${m} m (추정 축척)`, px, py-6);
  }

  function silhouette(f, mode){
    const {s,cx,base}=f;
    const N=160;
    const left=[], right=[];
    for(let i=0;i<=N;i++){
      const t=i/N;
      const half=BASE_M*stageP(t)/2*s;
      const y=base - t*H_M*s;
      left.push([cx-half,y]); right.push([cx+half,y]);
    }
    ctx.beginPath();
    left.forEach((p,i)=> i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]));
    right.slice().reverse().forEach(p=>ctx.lineTo(p[0],p[1]));
    ctx.closePath();
    ctx.fillStyle="rgba(120,160,190,0.22)"; ctx.fill();
    ctx.strokeStyle="#9ec5e6"; ctx.lineWidth=2; ctx.stroke();
    if(mode==="sect"){
      ctx.fillStyle="rgba(40,60,80,0.8)";
      for(let i=0;i<N;i++){
        const t0=i/N, t1=(i+1)/N;
        const hw=BASE_M*stageP((t0+t1)/2)*0.12*s;
        const y0=base-t0*H_M*s, y1=base-t1*H_M*s;
        ctx.fillRect(cx-hw,y1,2*hw,y0-y1+0.5);
      }
      ctx.strokeStyle="rgba(255,255,255,0.18)"; ctx.lineWidth=1;
      for(let fl=10; fl<=FLOORS; fl+=10){
        const t=fl/FLOORS, y=base-t*H_M*s, hw=BASE_M*stageP(t)/2*s;
        ctx.beginPath(); ctx.moveTo(cx-hw,y); ctx.lineTo(cx+hw,y); ctx.stroke();
      }
    }
  }

  function markerLine(f){
    const {s,cx,base}=f;
    const t=curT(), y=base-t*H_M*s, hw=BASE_M*stageP(t)/2*s;
    ctx.strokeStyle="#ff6b6b"; ctx.lineWidth=2; ctx.setLineDash([6,4]);
    ctx.beginPath(); ctx.moveTo(cx-hw-30,y); ctx.lineTo(cx+hw+30,y); ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle="#ff6b6b"; ctx.font="13px system-ui";
    ctx.fillText(`${state.floor}층`, cx+hw+34, y+4);
  }

  function heightAxis(f){
    const {s,cx,base}=f;
    ctx.strokeStyle="#5b6b7e"; ctx.fillStyle="#b7c3d0"; ctx.font="12px system-ui"; ctx.lineWidth=1;
    const x=cx-BASE_M*s/2-36;
    ctx.beginPath(); ctx.moveTo(x,base); ctx.lineTo(x,base-H_M*s); ctx.stroke();
    [0,123,246,369,492,555].forEach(m=>{
      const y=base-m*s;
      ctx.beginPath(); ctx.moveTo(x-5,y); ctx.lineTo(x+5,y); ctx.stroke();
      ctx.fillText(`${m} m`, x-44, y+4);
    });
    ctx.fillStyle="#ffd166";
    ctx.fillText("높이 555 m (공개 값)", cx+8, base-H_M*s-8);
  }

  function draw2D(){
    const W=cv.clientWidth, Hh=cv.clientHeight;
    if(!W||!Hh) return;
    ctx.clearRect(0,0,W,Hh);
    const v=state.view;
    if(v==="site"){
      const f=fit(SITE_W,SITE_H);
      const s=f.s;
      const top=(Hh-SITE_H*s)/2;
      ctx.strokeStyle="#8fa3b8"; ctx.setLineDash([8,6]); ctx.lineWidth=2;
      ctx.strokeRect(f.cx-SITE_W*s/2, top, SITE_W*s, SITE_H*s); ctx.setLineDash([]);
      ctx.fillStyle="#b7c3d0"; ctx.font="13px system-ui";
      ctx.fillText("대지 경계 (추정)", f.cx-SITE_W*s/2+6, top+16);
      const bs=BASE_M*s;
      ctx.fillStyle="rgba(158,197,230,0.35)"; ctx.strokeStyle="#9ec5e6"; ctx.lineWidth=2;
      ctx.fillRect(f.cx-bs/2, Hh/2-bs/2, bs, bs); ctx.strokeRect(f.cx-bs/2, Hh/2-bs/2, bs, bs);
      ctx.fillStyle="#eef2f6"; ctx.fillText("건물 기준 외곽 약 120 m (추정)", f.cx-bs/2, Hh/2+bs/2+18);
      ctx.fillStyle="#ffd166"; ctx.fillText("N ↑", f.cx+SITE_W*s/2-40, top+18);
      scaleBar(100, f.cx-SITE_W*s/2+6, top+SITE_H*s-12, s);
    }
    if(v==="plan"){
      const f=fit(220,220);
      const s=f.s, cx=f.cx, cy=Hh/2;
      ctx.strokeStyle="#1f2a36"; ctx.lineWidth=1;
      for(let i=-5;i<=5;i++){ ctx.beginPath(); ctx.moveTo(cx+i*20*s,cy-110*s); ctx.lineTo(cx+i*20*s,cy+110*s); ctx.stroke(); }
      for(let i=-5;i<=5;i++){ ctx.beginPath(); ctx.moveTo(cx-110*s,cy+i*20*s); ctx.lineTo(cx+110*s,cy+i*20*s); ctx.stroke(); }
      const b=BASE_M*s;
      ctx.setLineDash([6,5]); ctx.strokeStyle="#7d8fa3"; ctx.lineWidth=1.5;
      ctx.strokeRect(cx-b/2,cy-b/2,b,b); ctx.setLineDash([]);
      const t=curT(), side=BASE_M*stageP(t)*s, ang=STAGES[state.stage].tw*t;
      ctx.save(); ctx.translate(cx,cy); ctx.rotate(ang);
      ctx.fillStyle="rgba(255,209,102,0.18)"; ctx.strokeStyle="#ffd166"; ctx.lineWidth=2.5;
      ctx.fillRect(-side/2,-side/2,side,side); ctx.strokeRect(-side/2,-side/2,side,side);
      ctx.restore();
      ctx.fillStyle="#eef2f6"; ctx.font="13px system-ui";
      ctx.fillText(`선택 층 한 변 약 ${Math.round(BASE_M*stageP(t))} m (추정)`, cx-b/2, cy+b/2+22);
      ctx.fillText(`회전각 약 ${(ang*180/Math.PI).toFixed(1)}° (추정)`, cx-b/2, cy+b/2+40);
      ctx.fillStyle="#7d8fa3"; ctx.fillText("점선: 1층 외곽", cx-b/2, cy-b/2-10);
      scaleBar(100, cx-b/2, cy+b/2+64, s);
    }
    if(v==="elev"){
      const f=fit(160,H_M);
      silhouette(f,"elev"); heightAxis(f); markerLine(f);
      scaleBar(100, f.cx-BASE_M*f.s/2, f.base+20, f.s);
    }
    if(v==="sect"){
      const f=fit(160,H_M);
      silhouette(f,"sect"); heightAxis(f); markerLine(f);
      scaleBar(100, f.cx-BASE_M*f.s/2, f.base+20, f.s);
    }
    $("#drawInfo").innerHTML =
      `<b>${VIEWS[v].title}</b><br>${VIEWS[v].info}<br><span class="est">대응:</span> 선택 설계 단계 「${STAGES[state.stage].name}」, 선택 층 ${state.floor}층 (3D 마커와 동기화).`;
    $("#floorLabel").textContent = `${state.floor}층`;
    $("#floorInfo").textContent = floorTag(state.floor);
    updateView3DMarker();
  }

  /* ---------- 3D ---------- */
  let renderer=null, scene=null, camera=null, controls=null, model=null, marker=null, markerRing=null;
  let autoRotate = false;
  let threeOK = false;

  function init3D(){
    const host=$("#view3d");
    if(typeof THREE==="undefined" || !THREE || !THREE.WebGLRenderer){
      $("#fallback").style.display="flex";
      document.getElementById("btnRotate").disabled=true;
      document.getElementById("btnZoomIn").disabled=true;
      document.getElementById("btnZoomOut").disabled=true;
      document.getElementById("btnReset").disabled=true;
      return false;
    }
    try{
      renderer=new THREE.WebGLRenderer({antialias:true});
    }catch(e){
      $("#fallback").style.display="flex";
      return false;
    }
    renderer.setPixelRatio(Math.min(window.devicePixelRatio||1,2));
    renderer.setClearColor(0x0c1014,1);
    const canvas=renderer.domElement;
    canvas.setAttribute("aria-hidden","true");
    canvas.style.display="block";
    host.insertBefore(canvas, $("#fallback"));
    scene=new THREE.Scene();
    camera=new THREE.PerspectiveCamera(40,1,0.5,2000);
    camera.position.set(90,95,140);
    scene.add(new THREE.AmbientLight(0xffffff,0.75));
    const dl=new THREE.DirectionalLight(0xffffff,0.9); dl.position.set(80,160,60); scene.add(dl);
    scene.add(new THREE.GridHelper(200,40,0x2e3a48,0x1e2731));
    if(THREE.OrbitControls){
      controls=new THREE.OrbitControls(camera,canvas);
      controls.enableDamping=!reduceMotion;
      controls.target.set(0,H_U/2,0);
      controls.minDistance=40; controls.maxDistance=400;
      controls.autoRotate=false;
      controls.update();
    }
    marker=new THREE.Mesh(new THREE.SphereGeometry(0.9,16,12), new THREE.MeshBasicMaterial({color:0xff6b6b}));
    scene.add(marker);
    markerRing=new THREE.LineLoop(new THREE.BufferGeometry(), new THREE.LineBasicMaterial({color:0xff6b6b}));
    scene.add(markerRing);
    threeOK=true;
    buildModel();
    resize3D();
    return true;
  }

  function buildModel(){
    const st=STAGES[state.stage];
    if(scene){
      if(model){
        scene.remove(model);
        model.traverse(o=>{ if(o.geometry) o.geometry.dispose(); if(o.material) o.material.dispose(); });
      }
      model=new THREE.Group();
      const slabH=H_U/SLABS;
      const matFill=new THREE.MeshStandardMaterial({color:st.col, roughness:0.6, metalness:0.05, transparent:true, opacity:0.92});
      const matLine=new THREE.LineBasicMaterial({color:0x2a3340});
      for(let i=0;i<SLABS;i++){
        const t=(i+0.5)/SLABS;
        const w=BASE_U*st.p(t);
        const geo=new THREE.BoxGeometry(w,slabH*0.97,w);
        const mesh=new THREE.Mesh(geo,matFill);
        mesh.position.set(0,(i+0.5)*slabH,0);
        mesh.rotation.y=st.tw*t;
        model.add(mesh);
        const edgeGeo=new THREE.EdgesGeometry(geo);
        const edges=new THREE.LineSegments(edgeGeo, matLine);
        edges.position.copy(mesh.position); edges.rotation.copy(mesh.rotation);
        model.add(edges);
      }
      const base=new THREE.Mesh(new THREE.PlaneGeometry(BASE_U*2.2,BASE_U*2.2), new THREE.MeshBasicMaterial({color:0x1a2530}));
      base.rotation.x=-Math.PI/2; base.position.y=0.05; model.add(base);
      scene.add(model);
      updateView3DMarker();
    }
    $("#stageInfo").innerHTML =
      `<b>${st.name}</b><br>${st.text}<br><span class="note">형상 규칙: 높이 555 m(공개), 기준 외곽 약 120 m(추정), 비틀림 ${st.tw} rad(추정), 높이 방향 ${SLABS}개 구간으로 나눈 개념 모델입니다.</span>`;
    document.querySelectorAll("[data-stage]").forEach(b=>b.setAttribute("aria-pressed", String(+b.dataset.stage===state.stage)));
  }

  function updateView3DMarker(){
    if(!marker || !markerRing) return;
    const t=curT(), y=t*H_U;
    marker.position.set(0,y,0);
    const st=STAGES[state.stage];
    const side=BASE_U*st.p(t)/2, ang=st.tw*t;
    const pts=[];
    for(let k=0;k<4;k++){
      const a=ang+Math.PI/4+k*Math.PI/2;
      const r=side*Math.SQRT2;
      pts.push(new THREE.Vector3(Math.cos(a)*r,y,Math.sin(a)*r));
    }
    pts.push(pts[0].clone());
    markerRing.geometry.dispose();
    markerRing.geometry=new THREE.BufferGeometry().setFromPoints(pts);
  }

  function resize3D(){
    if(!renderer) return;
    const host=$("#view3d");
    const w=Math.max(1,host.clientWidth), h=Math.max(1,host.clientHeight);
    renderer.setSize(w,h,false);
    const c=renderer.domElement; c.style.width=w+"px"; c.style.height=h+"px";
    camera.aspect=w/h; camera.updateProjectionMatrix();
  }

  function animate(){
    requestAnimationFrame(animate);
    if(!renderer) return;
    if(controls){
      controls.autoRotate = autoRotate && !reduceMotion;
      controls.autoRotateSpeed = 1.2;
      controls.update();
    }
    renderer.render(scene,camera);
  }

  function orbitBy(angle){
    if(!controls) return;
    const up=new THREE.Vector3(0,1,0);
    const off=camera.position.clone().sub(controls.target).applyAxisAngle(up,angle);
    camera.position.copy(controls.target).add(off); controls.update();
  }
  function zoomBy(f){
    if(!controls) return;
    const off=camera.position.clone().sub(controls.target).multiplyScalar(f);
    const len=off.length();
    if(len<40||len>400) return;
    camera.position.copy(controls.target).add(off); controls.update();
  }
  function resetView(){
    if(!camera||!controls) return;
    camera.position.set(90,95,140);
    controls.target.set(0,H_U/2,0); controls.update();
  }

  /* ---------- 이벤트 ---------- */
  function setFloor(f){
    state.floor=Math.max(1,Math.min(FLOORS,f));
    floorInput.value=String(state.floor);
    draw2D();
  }

  document.querySelectorAll("[data-view]").forEach(b=>b.addEventListener("click",()=>{
    state.view=b.dataset.view;
    document.querySelectorAll("[data-view]").forEach(x=>x.setAttribute("aria-pressed",String(x===b)));
    draw2D();
  }));

  document.querySelectorAll("[data-stage]").forEach(b=>b.addEventListener("click",()=>{
    state.stage=+b.dataset.stage;
    buildModel();
    draw2D();
  }));

  floorInput.addEventListener("input",()=>{
    state.floor=+floorInput.value;
    draw2D();
  });

  cv.addEventListener("keydown",e=>{
    const k=e.key;
    if(k==="ArrowRight"||k==="ArrowUp"){ setFloor(state.floor+1); e.preventDefault(); }
    else if(k==="ArrowLeft"||k==="ArrowDown"){ setFloor(state.floor-1); e.preventDefault(); }
  });

  $("#view3d").addEventListener("keydown",e=>{
    const k=e.key;
    if(k==="ArrowLeft"){ orbitBy(0.08); e.preventDefault(); }
    else if(k==="ArrowRight"){ orbitBy(-0.08); e.preventDefault(); }
    else if(k==="+"||k==="="){ zoomBy(0.9); e.preventDefault(); }
    else if(k==="-"||k==="_"){ zoomBy(1.1); e.preventDefault(); }
    else if(k==="0"){ resetView(); e.preventDefault(); }
    else if(k===" "){ autoRotate=!autoRotate; syncRotateBtn(); e.preventDefault(); }
  });

  function syncRotateBtn(){
    const b=$("#btnRotate");
    b.setAttribute("aria-pressed",String(autoRotate));
    b.textContent = autoRotate ? "자동 회전 끄기" : "자동 회전 켜기";
  }
  $("#btnRotate").addEventListener("click",()=>{ autoRotate=!autoRotate; syncRotateBtn(); });
  $("#btnZoomIn").addEventListener("click",()=>zoomBy(0.9));
  $("#btnZoomOut").addEventListener("click",()=>zoomBy(1.1));
  $("#btnReset").addEventListener("click",resetView);

  window.addEventListener("resize",()=>{ resize2D(); resize3D(); });

  /* ---------- 시작 ---------- */
  floorInput.value=String(state.floor);
  syncRotateBtn();
  resize2D();
  const ok3d = init3D();
  if(ok3d){ animate(); }
  if(!ok3d){ buildModel(); }
  draw2D();
})();
