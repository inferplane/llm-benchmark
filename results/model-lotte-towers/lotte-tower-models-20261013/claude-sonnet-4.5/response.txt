  const h = canvas.height = canvas.offsetHeight * 2;

  ctx.fillRect(0, 0, w, h);

  ctx.fill();
  
  // Tower footprint
  ctx.fillStyle = '#424242';
  ctx.fillRect(w/2/2 - 30, h/2/2 - 30, 60, 60);
  
  // Surrounding buildings
  ctx.fillStyle = '#bdbdbd';
  ctx.fillRect(w/1.5/2, h/3/2, 40, 40);
  ctx.fillRect(w/1.3/2, h/1.5/2, 35, 35);
  
  // Labels
  ctx.fillStyle = '#000';
  ctx.font = '10px sans-serif';
  ctx.fillText('석촌호수', w/4/2 - 20, h/2/2);
  ctx.fillText('롯데월드타워', w/2/2 - 35, h/2/2 + 50);

  ctx.beginPath();
  ctx.moveTo(20, h/2 - 20);

}

function drawFloorPlan() {
  const canvas = document.getElementById('floorPlan');
  const ctx = canvas.getContext('2d');
  const w = canvas.width = canvas.offsetWidth * 2;
  const h = canvas.height = canvas.offsetHeight * 2;
  ctx.scale(2, 2);
  
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, w, h);
  
  const cx = w/2/2, cy = h/2/2;
  const radius = 80;
  
  // Outer curved shape
  ctx.strokeStyle = '#000';
  ctx.lineWidth = 3;
  ctx.beginPath();
  for (let i = 0; i < 8; i++) {
    const angle = (i / 8) * Math.PI * 2;
    const r = radius + Math.sin(i * 3) * 10;
    const x = cx + Math.cos(angle) * r;
    const y = cy + Math.sin(angle) * r;
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  }
  ctx.closePath();
  ctx.stroke();
  
  // Core
  ctx.fillStyle = '#9e9e9e';
  ctx.fillRect(cx - 25, cy - 25, 50, 50);
  
  // Elevators
  ctx.fillStyle = '#fff';
  for (let i = 0; i < 3; i++) {
    ctx.fillRect(cx - 20 + i * 15, cy - 20, 10, 10);
  }
  
  // Columns
  ctx.fillStyle = '#000';
  const colPositions = [
    [cx - 50, cy - 50], [cx + 50, cy - 50],
    [cx - 50, cy + 50], [cx + 50, cy + 50]
  ];
  colPositions.forEach(([x, y]) => {
    ctx.beginPath();
    ctx.arc(x, y, 4, 0, Math.PI * 2);
    ctx.fill();
  });
  
  // Labels
  ctx.font = '9px sans-serif';
  ctx.fillText('코어', cx - 12, cy + 3);
  ctx.fillText('임대 공간', cx + 30, cy - 40);
  
  // Dimensions
  ctx.strokeStyle = '#666';
  ctx.lineWidth = 1;
  ctx.setLineDash([3, 3]);
  ctx.beginPath();
  ctx.moveTo(cx - radius, h/2 - 10);
  ctx.lineTo(cx + radius, h/2 - 10);
  ctx.stroke();
  ctx.setLineDash([]);
  ctx.fillText('~50m (추정)', cx - 25, h/2 - 15);
}

function drawElevation() {
  const canvas = document.getElementById('elevation');
  const ctx = canvas.getContext('2d');
  const w = canvas.width = canvas.offsetWidth * 2;
  const h = canvas.height = canvas.offsetHeight * 2;
  ctx.scale(2, 2);
  
  ctx.fillStyle = '#e3f2fd';
  ctx.fillRect(0, 0, w, h);
  
  const baseW = 100;
  const topW = 40;
  const buildingH = h/2 - 40;
  const cx = w/2/2;
  
  // Tower silhouette with taper
  const gradient = ctx.createLinearGradient(0, h/2 - 20, 0, 20);
  gradient.addColorStop(0, '#4a90e2');
  gradient.addColorStop(0.4, '#7b68ee');
  gradient.addColorStop(0.7, '#ff6b6b');
  gradient.addColorStop(1, '#ffd93d');
  
  ctx.fillStyle = gradient;
  ctx.beginPath();
  ctx.moveTo(cx - baseW/2, h/2 - 20);
  ctx.lineTo(cx - topW/2, 20);
  ctx.lineTo(cx + topW/2, 20);
  ctx.lineTo(cx + baseW/2, h/2 - 20);
  ctx.closePath();
  ctx.fill();
  
  // Floor lines
  ctx.strokeStyle = 'rgba(255,255,255,0.3)';
  ctx.lineWidth = 1;
  for (let i = 0; i < 25; i++) {
    const y = h/2 - 20 - (i * buildingH / 25);
    const leftX = cx - baseW/2 + (baseW - topW) / 2 * (i / 25);
    const rightX = cx + baseW/2 - (baseW - topW) / 2 * (i / 25);
    ctx.beginPath();
    ctx.moveTo(leftX, y);
    ctx.lineTo(rightX, y);
    ctx.stroke();
  }
  
  // Ground
  ctx.fillStyle = '#795548';
  ctx.fillRect(0, h/2 - 20, w/2, 20);
  
  // Height annotation
  ctx.fillStyle = '#000';
  ctx.font = '11px sans-serif';
  ctx.fillText('555m', cx + baseW/2 + 10, h/4);
  ctx.strokeStyle = '#000';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(cx + baseW/2 + 5, 20);
  ctx.lineTo(cx + baseW/2 + 5, h/2 - 20);
  ctx.stroke();
  
  // Level markers
  ctx.font = '8px sans-serif';
  ctx.fillText('123F', cx - baseW/2 - 25, 25);
  ctx.fillText('1F', cx - baseW/2 - 20, h/2 - 20);
}

function drawSection() {
  const canvas = document.getElementById('section');
  const ctx = canvas.getContext('2d');
  const w = canvas.width = canvas.offsetWidth * 2;
  const h = canvas.height = canvas.offsetHeight * 2;
  ctx.scale(2, 2);
  
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, w, h);
  
  const baseW = 90;
  const topW = 35;
  const buildingH = h/2 - 50;
  const cx = w/2/2;
  const baseY = h/2 - 30;
  
  // Underground
  ctx.fillStyle = '#d7ccc8';
  ctx.fillRect(cx - baseW/2, baseY, baseW, 20);
  
  // Building sections
  const sections = [
    { height: 0.15, color: '#4a90e2', label: '상업' },
    { height: 0.35, color: '#7b68ee', label: '오피스' },
    { height: 0.35, color: '#ff6b6b', label: '호텔/주거' },
    { height: 0.15, color: '#ffd93d', label: '전망대' }
  ];
  
  let currentY = baseY;
  sections.forEach((section, idx) => {
    const sectionH = buildingH * section.height;
    const progress = (buildingH - (currentY - (baseY - buildingH))) / buildingH;
    const currentW = baseW - (baseW - topW) * progress;
    const nextProgress = (buildingH - (currentY - sectionH - (baseY - buildingH))) / buildingH;
    const nextW = baseW - (baseW - topW) * nextProgress;
    
    ctx.fillStyle = section.color;
    ctx.beginPath();
    ctx.moveTo(cx - currentW/2, currentY);
    ctx.lineTo(cx - nextW/2, currentY - sectionH);
    ctx.lineTo(cx + nextW/2, currentY - sectionH);
    ctx.lineTo(cx + currentW/2, currentY);
    ctx.closePath();
    ctx.fill();
    
    ctx.fillStyle = '#000';
    ctx.font = '9px sans-serif';
    ctx.fillText(section.label, cx + currentW/2 + 5, currentY - sectionH/2);
    
    currentY -= sectionH;
  });
  
  // Core
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.fillRect(cx - 8, baseY - buildingH, 16, buildingH);
  
  // Ground line
  ctx.strokeStyle = '#000';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(cx - baseW/2 - 20, baseY);
  ctx.lineTo(cx + baseW/2 + 20, baseY);
  ctx.stroke();
  
  ctx.font = '8px sans-serif';
  ctx.fillStyle = '#000';
  ctx.fillText('지상', cx + baseW/2 + 25, baseY);
  ctx.fillText('지하', cx + baseW/2 + 25, baseY + 15);
}

function createTower(stage) {
  if (currentTower) {
    scene.remove(currentTower);
  }
  
  const towerGroup = new THREE.Group();
  
  const totalHeight = 555;
  const segments = 40;
  const segmentHeight = totalHeight / segments;
  
  for (let i = 0; i < segments; i++) {
    const progress = i / segments;
    const y = i * segmentHeight;
    
    let baseRadius, topRadius;
    
    switch(stage) {
      case 0: // Simple box
        baseRadius = topRadius = 50;
        break;
      case 1: // Basic taper
        baseRadius = 50 - progress * 10;
        topRadius = 50 - (progress + 0.025) * 10;
        break;
      case 2: // More taper
        baseRadius = 50 - progress * 20;
        topRadius = 50 - (progress + 0.025) * 20;
        break;
      case 3: // Curved taper
        const curve1 = Math.pow(1 - progress, 0.7);
        baseRadius = 30 + curve1 * 20;
        topRadius = 30 + Math.pow(1 - (progress + 0.025), 0.7) * 20;
        break;
      case 4: // Final form with ceramic curve
        const curve2 = Math.pow(1 - progress, 0.6);
        const wave = Math.sin(progress * Math.PI * 2) * 2;
        baseRadius = 28 + curve2 * 22 + wave;
        topRadius = 28 + Math.pow(1 - (progress + 0.025), 0.6) * 22 + Math.sin((progress + 0.025) * Math.PI * 2) * 2;
        break;
    }
    
    // Color based on height
    let color;
    if (progress < 0.25) color = 0x4a90e2;
    else if (progress < 0.5) color = 0x7b68ee;
    else if (progress < 0.85) color = 0xff6b6b;
    else color = 0xffd93d;
    
    const geometry = new THREE.CylinderGeometry(topRadius, baseRadius, segmentHeight, stage === 0 ? 4 : 16);
    const material = new THREE.MeshStandardMaterial({
      color: color,
      metalness: 0.3,
      roughness: 0.4,
      transparent: true,
      opacity: 0.9
    });
    
    const segment = new THREE.Mesh(geometry, material);
    segment.position.y = y + segmentHeight / 2;
    
    // Edge lines for detail
    const edges = new THREE.EdgesGeometry(geometry);
    const lineMaterial = new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.2 });
    const wireframe = new THREE.LineSegments(edges, lineMaterial);
    segment.add(wireframe);
    
    towerGroup.add(segment);
  }
  
  currentTower = towerGroup;
  scene.add(towerGroup);
}
