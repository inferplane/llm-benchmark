  function initThree(){
    renderer = new THREE.WebGLRenderer({canvas:threeCanvas, antialias:true, alpha:false});
    renderer.setPixelRatio(Math.min(window.devicePixelRatio,2));
    scene = new THREE.Scene();
    scene.background = new THREE.Color(0x06080c);

    camera.position.set(60,70,120);

    const amb = new THREE.AmbientLight(0xffffff, 0.55);
    scene.add(amb);
    const dir = new THREE.DirectionalLight(0xfff2d8, 1.0);
    dir.position.set(80,120,60);
    scene.add(dir);
    const dir2 = new THREE.DirectionalLight(0x7ec8e3, 0.4);
    dir2.position.set(-80,40,-60);

    scene.add(grid);

    controls = new THREE.OrbitControls(camera, renderer.domElement);

    controls.dampingFactor = 0.08;
    controls.target.set(0,30,0);

    scene.add(towerGroup);

    resizeThree();
    buildStage(0);
    animate();
  }

  function resizeThree(){

    threeCanvas.width = w; threeCanvas.height = h;
    renderer.setSize(w,h,false);
    camera.aspect = w/h;
    camera.updateProjectionMatrix();
  }

  // 8각 단면 모양 생성(평면도와 대응)
  function octShape(radius, squashY){
    const pts=[];
    for(let i=0;i<8;i++){
      const a = Math.PI/8 + i*Math.PI/4;
      pts.push(new THREE.Vector2(Math.cos(a)*radius, Math.sin(a)*radius*squashY));
    }
    return pts;
  }

  function buildTaperedTower(segments, taperPow, roughness, color){
    const group = new THREE.Group();
    const totalH = 90;
    const baseR = 26, topR = 6;
    const layerH = totalH/segments;
    for(let i=0;i<segments;i++){
      const t0=i/segments, t1=(i+1)/segments;
      const r0 = baseR - (baseR-topR)*Math.pow(t0, taperPow);
      const r1 = baseR - (baseR-topR)*Math.pow(t1, taperPow);
      const geom = new THREE.CylinderGeometry(r1*0.9, r0*0.9, layerH, 8, 1, false);
      const mat = new THREE.MeshStandardMaterial({
        color: color, metalness:0.3, roughness: roughness,
        flatShading: true
      });
      const mesh = new THREE.Mesh(geom, mat);
      mesh.position.y = i*layerH + layerH/2;
      mesh.rotation.y = Math.PI/8;
      group.add(mesh);
    }
    return group;
  }

  function buildCurvedTower(color, roughness, segments){
    // 더 부드러운 곡면 근사: Lathe로 도자기 곡선 느낌
    const pts = [];
    const N = 24;
    for(let i=0;i<=N;i++){
      const t = i/N;
      const y = t*90;
      const r = 6 + (26-6)*Math.pow(1-t, 1.3) * (1 - 0.08*Math.sin(t*Math.PI*1.5));
      pts.push(new THREE.Vector2(Math.max(r,0.5), y));
    }
    const geom = new THREE.LatheGeometry(pts, 16);
    const mat = new THREE.MeshStandardMaterial({color, metalness:0.35, roughness, flatShading:false});
    const mesh = new THREE.Mesh(geom, mat);
    const group = new THREE.Group();
    group.add(mesh);
    return group;
  }

  window.addEventListener('keydown', (e)=>{

    switch(e.key){
      case 'ArrowLeft': controls.setAzimuthalAngle ? null : null; rotateCam(step); break;
      case 'ArrowRight': rotateCam(-step); break;
      case 'ArrowUp': pitchCam(step); break;
      case 'ArrowDown': pitchCam(-step); break;
    }
    if(e.shiftKey){
      if(e.key==='ArrowUp') dollyCam(-5);
      if(e.key==='ArrowDown') dollyCam(5);
    }
  });

  function rotateCam(a){
    const offset = new THREE.Vector3().copy(camera.position).sub(controls.target);
    const spherical = new THREE.Spherical().setFromVector3(offset);
    spherical.theta += a;
    offset.setFromSpherical(spherical);
    camera.position.copy(controls.target).add(offset);
    controls.update();
  }
  function pitchCam(a){
    const offset = new THREE.Vector3().copy(camera.position).sub(controls.target);
    const spherical = new THREE.Spherical().setFromVector3(offset);
    spherical.phi = Math.min(Math.max(spherical.phi - a, 0.15), Math.PI-0.15);
    offset.setFromSpherical(spherical);
    camera.position.copy(controls.target).add(offset);
    controls.update();
  }
  function dollyCam(d){
    const dir = new THREE.Vector3().copy(camera.position).sub(controls.target).normalize();
    camera.position.addScaledVector(dir, d);
    controls.update();
  }
