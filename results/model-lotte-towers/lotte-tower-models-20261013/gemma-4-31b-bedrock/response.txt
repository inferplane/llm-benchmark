    <script>
        (function() {
            if (!window.THREE) {
                const script = document.createElement('script');
                script.src = "../runtime/three-runtime.js";
                script.onload = init;
                document.head.appendChild(script);
            } else {
                init();
            }

            let scene, camera, renderer, controls, towerGroup;
            const container = document.getElementById('canvas-container');

            function init() {
                scene = new THREE.Scene();
                scene.background = new THREE.Color(0x0a0a0c);
                scene.fog = new THREE.FogExp2(0x0a0a0c, 0.0015);

                camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 1, 10000);
                camera.position.set(300, 300, 600);

                renderer = new THREE.WebGLRenderer({ antialias: true });
                renderer.setSize(window.innerWidth, window.innerHeight);
                renderer.setPixelRatio(window.devicePixelRatio);
                container.appendChild(renderer.domElement);

                controls = new THREE.OrbitControls(camera, renderer.domElement);
                controls.enableDamping = true;
                controls.target.set(0, 200, 0);

                const ambientLight = new THREE.AmbientLight(0x404040, 2);
                scene.add(ambientLight);

                const sunLight = new THREE.DirectionalLight(0xffffff, 1.5);
                sunLight.position.set(200, 500, 200);
                scene.add(sunLight);

                const blueLight = new THREE.PointLight(0x00d4ff, 1);
                blueLight.position.set(-200, 200, -200);
                scene.add(blueLight);

                towerGroup = new THREE.Group();
                scene.add(towerGroup);

                createGround();
                setDesignStep(4); // Default to final
                
                window.addEventListener('resize', onWindowResize);
                setupInputs();
                animate();
            }

            function createGround() {
                const grid = new THREE.GridHelper(2000, 40, 0x333333, 0x222222);
                scene.add(grid);
                const planeGeo = new THREE.PlaneGeometry(2000, 2000);
                const planeMat = new THREE.MeshStandardMaterial({ color: 0x08080a });
                const plane = new THREE.Mesh(planeGeo, planeMat);
                plane.rotation.x = -Math.PI / 2;
                scene.add(plane);
            }

            function setDesignStep(step) {
                towerGroup.clear();
                const material = new THREE.MeshStandardMaterial({
                    color: 0xcccccc,
                    metalness: 0.8,
                    roughness: 0.2,
                    transparent: true,
                    opacity: 0.8
                });

                const TOTAL_HEIGHT = 555;

                if (step === 1) { // Massing
                    const geo = new THREE.BoxGeometry(60, TOTAL_HEIGHT, 60);
                    const mesh = new THREE.Mesh(geo, material);
                    mesh.position.y = TOTAL_HEIGHT / 2;
                    towerGroup.add(mesh);
                } else if (step === 2) { // Tapering
                    const geo = new THREE.CylinderGeometry(15, 40, TOTAL_HEIGHT, 32);
                    const mesh = new THREE.Mesh(geo, material);
                    mesh.position.y = TOTAL_HEIGHT / 2;
                    towerGroup.add(mesh);
                } else if (step === 3) { // Segmented/Split
                    for(let i=0; i<3; i++) {
                        const h = TOTAL_HEIGHT / 3;
                        const geo = new THREE.CylinderGeometry(15 + (25-15)*(2-i/3), 15 + (25-15)*(3-i/3), h, 32);
                        const mesh = new THREE.Mesh(geo, material);
                        mesh.position.y = (h/2) + (i * h);
                        towerGroup.add(mesh);
                    }
                } else { // Final
                    // Body
                    const bodyGeo = new THREE.CylinderGeometry(12, 38, 510, 32);
                    const body = new THREE.Mesh(bodyGeo, material);
                    body.position.y = 510 / 2;
                    towerGroup.add(body);

                    // Crown (integrating into total 555m)
                    const crownGeo = new THREE.CylinderGeometry(0, 12, 45, 32);
                    const crown = new THREE.Mesh(crownGeo, material);
                    crown.position.y = 510 + (45/2);
                    towerGroup.add(crown);
                }

                document.querySelectorAll('.controls-grid button').forEach(btn => btn.classList.remove('active'));
                document.getElementById('btn-step' + step).classList.add('active');
            }

            function onWindowResize() {
                camera.aspect = window.innerWidth / window.innerHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(window.innerWidth, window.innerHeight);
            }

            function setupInputs() {
                const resetView = () => {
                    camera.position.set(300, 300, 600);
                    controls.target.set(0, 200, 0);
                    controls.update();
                };

                const rotateCam = (angle) => {
                    const pos = camera.position.clone();
                    const x = pos.x * Math.cos(angle) - pos.z * Math.sin(angle);
                    const z = pos.x * Math.sin(angle) + pos.z * Math.cos(angle);
                    camera.position.set(x, pos.y, z);
                    controls.update();
                };

                document.getElementById('btn-reset').addEventListener('click', resetView);
                document.getElementById('btn-rotate').addEventListener('click', () => rotateCam(Math.PI / 12));
                
                document.getElementById('btn-step1').addEventListener('click', () => setDesignStep(1));
                document.getElementById('btn-step2').addEventListener('click', () => setDesignStep(2));
                document.getElementById('btn-step3').addEventListener('click', () => setDesignStep(3));
                document.getElementById('btn-step4').addEventListener('click', () => setDesignStep(4));

                window.addEventListener('keydown', (e) => {
                    switch(e.key.toLowerCase()) {
                        case 'r': resetView(); break;
                        case '+': camera.position.multiplyScalar(0.9); break;
                        case '-': camera.position.multiplyScalar(1.1); break;
                        case 'arrowleft': rotateCam(-Math.PI/12); break;
                        case 'arrowright': rotateCam(Math.PI/12); break;
                    }
                });
            }

            window.updateView = function(type) {
                switch(type) {
                    case 'site': 
                        camera.position.set(0, 800, 0); 
                        controls.target.set(0, 0, 0); 
                        break;
                    case 'floor': 
                        camera.position.set(0, 100, 0); 
                        controls.target.set(0, 100, 0); 
                        break;
                    case 'ele': 
                        camera.position.set(600, 200, 0); 
                        controls.target.set(0, 200, 0); 
                        break;
                    case 'sec': 
                        camera.position.set(0, 200, 600); 
                        controls.target.set(0, 200, 0); 
                        break;
                }
                controls.update();
            };

            function animate() {
                requestAnimationFrame(animate);
                controls.update();
                renderer.render(scene, camera);
            }
        })();
    </script>
