    <script>
        /**
         * zai.glm-4.7-flash
         * Interactive Lotte World Tower Visualization
         * 
         * Architecture:
         * 1. State Management: Handles the current stage (Site, Plan, Elev, Section, Model).
         * 2. Scene Graph: A single scene with multiple camera positions and object groups.
         * 3. Procedural Generation: Generates the tower geometry based on the "taper" logic.
         * 4. Interaction: Keyboard and UI event listeners.
         */

        const CONFIG = {
            towerHeight: 555, // meters
            floors: 123,
            baseRadius: 45,   // meters
            topRadius: 12,    // meters
            colors: {
                glass: 0x88ccff,
                structure: 0x333333,
                ground: 0x1a1a1a,
                grid: 0x444444,
                diagram: 0xffaa00
            }
        };

        class LotteApp {
            constructor() {
                this.container = document.getElementById('canvas-container');
                this.scene = null;
                this.camera = null;
                this.renderer = null;
                this.controls = null;
                
                // Objects
                this.towerGroup = new THREE.Group();
                this.groundPlane = null;
                this.gridHelper = null;
                this.raycaster = new THREE.Raycaster();
                this.mouse = new THREE.Vector2();

                // State
                this.currentStage = 'site';
                this.isAnimating = false;
                this.targetCameraPos = new THREE.Vector3();
                this.targetLookAt = new THREE.Vector3(0, 50, 0);

                this.init();
            }

            init() {
                // 1. Scene Setup
                this.scene = new THREE.Scene();
                this.scene.background = new THREE.Color(CONFIG.colors.ground);
                this.scene.fog = new THREE.FogExp2(CONFIG.colors.ground, 0.002);

                // 2. Camera
                this.camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 1, 2000);
                this.camera.position.set(200, 150, 200);

                // 3. Renderer
                this.renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
                this.renderer.setSize(window.innerWidth, window.innerHeight);
                this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
                this.renderer.shadowMap.enabled = true;
                this.container.appendChild(this.renderer.domElement);

                // 4. Controls (OrbitControls from runtime)
                if (typeof THREE !== 'undefined' && THREE.OrbitControls) {
                    this.controls = new THREE.OrbitControls(this.camera, this.renderer.domElement);
                    this.controls.enableDamping = true;
                    this.controls.dampingFactor = 0.05;
                    this.controls.maxPolarAngle = Math.PI / 2 - 0.05; // Don't go below ground
                    this.controls.minDistance = 50;
                    this.controls.maxDistance = 800;
                }

                // 5. Lighting
                const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
                this.scene.add(ambientLight);

                const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
                dirLight.position.set(100, 200, 100);
                dirLight.castShadow = true;
                dirLight.shadow.mapSize.width = 2048;
                dirLight.shadow.mapSize.height = 2048;
                this.scene.add(dirLight);

                const spotLight = new THREE.SpotLight(CONFIG.colors.glass, 1);
                spotLight.position.set(-100, 300, -100);
                spotLight.lookAt(0, 100, 0);
                this.scene.add(spotLight);

                // 6. Build World
                this.createEnvironment();
                this.createTower();
                this.createDesignStages();
                this.createDiagrams();

                // 7. Event Listeners
                window.addEventListener('resize', () => this.onWindowResize(), false);
                document.addEventListener('keydown', (e) => this.onKeyDown(e), false);

                // Remove Loader
                setTimeout(() => {
                    document.getElementById('loader').style.opacity = '0';
                    setTimeout(() => document.getElementById('loader').remove(), 500);
                }, 1000);

                // Start Loop
                this.animate();
            }

            createEnvironment() {
                // Ground
                const geometry = new THREE.PlaneGeometry(2000, 2000);
                const material = new THREE.MeshStandardMaterial({ 
                    color: CONFIG.colors.ground,
                    roughness: 0.8
                });
                this.groundPlane = new THREE.Mesh(geometry, material);
                this.groundPlane.rotation.x = -Math.PI / 2;
                this.groundPlane.receiveShadow = true;
                this.scene.add(this.groundPlane);

                // Grid
                this.gridHelper = new THREE.GridHelper(2000, 100, CONFIG.colors.accent, CONFIG.colors.grid);
                this.gridHelper.position.y = 0.1;
                this.scene.add(this.gridHelper);
            }

            // Procedural Tower Generation
            createTower() {
                // We create a custom geometry to represent the taper
                // Using a CylinderGeometry with radial segments for the "porcelain" look
                const segments = 32;
                const heightSegments = 60; // Detail level
                
                // Create a custom shape for the profile
                const shape = new THREE.Shape();
                const h = CONFIG.towerHeight;
                const rBase = CONFIG.baseRadius;
                const rTop = CONFIG.topRadius;
                
                // Quadratic Bezier curve for smooth taper
                shape.moveTo(-rBase, 0);
                shape.quadraticCurveTo(0, h * 0.3, -rTop, h);
                shape.lineTo(rTop, h);
                shape.quadraticCurveTo(0, h * 0.3, rBase, 0);
                shape.lineTo(-rBase, 0);

                const extrudeSettings = {
                    steps: heightSegments,
                    depth: 1, // Thickness
                    bevelEnabled: true,
                    bevelThickness: 0.5,
                    bevelSize: 0.5,
                    bevelSegments: 2
                };

                const geometry = new THREE.ExtrudeGeometry(shape, extrudeSettings);
                geometry.center(); // Center the geometry vertically

                // Material: Glassy, reflective
                const material = new THREE.MeshPhysicalMaterial({
                    color: CONFIG.colors.glass,
                    metalness: 0.1,
                    roughness: 0.1,
                    transmission: 0.6, // Glass-like
                    thickness: 1.0,
                    clearcoat: 1.0,
                    clearcoatRoughness: 0.1
                });

                this.towerMesh = new THREE.Mesh(geometry, material);
                this.towerMesh.castShadow = true;
                this.towerMesh.receiveShadow = true;
                
                // Add a structural core (visual only)
                const coreGeo = new THREE.CylinderGeometry(5, 5, CONFIG.towerHeight, 8);
                const coreMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.5 });
                const core = new THREE.Mesh(coreGeo, coreMat);
                core.position.y = CONFIG.towerHeight / 2;
                this.towerMesh.add(core);

                this.towerGroup.add(this.towerMesh);
                this.scene.add(this.towerGroup);
            }

            // Create visual markers for design stages
            createDesignStages() {
                // 1. Site Plan Marker (Top View)
                const siteRing = new THREE.RingGeometry(CONFIG.baseRadius + 5, CONFIG.baseRadius + 10, 64);
                const siteMat = new THREE.MeshBasicMaterial({ color: CONFIG.colors.accent, side: THREE.DoubleSide, transparent: true, opacity: 0.5 });
                const siteMesh = new THREE.Mesh(siteRing, siteMat);
                siteMesh.rotation.x = -Math.PI / 2;
                siteMesh.position.y = 0.5;
                siteMesh.userData = { type: 'site' };
                this.towerGroup.add(siteMesh);

                // 2. Floor Plan Marker (Top View)
                const floorRing = new THREE.RingGeometry(CONFIG.topRadius, CONFIG.topRadius + 2, 32);
                const floorMat = new THREE.MeshBasicMaterial({ color: 0xffaa00, side: THREE.DoubleSide, transparent: true, opacity: 0.5 });
                const floorMesh = new THREE.Mesh(floorRing, floorMat);
                floorMesh.rotation.x = -Math.PI / 2;
                floorMesh.position.y = CONFIG.towerHeight;
                floorMesh.userData = { type: 'plan' };
                this.towerGroup.add(floorMesh);

                // 3. Elevation Marker (Front View)
                const elevLine = new THREE.Line(
                    new THREE.BufferGeometry().setFromPoints([
                        new THREE.Vector3(0, 0, 0),
                        new THREE.Vector3(0, CONFIG.towerHeight, 0)
                    ]),
                    new THREE.LineBasicMaterial({ color: 0xffaa00, linewidth: 2 })
                );
                elevLine.userData = { type: 'elevation' };
                this.towerGroup.add(elevLine);

                // 4. Section Marker (Side View)
                const sectionLine = new THREE.Line(
                    new THREE.BufferGeometry().setFromPoints([
                        new THREE.Vector3(-CONFIG.baseRadius, 0, 0),
                        new THREE.Vector3(CONFIG.baseRadius, 0, 0)
                    ]),
                    new THREE.LineBasicMaterial({ color: 0xffaa00, linewidth: 2 })
                );
                sectionLine.userData = { type: 'section' };
                this.towerGroup.add(sectionLine);
            }

            // FIX: Create 4 distinct diagrams with labels and units
            createDiagrams() {
                const diagramGroup = new THREE.Group();
                diagramGroup.visible = false; // Hidden by default, shown in specific stages
                this.diagramGroup = diagramGroup;
                this.scene.add(diagramGroup);

                // Helper to create text sprites
                function createTextSprite(message, x, y, z, color = '#fff') {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    const fontSize = 48; // High res for crisp text
                    canvas.width = 1024;
                    canvas.height = 512;
                    ctx.font = `bold ${fontSize}px "Noto Sans KR", sans-serif`;
                    ctx.fillStyle = color;
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(message, canvas.width / 2, canvas.height / 2);
                    
                    const texture = new THREE.CanvasTexture(canvas);
                    const spriteMaterial = new THREE.SpriteMaterial({ map: texture, transparent: true });
                    const sprite = new THREE.Sprite(spriteMaterial);
                    sprite.position.set(x, y, z);
                    sprite.scale.set(60, 30, 1);
                    return sprite;
                }

                // 1. Site Plan Diagram (Top Down)
                const siteDiagram = new THREE.Group();
                // Draw a circle representing the site
                const siteGeo = new THREE.RingGeometry(40, 50, 64);
                const siteMat = new THREE.MeshBasicMaterial({ color: CONFIG.colors.accent, side: THREE.DoubleSide, transparent: true, opacity: 0.8 });
                const siteMesh = new THREE.Mesh(siteGeo, siteMat);
                siteMesh.rotation.x = -Math.PI / 2;
                siteMesh.position.y = 0.5;
                siteDiagram.add(siteMesh);
                // Label
                siteDiagram.add(createTextSprite("배치도\nSite Plan\nScale: 1:2000", 0, 0, 0, CONFIG.colors.accent));
                siteDiagram.position.set(0, 0, 0);
                diagramGroup.add(siteDiagram);

                // 2. Floor Plan Diagram (Top Down)
                const planDiagram = new THREE.Group();
                // Draw the tapering shape
                const planShape = new THREE.Shape();
                planShape.moveTo(-CONFIG.baseRadius, 0);
                planShape.quadraticCurveTo(0, CONFIG.towerHeight * 0.3, -CONFIG.topRadius, CONFIG.towerHeight);
                planShape.lineTo(CONFIG.topRadius, CONFIG.towerHeight);
                planShape.quadraticCurveTo(0, CONFIG.towerHeight * 0.3, CONFIG.baseRadius, 0);
                planShape.lineTo(-CONFIG.baseRadius, 0);
                
                const planGeo = new THREE.ShapeGeometry(planShape);
                const planMat = new THREE.MeshBasicMaterial({ color: 0xffaa00, side: THREE.DoubleSide, transparent: true, opacity: 0.8 });
                const planMesh = new THREE.Mesh(planGeo, planMat);
                planMesh.position.y = CONFIG.towerHeight / 2;
                planDiagram.add(planMesh);
                // Label
                planDiagram.add(createTextSprite("기준층 평면도\nFloor Plan\nScale: 1:500", 0, CONFIG.towerHeight / 2, 0, 0xffaa00));
                planDiagram.position.set(0, 0, 0);
                diagramGroup.add(planDiagram);

                // 3. Elevation Diagram (Front)
                const elevDiagram = new THREE.Group();
                // Draw the profile curve
                const elevShape = new THREE.Shape();
                elevShape.moveTo(-CONFIG.baseRadius, 0);
                elevShape.quadraticCurveTo(0, CONFIG.towerHeight * 0.3, -CONFIG.topRadius, CONFIG.towerHeight);
                elevShape.lineTo(CONFIG.topRadius, CONFIG.towerHeight);
                elevShape.quadraticCurveTo(0, CONFIG.towerHeight * 0.3, CONFIG.baseRadius, 0);
                elevShape.lineTo(-CONFIG.baseRadius, 0);
                
                const elevGeo = new THREE.ShapeGeometry(elevShape);
                const elevMat = new THREE.MeshBasicMaterial({ color: 0xffaa00, side: THREE.DoubleSide, transparent: true, opacity: 0.8 });
                const elevMesh = new THREE.Mesh(elevGeo, elevMat);
                elevMesh.position.y = CONFIG.towerHeight / 2;
                elevDiagram.add(elevMesh);
                // Label
                elevDiagram.add(createTextSprite("입면도\nElevation\nHeight: 555m", 0, CONFIG.towerHeight / 2, 0, 0xffaa00));
                elevDiagram.position.set(0, 0, 0);
                diagramGroup.add(elevDiagram);

                // 4. Section Diagram (Side)
                const sectionDiagram = new THREE.Group();
                // Draw the profile curve (same as elevation but rotated)
                const sectionShape = new THREE.Shape();
                sectionShape.moveTo(0, -CONFIG.baseRadius);
                sectionShape.quadraticCurveTo(CONFIG.towerHeight * 0.3, 0, CONFIG.towerHeight, -CONFIG.topRadius);
                sectionShape.lineTo(CONFIG.towerHeight, CONFIG.topRadius);
                sectionShape.quadraticCurveTo(CONFIG.towerHeight * 0.3, 0, 0, -CONFIG.baseRadius);
                sectionShape.lineTo(0, -CONFIG.baseRadius);
                
                const sectionGeo = new THREE.ShapeGeometry(sectionShape);
                const sectionMat = new THREE.MeshBasicMaterial({ color: 0xffaa00, side: THREE.DoubleSide, transparent: true, opacity: 0.8 });
                const sectionMesh = new THREE.Mesh(sectionGeo, sectionMat);
                sectionMesh.position.y = CONFIG.towerHeight / 2;
                sectionDiagram.add(sectionMesh);
                // Label
                sectionDiagram.add(createTextSprite("단면도\nSection\nScale: 1:500", 0, CONFIG.towerHeight / 2, 0, 0xffaa00));
                sectionDiagram.position.set(0, 0, 0);
                diagramGroup.add(sectionDiagram);
            }

            setStage(stage) {
                this.currentStage = stage;
                
                // FIX: Use consistent button-ID/stage mapping
                const buttons = document.querySelectorAll('.stage-selector button');
                buttons.forEach(btn => btn.classList.remove('active'));

                if (activeBtn) activeBtn.classList.add('active');

                // Update Text
                const title = document.getElementById('info-title');
                const content = document.getElementById('info-content');
                
                // Reset Diagram Visibility
                if (this.diagramGroup) {
                    this.diagramGroup.children.forEach(d => d.visible = false);
                }

                switch(stage) {
                    case 'site':
                        title.innerText = "배치도";
                        content.innerHTML = `서울 송파구 올림픽공로 300번길 300.<br>지상 123층, 높이 555m.<br>롯데월드타워는 서울의 지표면을 기준으로 배치되었습니다.`;

                        if (this.diagramGroup) this.diagramGroup.children[0].visible = true; // Show Site Diagram
                        break;
                    case 'plan':
                        title.innerText = "기준층 평면도";

                        this.targetCameraPos.set(0, 600, 0); // Top down
                        this.targetLookAt.set(0, 0, 0);
                        if (this.diagramGroup) this.diagramGroup.children[1].visible = true; // Show Plan Diagram
                        break;

                        title.innerText = "입면도";
                        content.innerHTML = `북측 입면.<br>도자기의 곡선을 따른 외관.<br>555m 높이의 측면 형상.`;
                        this.targetCameraPos.set(0, 100, 500);
                        this.targetLookAt.set(0, 277, 0);
                        if (this.diagramGroup) this.diagramGroup.children[2].visible = true; // Show Elev Diagram
                        break;
                    case 'section':
                        title.innerText = "단면도";

                        this.targetCameraPos.set(500, 100, 0);
                        this.targetLookAt.set(0, 277, 0);
                        if (this.diagramGroup) this.diagramGroup.children[3].visible = true; // Show Section Diagram
                        break;
                    case 'model':
                        title.innerText = "완성 모델";
                        content.innerHTML = `3D 형상의 전체 표현.<br>조작: 마우스 드래그(회전), 휠(확대/축소).<br>키보드: [WASD] 회전, [Q/E] 줌, [R] 초기화.`;
                        this.targetCameraPos.set(200, 150, 200);
                        this.targetLookAt.set(0, 50, 0);
                        break;
                }
            }

            onKeyDown(event) {
                const step = 10;
                const zoomStep = 20;
                
                // FIX: Keyboard controls for camera
                switch(event.key.toLowerCase()) {
                    case 'w': this.targetCameraPos.z -= step; break;
                    case 's': this.targetCameraPos.z += step; break;
                    case 'a': this.targetCameraPos.x -= step; break;
                    case 'd': this.targetCameraPos.x += step; break;
                    case 'q': this.targetCameraPos.y += zoomStep; break;
                    case 'e': this.targetCameraPos.y -= zoomStep; break;
                    case 'r': 
                        this.setStage('site'); 
                        break;
                }
            }

            onWindowResize() {
                this.camera.aspect = window.innerWidth / window.innerHeight;
                this.camera.updateProjectionMatrix();
                this.renderer.setSize(window.innerWidth, window.innerHeight);
            }

            animate() {
                requestAnimationFrame(() => this.animate());

                // Smooth Camera Transition
                this.camera.position.lerp(this.targetCameraPos, 0.05);
                this.controls.target.lerp(this.targetLookAt, 0.05);
                this.controls.update();

                // Subtle rotation for the "Model" stage if idle
                if (this.currentStage === 'model' && this.controls.getAzimuthalAngle() % (Math.PI * 2) < 0.01) {
                    // Optional: Auto rotate if user isn't interacting? 
                    // Better to let user control, but we can rotate the tower slightly for effect
                    // this.towerGroup.rotation.y += 0.001;
                }

                this.renderer.render(this.scene, this.camera);
            }
        }

        // Initialize App
        const app = new LotteApp();

    </script>
