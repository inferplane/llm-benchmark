      const e = new THREE.LineSegments(new THREE.EdgesGeometry(g), edgeMaterial);
      e.position.copy(m.position); tower.add(e);
    }
  }
  function build(stage) {
    while(tower.children.length) {
      const x=tower.children.pop(); x.geometry?.dispose();
    }
    if(stage===0) {
      frustum(0,509,45,45,8,material,true);
      frustum(509,555,45,8,8,crownMaterial,true);
    } else if(stage===1) {
      [[0,110,46,43],[110,230,43,36],[230,350,36,29],[350,440,29,21],[440,509,21,15]].forEach(a=>frustum(...a,8,material,true));
      frustum(509,555,15,7,8,crownMaterial,true);
    } else {
      [[0,90,46,44],[90,185,44,39],[185,285,39,33],[285,375,33,26],[375,445,26,19],[445,509,19,14]].forEach(a=>frustum(...a,12,material,true));
      frustum(509,532,14,10,12,crownMaterial,true);
      frustum(532,555,10,3.5,12,crownMaterial,true);
    }
  }
  let current=3, spinning=false;
  build(current);

  function resetView() {
    camera.position.set(760, 560, 760);
    controls.target.set(0,275,0);
    controls.update();
  }
  function setView(kind) {
    const d=1030;
    if(kind==='front') camera.position.set(0,390,d);
    if(kind==='side') camera.position.set(d,390,0);
    controls.target.set(0,275,0); controls.update();
  }
  function zoom(factor) {
    const v=camera.position.clone().sub(controls.target).multiplyScalar(factor);
    camera.position.copy(controls.target.clone().add(v));
    controls.update();
  }
  resetView();

  function selectStage(n) {
    current=n; build(n);
    label.innerHTML=stageInfo[n][0]+'<small>'+ (n===3?'555m = 본체 509m + 상부 46m':'설계 논리의 개념 단계') +'</small>';
    stageText.textContent=stageInfo[n][1];
    document.querySelectorAll('.stage').forEach((b,i)=>b.classList.toggle('active',i===n));
  }
  document.querySelectorAll('.stage').forEach((b,i)=>b.addEventListener('click',()=>selectStage(i)));
  document.getElementById('front').onclick=()=>setView('front');
  document.getElementById('side').onclick=()=>setView('side');
  document.getElementById('home').onclick=resetView;
  document.getElementById('zoomIn').onclick=()=>zoom(.82);
  document.getElementById('zoomOut').onclick=()=>zoom(1.22);
  document.getElementById('auto').onclick=function(){spinning=!spinning;this.textContent=spinning?'회전 끄기':'회전 켜기'};

  window.addEventListener('keydown', e=>{
    if(e.target.matches('button,a')) return;
    if(e.key>='1'&&e.key<='4') selectStage(+e.key-1);
    if(e.key.toLowerCase()==='r') resetView();
    if(e.key==='+'||e.key==='=') {e.preventDefault();zoom(.82)}
    if(e.key==='-'||e.key==='_') {e.preventDefault();zoom(1.22)}
    if(e.key==='ArrowLeft') tower.rotation.y-=.12;
    if(e.key==='ArrowRight') tower.rotation.y+=.12;
  });

  function resize() {
    const w=holder.clientWidth, h=holder.clientHeight;
    camera.aspect=w/h; camera.updateProjectionMatrix(); renderer.setSize(w,h,false);
  }
  new ResizeObserver(resize).observe(holder); resize();
  function animate() {
    requestAnimationFrame(animate);
    if(spinning && !matchMedia('(prefers-reduced-motion: reduce)').matches) tower.rotation.y+=.004;
    controls.update(); renderer.render(scene,camera);
  }
  animate();
})();
