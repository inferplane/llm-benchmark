  const T=THREE, scene=new T.Scene();scene.background=new T.Color(0x151e19);
  const camera=new T.PerspectiveCamera(36,1,.1,3000);camera.position.set(390,310,610);

  const goldMat=new T.MeshStandardMaterial({color:0xb5a071,roughness:.42,metalness:.24});
  const glassMat=new T.MeshStandardMaterial({color:0x83a99b,roughness:.23,metalness:.35,transparent:true,opacity:.78});
  const coreMat=new T.MeshStandardMaterial({color:0xc9b987,roughness:.65});
  const siteMat=new T.MeshStandardMaterial({color:0x586b5a,roughness:.82});
  function createTower(){
    while(building.children.length)building.remove(building.children[0]);
    if(stage===0){
      const base=new T.Mesh(new T.CylinderGeometry(55,67,5,48),siteMat);base.position.y=1;building.add(base);
      const footprint=new T.Mesh(new T.CylinderGeometry(13,17,18,32),goldMat);footprint.position.y=13;building.add(footprint);
      for(let i=0;i<8;i++){const a=i*Math.PI/4;const b=new T.Mesh(new T.BoxGeometry(10,2,7),new T.MeshStandardMaterial({color:0x71806d}));b.position.set(Math.cos(a)*39,4,Math.sin(a)*39);b.rotation.y=-a;building.add(b)}
      return;
    }
    const H=stage===1?250:stage===2?330:370;
    const levels=stage===1?12:stage===2?18:24, segments=48;
    const positions=[],indices=[];
    function radius(y){let t=y/H;return (stage===1?40:54)*(1-.46*t+.035*Math.sin(t*Math.PI));}
    for(let j=0;j<=levels;j++){
      const t=j/levels,y=t*H,r=radius(y),twist=(stage>=2?.10*t:0);
      for(let i=0;i<=segments;i++){
        const a=i/segments*Math.PI*2+twist;
        const power=stage>=2?2.25:2;
        const c=Math.sign(Math.cos(a))*Math.pow(Math.abs(Math.cos(a)),2/power);
        const s=Math.sign(Math.sin(a))*Math.pow(Math.abs(Math.sin(a)),2/power);
        positions.push(c*r,y,s*r);
      }
    }
    for(let j=0;j<levels;j++)for(let i=0;i<segments;i++){let a=j*(segments+1)+i,b=a+segments+1;indices.push(a,b,a+1,b,b+1,a+1)}
    const geo=new T.BufferGeometry();geo.setAttribute("position",new T.Float32BufferAttribute(positions,3));geo.setIndex(indices);geo.computeVertexNormals();
    const shell=new T.Mesh(geo,new T.MeshStandardMaterial({color:stage>=3?0x9db8ad:0xb6a578,roughness:stage>=3?.24:.46,metalness:.2,side:T.DoubleSide}));building.add(shell);
    if(stage===1||stage===2){
      const core=new T.Mesh(new T.CylinderGeometry(13,17,H*.98,12),coreMat);core.position.y=H/2;building.add(core);
    }
    if(stage>=2){
      const lines=stage===2?8:16;
      for(let k=0;k<lines;k++){
        const a=k/lines*Math.PI*2, pts=[];
        for(let j=0;j<=20;j++){const t=j/20,y=t*H,r=radius(y)+.35;pts.push(new T.Vector3(Math.cos(a+.1*t)*r,y,Math.sin(a+.1*t)*r))}
        const curve=new T.CatmullRomCurve3(pts);
        const line=new T.Mesh(new T.TubeGeometry(curve,40,stage===3?.52:.68,5,false),new T.MeshStandardMaterial({color:stage===3?0xd7c58f:0xe0c78d,roughness:.3,metalness:.3}));
        building.add(line);
      }
    }
    const base=new T.Mesh(new T.CylinderGeometry(58,62,3,48),siteMat);base.position.y=1;building.add(base);
    building.position.y=0;
  }
  window.updateTower=createTower;createTower();
  let controls;
  if(T.OrbitControls){controls=new T.OrbitControls(camera,renderer.domElement);controls.target.set(0,175,0);controls.enableDamping=true;controls.dampingFactor=.07;controls.minDistance=220;controls.maxDistance=1150;controls.maxPolarAngle=Math.PI*.88}
  else {
    let down=false,x=0,y=0,theta=Math.atan2(camera.position.x,camera.position.z),phi=Math.acos(camera.position.y/camera.position.length()),dist=camera.position.length();
    function place(){phi=Math.max(.12,Math.min(2.8,phi));camera.position.set(dist*Math.sin(phi)*Math.sin(theta),dist*Math.cos(phi),dist*Math.sin(phi)*Math.cos(theta));camera.lookAt(0,175,0)}
    canvas.addEventListener("pointerdown",e=>{down=true;x=e.clientX;y=e.clientY;canvas.setPointerCapture(e.pointerId)});
    canvas.addEventListener("pointermove",e=>{if(!down)return;theta-=(e.clientX-x)*.006;phi+=(e.clientY-y)*.006;x=e.clientX;y=e.clientY;place()});
    canvas.addEventListener("pointerup",()=>down=false);
    canvas.addEventListener("wheel",e=>{dist=Math.max(220,Math.min(1150,dist*Math.exp(e.deltaY*.001)));place()},{passive:true});
    canvas.addEventListener("keydown",e=>{if(e.key==="ArrowLeft")theta-=.08;if(e.key==="ArrowRight")theta+=.08;if(e.key==="ArrowUp")phi-=.06;if(e.key==="ArrowDown")phi+=.06;if(e.key==="+"||e.key==="=")dist*=.9;if(e.key==="-")dist*=1.1;place()});
  }

  setStage(3);
