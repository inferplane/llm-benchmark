<script src="../runtime/three-runtime.js"></script>
<script>
(() => {
    const container = document.getElementById('canvasContainer');
    const info = document.getElementById('infoPanel');
    const resetBtn = document.getElementById('resetBtn');
    const stageButtons = document.querySelectorAll('.navBtn');

    // Three.js 기본 설정
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 1, 3000);
    camera.position.set(0, 300, 800);
    const renderer = new THREE.WebGLRenderer({antialias:true});
    renderer.setSize(container.clientWidth, container.clientHeight);
    container.appendChild(renderer.domElement);

    const controls = new THREE.OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.screenSpacePanning = false;
    controls.minDistance = 200;
    controls.maxDistance = 2000;
    controls.maxPolarAngle = Math.PI/2;

    // 조명
    const hemi = new THREE.HemisphereLight(0xffffff,0x444444,1);
    scene.add(hemi);
    const dirLight = new THREE.DirectionalLight(0xffffff,0.8);
    dirLight.position.set(300,400,200);
    scene.add(dirLight);

    const mat = new THREE.MeshStandardMaterial({color:0x8da5b3, roughness:0.6, metalness:0.2});

    const baseHeight = 555; // 전체 높이 (예시)
    const baseRadius = 30;  // 바닥 반경 (예시)
    const topRadius = 5;    // 꼭대기 반경 (예시)

    function createTaperedCylinder(height, bottomR, topR) {
        const geo = new THREE.CylinderGeometry(bottomR, topR, height, 32, 1, true);
        const mesh = new THREE.Mesh(geo, mat);
        return mesh;
    }

    // 개념 단계: 단순 원통형 (높이 40%)
    const concept = createTaperedCylinder(baseHeight*0.4, baseRadius, baseRadius);
    concept.position.y = (baseHeight*0.4)/2;
    // 중간 단계: 두 구간 테이퍼
    const groupMid = new THREE.Group();
    const part1 = createTaperedCylinder(baseHeight*0.5, baseRadius, baseRadius*0.6);
    part1.position.y = (baseHeight*0.5)/2;
    const part2 = createTaperedCylinder(baseHeight*0.5, baseRadius*0.6, topRadius);
    part2.position.y = baseHeight*0.5 + (baseHeight*0.5)/2;
    groupMid.add(part1, part2);
    // 최종 단계: 연속 테이퍼 전체 높이
    const final = createTaperedCylinder(baseHeight, baseRadius, topRadius);
    final.position.y = baseHeight/2;

    const models = {concept, mid: groupMid, final};

    let current = null;
    function showStage(stage){
        if(current) scene.remove(current);
        current = models[stage];
        scene.add(current);
        info.innerHTML = `<strong>설명:</strong> 현재 <em>${stage}</em> 단계 모델을 표시합니다.`;
        highlightDrawing(stage);
    }

    function highlightDrawing(stage){
        const ids = ['sitePlan','floorPlan','elevation','section'];
        ids.forEach(id=>document.getElementById(id).style.borderColor='#999');
        if(stage==='concept') document.getElementById('sitePlan').style.borderColor='#0066cc';
        else if(stage==='mid') document.getElementById('floorPlan').style.borderColor='#0066cc';
        else if(stage==='final') document.getElementById('elevation').style.borderColor='#0066cc';
    }

    // 초기 표시
    showStage('concept');

    stageButtons.forEach(btn=>{
        btn.addEventListener('click',()=>showStage(btn.dataset.stage));
        btn.addEventListener('keydown',e=>{
            if(e.key==='Enter' || e.key===' ') {e.preventDefault(); showStage(btn.dataset.stage);}
        });
    });

    function resetView(){
        camera.position.set(0,300,800);
        controls.target.set(0,baseHeight/2,0);
        controls.update();
    }
    resetBtn.addEventListener('click',resetView);
    resetBtn.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' ') resetView();});

    // 키보드 제어
    window.addEventListener('keydown',e=>{
        if(e.target.tagName==='INPUT' || e.target.isContentEditable) return;
        switch(e.key){
            case 'r': case 'R': resetView(); break;
            case '+': controls.dollyOut(1.1); controls.update(); break; // zoom in
            case '-': controls.dollyIn(1.1); controls.update(); break;  // zoom out
            case 'ArrowUp': controls.rotateUp(0.05); controls.update(); break;
            case 'ArrowDown': controls.rotateDown(0.05); controls.update(); break;
            case 'ArrowLeft': controls.rotateLeft(0.05); controls.update(); break;
            case 'ArrowRight': controls.rotateRight(0.05); controls.update(); break;
            case 'w': case 'W': controls.rotateUp(0.05); controls.update(); break;
            case 's': case 'S': controls.rotateDown(0.05); controls.update(); break;
            case 'a': case 'A': controls.rotateLeft(0.05); controls.update(); break;
            case 'd': case 'D': controls.rotateRight(0.05); controls.update(); break;
        }
    });

    // 반응형
    window.addEventListener('resize',()=>{
        const w = container.clientWidth, h = container.clientHeight;
        camera.aspect = w/h;
        camera.updateProjectionMatrix();
        renderer.setSize(w,h);
    });

    function animate(){
        requestAnimationFrame(animate);
        controls.update();
        renderer.render(scene,camera);
    }
    animate();
})();
</script>
