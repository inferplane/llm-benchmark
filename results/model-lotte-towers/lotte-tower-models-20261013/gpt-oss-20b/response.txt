<script>
const stages = [
{label:'기본형', height:200, radiusTop:30, radiusBottom:50},
{label:'중간형', height:350, radiusTop:20, radiusBottom:40},
{label:'상단형', height:500, radiusTop:10, radiusBottom:30},
{label:'완성 모델', height:555, radiusTop:5, radiusBottom:25}
];
let currentStage = 3;
const stageLabel = document.getElementById('stageLabel');
const resetBtn = document.getElementById('resetBtn');
const stagePrev = document.getElementById('stagePrev');
const stageNext = document.getElementById('stageNext');
const canvas = document.getElementById('threeCanvas');
const renderer = new THREE.WebGLRenderer({canvas, antialias:true});
const container = document.getElementById('viewer');
function resize(){
const width = container.clientWidth;
const height = container.clientHeight;
renderer.setSize(width, height);
camera.aspect = width / height;
camera.updateProjectionMatrix();
}
resize();
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(45, container.clientWidth/container.clientHeight, 0.1, 2000);
camera.position.set(0, -400, 300);
const controls = new THREE.OrbitControls(camera, canvas);
controls.enableDamping = true;
controls.dampingFactor = 0.05;
controls.enablePan = false;
controls.minDistance = 200;
controls.maxDistance = 1000;
const light = new THREE.DirectionalLight(0xffffff, 1);
light.position.set(0, -1, 1);
scene.add(light);
const ambient = new THREE.AmbientLight(0x404040);
scene.add(ambient);
let tower;
function createTower(stage){
if(tower){scene.remove(tower);}
const geom = new THREE.CylinderGeometry(stage.radiusTop, stage.radiusBottom, stage.height, 64, 1, true);
const mat = new THREE.MeshPhongMaterial({color:0x8888ff, flatShading:true});
tower = new THREE.Mesh(geom, mat);
tower.position.y = stage.height/2;
scene.add(tower);
}
function updateStage(){
stageLabel.textContent = stages[currentStage].label;
createTower(stages[currentStage]);
}
stagePrev.onclick = () =>{currentStage=Math.max(0,currentStage-1);updateStage();};
stageNext.onclick = () =>{currentStage=Math.min(stages.length-1,currentStage+1);updateStage();};
resetBtn.onclick = () =>{controls.reset();};
window.addEventListener('resize',resize);
window.addEventListener('keydown',(e)=>{
if(e.key==='ArrowLeft'){stagePrev.click();}
if(e.key==='ArrowRight'){stageNext.click();}
if(e.key==='r'||e.key==='R'){resetBtn.click();}
if(e.key==='w'){controls.target.y+=10;controls.update();}
if(e.key==='s'){controls.target.y-=10;controls.update();}
if(e.key==='a'){controls.target.x-=10;controls.update();}
if(e.key==='d'){controls.target.x+=10;controls.update();}
if(e.key==='q'){controls.target.z+=10;controls.update();}
if(e.key==='e'){controls.target.z-=10;controls.update();}
});
updateStage();
function animate(){requestAnimationFrame(animate);controls.update();renderer.render(scene,camera);}
animate();
</script>
