            width: 25%;

<body>
    <div id="controls">
        <button id="stage-button">Stage</button>
        <button id="zoom-button">Zoom</button>
        <button id="reset-button">Reset</button>
        <button id="orbit-button">Orbit</button>
        <button id="tilt-button">Tilt</button>
        <button id="pointer-button">Pointer</button>
    </div>
    <canvas id="canvas"></canvas>
    <div id="diagrams">
        <canvas id="diagram1"></canvas>
        <canvas id="diagram2"></canvas>
        <canvas id="diagram3"></canvas>
        <canvas id="diagram4"></canvas>
    </div>
    <div id="model-name">llama-3.1-8b</div>
    <script src="../runtime/three-runtime.js"></script>
    <script>
        const canvas = document.getElementById('canvas');
        const diagrams = document.querySelectorAll('#diagrams canvas');
        const controls = document.getElementById('controls');
        const modelNameElement = document.getElementById('model-name');
        
        const renderer = new THREE.WebGLRenderer({
            canvas: canvas,
            antialias: true
        });
        renderer.setSize(canvas.width, canvas.height);
        
        const camera = new THREE.PerspectiveCamera(75, canvas.width / canvas.height, 0.1, 1000);
        camera.position.z = 500;
        
        const scene = new THREE.Scene();
        
        const diagram1 = new THREE.CanvasRenderer();
        diagram1.setSize(diagrams[0].width, diagrams[0].height);
        diagrams[0].appendChild(diagram1.domElement);
        
        const diagram2 = new THREE.CanvasRenderer();
        diagram2.setSize(diagrams[1].width, diagrams[1].height);
        diagrams[1].appendChild(diagram2.domElement);
        
        const diagram3 = new THREE.CanvasRenderer();
        diagram3.setSize(diagrams[2].width, diagrams[2].height);
        diagrams[2].appendChild(diagram3.domElement);
        
        const diagram4 = new THREE.CanvasRenderer();
        diagram4.setSize(diagrams[3].width, diagrams[3].height);
        diagrams[3].appendChild(diagram4.domElement);
        
        const stageButton = document.getElementById('stage-button');
        const zoomButton = document.getElementById('zoom-button');
        const resetButton = document.getElementById('reset-button');
        const orbitButton = document.getElementById('orbit-button');
        const tiltButton = document.getElementById('tilt-button');
        const pointerButton = document.getElementById('pointer-button');
        
        let stage = 0;
        let zoom = 1;
        let cameraPosition = new THREE.Vector3(0, 0, 500);
        let cameraTarget = new THREE.Vector3(0, 0, 0);
        let projection = new THREE.PerspectiveCamera(75, canvas.width / canvas.height, 0.1, 1000);
        
        function render() {
            requestAnimationFrame(render);
            renderer.render(scene, camera);
            diagram1.render(scene, camera);
            diagram2.render(scene, camera);
            diagram3.render(scene, camera);
            diagram4.render(scene, camera);
        }
        
        function onStageButtonClick() {
            stage = (stage + 1) % 4;
            updateScene();
        }
        
        function onZoomButtonClick() {
            zoom = (zoom + 0.1) % 2;
            camera.position.z = 500 * zoom;
            updateScene();
        }
        
        function onResetButtonClick() {
            stage = 0;
            zoom = 1;
            camera.position.copy(cameraPosition);
            camera.lookAt(cameraTarget);
            projection.copy(camera);
            updateScene();
        }
        
        function onOrbitButtonClick() {
            camera.position.x += 10;
            camera.lookAt(cameraTarget);
            updateScene();
        }
        
        function onTiltButtonClick() {
            camera.rotation.x += 0.1;
            updateScene();
        }
        
        function onPointerButtonClick() {
            camera.position.x += 10;
            camera.lookAt(cameraTarget);
            updateScene();
        }
        
        function updateScene() {
            scene.clear();
            switch (stage) {
                case 0:
                    drawCylinder(scene);
                    break;
                case 1:
                    drawFloor(scene);
                    break;
                case 2:
                    drawSection(scene);
                    break;
                case 3:
                    drawElevation(scene);
                    break;
            }
            camera.position.copy(cameraPosition);
            camera.lookAt(cameraTarget);
            projection.copy(camera);
            renderer.setSize(canvas.width, canvas.height);
            renderer.render(scene, camera);
            diagram1.setSize(diagrams[0].width, diagrams[0].height);
            diagram2.setSize(diagrams[1].width, diagrams[1].height);
            diagram3.setSize(diagrams[2].width, diagrams[2].height);
            diagram4.setSize(diagrams[3].width, diagrams[3].height);
            diagram1.render(scene, camera);
            diagram2.render(scene, camera);
            diagram3.render(scene, camera);
            diagram4.render(scene, camera);
        }
        
        function drawCylinder(scene) {
            const geometry = new THREE.CylinderGeometry(50, 50, 500, 32);
            const material = new THREE.MeshBasicMaterial({ color: 0xffffff });
            const cylinder = new THREE.Mesh(geometry, material);
            scene.add(cylinder);
        }
        
        function drawFloor(scene) {
            const geometry = new THREE.PlaneGeometry(100, 100);
            const material = new THREE.MeshBasicMaterial({ color: 0xffffff });
            const floor = new THREE.Mesh(geometry, material);
            floor.position.y = -500;
            scene.add(floor);
        }
        
        function drawSection(scene) {
            const geometry = new THREE.BoxGeometry(100, 100, 100);
            const material = new THREE.MeshBasicMaterial({ color: 0xffffff });
            const section = new THREE.Mesh(geometry, material);
            section.position.x = 50;
            section.position.y = -250;
            scene.add(section);
        }
        
        function drawElevation(scene) {
            const geometry = new THREE.BoxGeometry(100, 100, 100);
            const material = new THREE.MeshBasicMaterial({ color: 0xffffff });
            const elevation = new THREE.Mesh(geometry, material);
            elevation.position.x = -50;
            elevation.position.y = -250;
            scene.add(elevation);
        }
        
        render();
        
        stageButton.addEventListener('click', onStageButtonClick);
        zoomButton.addEventListener('click', onZoomButtonClick);
        resetButton.addEventListener('click', onResetButtonClick);
        orbitButton.addEventListener('click', onOrbitButtonClick);
        tiltButton.addEventListener('click', onTiltButtonClick);
        pointerButton.addEventListener('click', onPointerButtonClick);
        
        modelNameElement.textContent = 'llama-3.1-8b';
    </script>
</body>
