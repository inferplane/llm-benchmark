    <script src="../runtime/three-runtime.js"></script>
    <script>
        let scene, camera, renderer, controls;
        let stage = 0;
        const stages = [
            { name: '초기 단계', geometry: createTowerGeometry(100, 50) },
            { name: '중간 단계', geometry: createTowerGeometry(200, 30) },
            { name: '완료 단계', geometry: createTowerGeometry(300, 20) },
            { name: '최종 단계', geometry: createTowerGeometry(400, 10) },
        ];

        function createTowerGeometry(height, radius) {
            const geometry = new THREE.CylinderGeometry(radius, radius / 2, height, 64);
            return geometry;
        }

        init();
        animate();

        function init() {
            scene = new THREE.Scene();
            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
            renderer = new THREE.WebGLRenderer({
                canvas: document.getElementById('canvas'),
                antialias: true
            });
            renderer.setSize(window.innerWidth, window.innerHeight * 0.8);
            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;

            const geometry = stages[stage].geometry;
            const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
            const mesh = new THREE.Mesh(geometry, material);
            scene.add(mesh);

            camera.position.z = 500;
            controls.update();

            document.getElementById('zoom-in').addEventListener('click', () => {
                controls.dollyIn(0.5);
            });
            document.getElementById('zoom-out').addEventListener('click', () => {
                controls.dollyOut(0.5);
            });
            document.getElementById('reset-view').addEventListener('click', () => {
                camera.position.z = 500;
                controls.update();
            });
            document.getElementById('prev-stage').addEventListener('click', () => {
                stage = (stage - 1 + stages.length) % stages.length;
                scene.clear();
                const geometry = stages[stage].geometry;
                const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
                const mesh = new THREE.Mesh(geometry, material);
                scene.add(mesh);
            });
            document.getElementById('next-stage').addEventListener('click', () => {
                stage = (stage + 1) % stages.length;
                scene.clear();
                const geometry = stages[stage].geometry;
                const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
                const mesh = new THREE.Mesh(geometry, material);
                scene.add(mesh);
            });

            document.addEventListener('keydown', (event) => {
                switch (event.key) {
                    case 'ArrowUp':
                        controls.rotateUp(Math.PI / 180);
                        break;
                    case 'ArrowDown':
                        controls.rotateDown(Math.PI / 180);
                        break;
                    case 'ArrowLeft':
                        controls.rotateLeft(Math.PI / 180);
                        break;
                    case 'ArrowRight':
                        controls.rotateRight(Math.PI / 180);
                        break;
                }
            });

            window.addEventListener('resize', () => {
                renderer.setSize(window.innerWidth, window.innerHeight * 0.8);
                camera.aspect = window.innerWidth / window.innerHeight;
                camera.updateProjectionMatrix();
            });
        }

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }
    </script>
