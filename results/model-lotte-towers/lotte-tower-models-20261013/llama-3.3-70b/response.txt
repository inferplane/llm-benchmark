        let scene = new THREE.Scene();
        let camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        let renderer = new THREE.WebGLRenderer({
            antialias: true
        });
        renderer.setSize(window.innerWidth, window.innerHeight * 0.7);
        document.getElementById('three-container').appendChild(renderer.domElement);
        let towerGeometry = new THREE.BufferGeometry();
        let vertices = [];
        for (let i = 0; i <= 123; i++) {
            let width = 100 - (i / 123) * 50;
            let height = i * 555 / 123;
            vertices.push(-width / 2, 0, -width / 2);
            vertices.push(width / 2, 0, -width / 2);
            vertices.push(width / 2, 0, width / 2);
            vertices.push(-width / 2, 0, width / 2);
            vertices.push(-width / 2, height, -width / 2);
            vertices.push(width / 2, height, -width / 2);
            vertices.push(width / 2, height, width / 2);
            vertices.push(-width / 2, height, width / 2);
        }
        let indices = [];
        for (let i = 0; i < 123; i++) {
            let base = i * 8;
            let top = (i + 1) * 8;
            indices.push(base, base + 1, base + 2);
            indices.push(base, base + 2, base + 3);
            indices.push(top, top + 1, top + 2);
            indices.push(top, top + 2, top + 3);
            indices.push(base, top, base + 1);
            indices.push(top, base + 1, top + 1);
            indices.push(base + 2, top + 2, base + 3);
            indices.push(top + 2, base + 3, top + 3);
        }
        towerGeometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
        towerGeometry.setIndex(indices);
        let towerMaterial = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
        let tower = new THREE.Mesh(towerGeometry, towerMaterial);
        scene.add(tower);
        camera.position.z = 500;
        let orbitControls = new THREE.OrbitControls(camera, renderer.domElement);
        function animate() {
            requestAnimationFrame(animate);
            renderer.render(scene, camera);
        }
        animate();
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight * 0.7);
        });
        document.getElementById('stage-1').addEventListener('click', () => {
            tower.geometry.dispose();
            tower.geometry = new THREE.BoxGeometry(100, 100, 100);
            tower.position.y = 50;
        });
        document.getElementById('stage-2').addEventListener('click', () => {
            tower.geometry.dispose();
            tower.geometry = new THREE.BoxGeometry(80, 200, 80);
            tower.position.y = 100;
        });
        document.getElementById('stage-3').addEventListener('click', () => {
            tower.geometry.dispose();
            tower.geometry = towerGeometry;
            tower.position.y = 0;
        });
        document.getElementById('rotate').addEventListener('click', () => {
            tower.rotation.x += 0.1;
            tower.rotation.y += 0.1;
        });
        document.getElementById('zoom').addEventListener('click', () => {
            camera.position.z -= 50;
        });
        document.getElementById('reset').addEventListener('click', () => {
            camera.position.z = 500;
            camera.position.x = 0;
            camera.position.y = 0;
            camera.rotation.x = 0;
            camera.rotation.y = 0;
            tower.rotation.x = 0;
            tower.rotation.y = 0;
        });
        document.addEventListener('keydown', (event) => {
            if (event.key === ' ') {
                tower.rotation.x += 0.1;
                tower.rotation.y += 0.1;
            } else if (event.key === '+') {
                camera.position.z -= 50;
            } else if (event.key === '-') {
                camera.position.z += 50;
            } else if (event.key === 'r') {
                camera.position.z = 500;
                camera.position.x = 0;
                camera.position.y = 0;
                camera.rotation.x = 0;
                camera.rotation.y = 0;
                tower.rotation.x = 0;
                tower.rotation.y = 0;
            }
        });
