<body>
    <canvas id="canvas"></canvas>
    <div class="controls">
        <button id="reset-camera">초기화</button>
        <button id="stage-1">1단계</button>
        <button id="stage-2">2단계</button>
        <button id="stage-3">3단계</button>
        <button id="complete">완료 모델</button>
    </div>
    <div class="info">llama4-scout</div>

    <div id="site-plan" class="diagram">
        <canvas width="200" height="200"></canvas>
    </div>
    <div id="floor-plan" class="diagram">
        <canvas width="200" height="200"></canvas>
    </div>
    <div id="elevation" class="diagram">
        <canvas width="200" height="200"></canvas>
    </div>
    <div id="section" class="diagram">
        <canvas width="200" height="200"></canvas>
    </div>

    <script src="../runtime/three-runtime.js"></script>
    <script>
        let camera, scene, renderer, controls;
        let stage1, stage2, stage3, completeModel;
        let currentStage = null;

        // Initialize Three.js
        init();

        function init() {
            // Create scene and camera
            scene = new THREE.Scene();
            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.set(0, 0, 500);

            // Create renderer
            renderer = new THREE.WebGLRenderer({
                canvas: document.getElementById('canvas'),
                antialias: true
            });
            renderer.setSize(window.innerWidth, window.innerHeight);

            // Create OrbitControls
            controls = new THREE.OrbitControls(camera, renderer.domElement);

            // Create models
            createModels();

            // Add event listeners
            document.getElementById('reset-camera').addEventListener('click', resetCamera);
            document.getElementById('stage-1').addEventListener('click', () => showStage(stage1));
            document.getElementById('stage-2').addEventListener('click', () => showStage(stage2));
            document.getElementById('stage-3').addEventListener('click', () => showStage(stage3));
            document.getElementById('complete').addEventListener('click', () => showStage(completeModel));

            // Show initial stage
            showStage(stage1);

            // Handle window resize
            window.addEventListener('resize', onWindowResize);

            // Handle keydown event
            document.addEventListener('keydown', onKeyDown);

            // Reduce motion
            const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
            if (reduceMotion.matches) {
                controls.enableDamping = false;
            }

            // Draw diagrams
            drawDiagrams();
        }

        function createModels() {
            // Create stage 1
            stage1 = createStage1();

            // Create stage 2
            stage2 = createStage2();

            // Create stage 3
            stage3 = createStage3();

            // Create complete model
            completeModel = createCompleteModel();

            // Add models to scene
            scene.add(stage1);
            scene.add(stage2);
            scene.add(stage3);
            scene.add(completeModel);

            // Hide all models except stage 1
            stage2.visible = false;
            stage3.visible = false;
            completeModel.visible = false;
        }

        function createStage1() {
            // Create a simple box
            const geometry = new THREE.BoxGeometry(100, 100, 100);
            const material = new THREE.MeshBasicMaterial({ color: 0xff0000 });
            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.set(0, 50, 0);
            return mesh;
        }

        function createStage2() {
            // Create a tapered box
            const geometry = new THREE.BoxGeometry(90, 120, 90);
            const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.set(0, 60, 0);
            return mesh;
        }

        function createStage3() {
            // Create a more tapered box
            const geometry = new THREE.BoxGeometry(80, 150, 80);
            const material = new THREE.MeshBasicMaterial({ color: 0x0000ff });
            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.set(0, 75, 0);
            return mesh;
        }

        function createCompleteModel() {
            // Create a final tapered model
            const geometry = new THREE.ConeGeometry(50, 200, 32);
            const material = new THREE.MeshBasicMaterial({ color: 0xffff00 });
            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.set(0, 100, 0);
            return mesh;
        }

        function drawDiagrams() {
            // Draw site plan
            const sitePlanCanvas = document.getElementById('site-plan').querySelector('canvas');
            const sitePlanCtx = sitePlanCanvas.getContext('2d');
            sitePlanCtx.clearRect(0, 0, 200, 200);
            sitePlanCtx.beginPath();
            sitePlanCtx.rect(50, 50, 100, 100);
            sitePlanCtx.stroke();
            sitePlanCtx.font = '12px Arial';
            sitePlanCtx.fillStyle = '#000';
            sitePlanCtx.textAlign = 'left';
            sitePlanCtx.textBaseline = 'top';
            sitePlanCtx.fillText('배치도 (m)', 10, 10);
            sitePlanCtx.fillText('100m', 50, 160);

            // Draw floor plan
            const floorPlanCanvas = document.getElementById('floor-plan').querySelector('canvas');
            const floorPlanCtx = floorPlanCanvas.getContext('2d');
            floorPlanCtx.clearRect(0, 0, 200, 200);
            floorPlanCtx.beginPath();
            floorPlanCtx.rect(25, 25, 50, 50);
            floorPlanCtx.stroke();
            floorPlanCtx.font = '12px Arial';
            floorPlanCtx.fillStyle = '#000';
            floorPlanCtx.textAlign = 'left';
            floorPlanCtx.textBaseline = 'top';
            floorPlanCtx.fillText('기준층 평면도 (m)', 10, 10);
            floorPlanCtx.fillText('50m', 25, 80);

            // Draw elevation
            const elevationCanvas = document.getElementById('elevation').querySelector('canvas');
            const elevationCtx = elevationCanvas.getContext('2d');
            elevationCtx.clearRect(0, 0, 200, 200);
            elevationCtx.beginPath();
            elevationCtx.rect(50, 50, 20, 100);
            elevationCtx.stroke();
            elevationCtx.font = '12px Arial';
            elevationCtx.fillStyle = '#000';
            elevationCtx.textAlign = 'left';
            elevationCtx.textBaseline = 'top';
            elevationCtx.fillText('입면도 (m)', 10, 10);
            elevationCtx.fillText('100m', 70, 150);

            // Draw section
            const sectionCanvas = document.getElementById('section').querySelector('canvas');
            const sectionCtx = sectionCanvas.getContext('2d');
            sectionCtx.clearRect(0, 0, 200, 200);
            sectionCtx.beginPath();
            sectionCtx.rect(50, 50, 20, 100);
            sectionCtx.stroke();
            sectionCtx.font = '12px Arial';
            sectionCtx.fillStyle = '#000';
            sectionCtx.textAlign = 'left';
            sectionCtx.textBaseline = 'top';
            sectionCtx.fillText('단면도 (m)', 10, 10);
            sectionCtx.fillText('100m', 70, 150);
        }

        function showStage(stage) {
            if (currentStage) {
                currentStage.visible = false;
            }
            stage.visible = true;
            currentStage = stage;
            controls.reset();
            camera.position.set(0, 0, 500);
            camera.lookAt(0, 0, 0);
        }

        function resetCamera() {
            camera.position.set(0, 0, 500);
            camera.lookAt(0, 0, 0);
            controls.reset();
        }

        function onWindowResize() {
            const container = document.getElementById('canvas').parentNode;
            camera.aspect = container.offsetWidth / container.offsetHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(container.offsetWidth, container.offsetHeight);
        }

        function onKeyDown(event) {
            switch (event.key) {
                case 'r':
                    controls.reset();
                    break;
                case '+':
                case '=':
                    camera.zoom += 0.1;
                    camera.updateProjectionMatrix();
                    break;
                case '-':
                    camera.zoom -= 0.1;
                    camera.updateProjectionMatrix();
                    break;
                case 'ArrowUp':
                    camera.rotateX(0.1);
                    break;
                case 'ArrowDown':
                    camera.rotateX(-0.1);
                    break;
                case 'ArrowLeft':
                    camera.rotateY(0.1);
                    break;
                case 'ArrowRight':
                    camera.rotateY(-0.1);
                    break;
            }
        }

        // Animate
        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }
        animate();
    </script>
</body>
</html>
