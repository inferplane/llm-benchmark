        document.addEventListener('DOMContentLoaded', function() {
            // Setup Three.js environment
            const scene = new THREE.Scene();
            const camera = new THREE.PerspectiveCamera(75, 1, 0.1, 10000);
            const renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setPixelRatio(window.devicePixelRatio);
            renderer.setSize(800, 500);

            const canvasContainer = document.getElementById('model-canvas');
            canvasContainer.appendChild(renderer.domElement);

            // OrbitControls
            const controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.screenSpacePanning = false;
            controls.minDistance = 200;
            controls.maxDistance = 2000;
            controls.maxPolarAngle = Math.PI * 0.9;

            // Keyboard controls
            const keys = {
                up: false,
                down: false,
                left: false,
                right: false,
                shift: false,
                ctrl: false,
                alt: false
            };

            function setupKeyboardControls() {
                window.addEventListener('keydown', function(e) {
                    if (e.key === 'Shift') keys.shift = true;
                    else if (e.key === 'Control') keys.ctrl = true;
                    else if (e.key === 'Alt') keys.alt = true;
                    else if (e.key === 'ArrowUp') keys.up = true;
                    else if (e.key === 'ArrowDown') keys.down = true;
                    else if (e.key === 'ArrowLeft') keys.left = true;
                    else if (e.key === 'ArrowRight') keys.right = true;
                });

                window.addEventListener('keyup', function(e) {
                    if (e.key === 'Shift') keys.shift = false;
                    else if (e.key === 'Control') keys.ctrl = false;
                    else if (e.key === 'Alt') keys.alt = false;
                    else if (e.key === 'ArrowUp') keys.up = false;
                    else if (e.key === 'ArrowDown') keys.down = false;
                    else if (e.key === 'ArrowLeft') keys.left = false;
                    else if (e.key === 'ArrowRight') keys.right = false;
                });
            }

            // Mouse wheel zoom
            renderer.domElement.addEventListener('wheel', function(e) {
                e.preventDefault();
                const delta = e.deltaY > 0 ? -1 : 1;
                controls.dollyOut(delta * 0.01 * (keys.shift ? 0.1 : 1));
            }, { passive: false });

            // Animation loop
            function animate() {
                requestAnimationFrame(animate);

                // Keyboard movement
                if (keys.up) controls.pan(new THREE.Vector3(0, 0, -1 * (keys.shift ? 0.5 : keys.ctrl ? 0.1 : 0.2)));
                if (keys.down) controls.pan(new THREE.Vector3(0, 0, 1 * (keys.shift ? 0.5 : keys.ctrl ? 0.1 : 0.2)));
                if (keys.left) controls.pan(new THREE.Vector3(-1 * (keys.shift ? 0.5 : keys.ctrl ? 0.1 : 0.2), 0, 0));
                if (keys.right) controls.pan(new THREE.Vector3(1 * (keys.shift ? 0.5 : keys.ctrl ? 0.1 : 0.2), 0, 0));

                // Alt key for faster rotation
                if (keys.alt) {
                    controls.rotateLeft(0.01);
                }

                controls.update();
                renderer.render(scene, camera);
            }

            // Reset view button
            document.getElementById('reset-view').addEventListener('click', function() {
                camera.position.set(500, 500, 500);
                camera.lookAt(0, 0, 0);
                controls.target.set(0, 0, 0);
                controls.update();
            });

            // Create models for different stages
            function createConceptModel() {
                scene.children = [];

                // Base platform
                const platformGeometry = new THREE.BufferGeometry();
                const platformVertices = new Float32Array([
                    // Base
                    -100, 0, -100, 100, 0, -100, 100, 0, 100, -100, 0, 100,
                    // Top (smaller)
                    -50, 500, -50, 50, 500, -50, 50, 500, 50, -50, 500, 50
                ]);
                platformGeometry.setAttribute('position', new THREE.BufferAttribute(platformVertices, 3));

                const platformMaterial = new THREE.MeshBasicMaterial({
                    color: 0xcccccc,
                    wireframe: true
                });
                const platform = new THREE.Mesh(platformGeometry, platformMaterial);
                scene.add(platform);

                // Simple tower outline
                const towerGeometry = new THREE.BufferGeometry();
                const towerVertices = [];
                const segments = 12;
                const height = 500;

                for (let i = 0; i <= segments; i++) {
                    const angle = (i / segments) * Math.PI * 2;
                    const radius = 100 - (i / segments) * 50;
                    towerVertices.push(
                        radius * Math.cos(angle), 0, radius * Math.sin(angle),
                        radius * Math.cos(angle), height, radius * Math.sin(angle)
                    );
                }

                towerGeometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array(towerVertices), 3));

                const towerMaterial = new THREE.MeshBasicMaterial({
                    color: 0x666666,
                    wireframe: true,
                    transparent: true,
                    opacity: 0.7
                });
                const tower = new THREE.Mesh(towerGeometry, towerMaterial);
                scene.add(tower);

                // Add some conceptual elements
                const coreGeometry = new THREE.BufferGeometry();
                const coreVertices = new Float32Array([
                    -10, 0, -10, 10, 0, -10, 10, 0, 10, -10, 0, 10,
                    -10, height, -10, 10, height, -10, 10, height, 10, -10, height, 10
                ]);
                coreGeometry.setAttribute('position', new THREE.BufferAttribute(coreVertices, 3));

                const coreMaterial = new THREE.MeshBasicMaterial({
                    color: 0x333333,
                    wireframe: true
                });
                const core = new THREE.Mesh(coreGeometry, coreMaterial);
                scene.add(core);

                // Position camera
                camera.position.set(300, 300, 300);
                camera.lookAt(0, 0, 0);
                controls.target.set(0, 0, 0);
            }

                    const angle = (i / segments) * Math.PI * 2;
                    const radius = 100 - (progress * progress * 0.8) * 50;
                    const nextProgress = Math.min(progress + 0.01, 1);
                    const nextRadius = 100 - (nextProgress * nextProgress * 0.8) * 50;

                    towerVertices.push(
                        radius * Math.cos(angle), progress * height, radius * Math.sin(angle),
                        nextRadius * Math.cos(angle), nextProgress * height, nextRadius * Math.sin(angle)
                    );
                }

                towerGeometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array(towerVertices), 3));

                    wireframe: true,
                    transparent: true,
