<body>
    <h1>롯데월드타워 건축 설계 과정</h1>
    <div class="container">
        <div class="drawing">
            <canvas id="site-plan"></canvas>
            <canvas id="floor-plan"></canvas>
            <canvas id="elevation"></canvas>
            <canvas id="section"></canvas>
        </div>
        <div class="model">
            <canvas id="model"></canvas>
        </div>
    </div>
    <div class="controls">
        <button class="control-button" id="rotate">회전</button>
        <button class="control-button" id="zoom-in">확대</button>
        <button class="control-button" id="zoom-out">축소</button>
        <button class="control-button" id="reset-view">시점 초기화</button>

    <script src="../runtime/three-runtime.js"></script>
    <script>
        const canvasWidth = 600;
        const canvasHeight = 400;
        const sitePlanCanvas = document.getElementById('site-plan');
        const floorPlanCanvas = document.getElementById('floor-plan');
        const elevationCanvas = document.getElementById('elevation');
        const sectionCanvas = document.getElementById('section');
        const modelCanvas = document.getElementById('model');

        const sitePlanCtx = sitePlanCanvas.getContext('2d');
        const floorPlanCtx = floorPlanCanvas.getContext('2d');
        const elevationCtx = elevationCanvas.getContext('2d');
        const sectionCtx = sectionCanvas.getContext('2d');

        sitePlanCanvas.width = canvasWidth;
        sitePlanCanvas.height = canvasHeight;
        floorPlanCanvas.width = canvasWidth;
        floorPlanCanvas.height = canvasHeight;
        elevationCanvas.width = canvasWidth;
        elevationCanvas.height = canvasHeight;
        sectionCanvas.width = canvasWidth;
        sectionCanvas.height = canvasHeight;
        modelCanvas.width = canvasWidth;
        modelCanvas.height = canvasHeight;

        // Drawing the conceptual diagrams
        drawSitePlan();
        drawFloorPlan();
        drawElevation();
        drawSection();

        function drawSitePlan() {
            sitePlanCtx.fillStyle = '#ffffff';
            sitePlanCtx.fillRect(0, 0, canvasWidth, canvasHeight);
            sitePlanCtx.fillStyle = '#000000';
            sitePlanCtx.font = '24px Arial';
            sitePlanCtx.fillText('부지 평면도', 10, 30);
            drawGrid(sitePlanCtx, canvasWidth, canvasHeight, 20);
        }

        function drawFloorPlan() {
            floorPlanCtx.fillStyle = '#ffffff';
            floorPlanCtx.fillRect(0, 0, canvasWidth, canvasHeight);
            floorPlanCtx.fillStyle = '#000000';
            floorPlanCtx.font = '24px Arial';
            floorPlanCtx.fillText('기준층 평면도', 10, 30);
            drawGrid(floorPlanCtx, canvasWidth, canvasHeight, 20);
        }

        function drawElevation() {
            elevationCtx.fillStyle = '#ffffff';
            elevationCtx.fillRect(0, 0, canvasWidth, canvasHeight);
            elevationCtx.fillStyle = '#000000';
            elevationCtx.font = '24px Arial';
            elevationCtx.fillText('입면도', 10, 30);
            drawGrid(elevationCtx, canvasWidth, canvasHeight, 20);
        }

        function drawSection() {
            sectionCtx.fillStyle = '#ffffff';
            sectionCtx.fillRect(0, 0, canvasWidth, canvasHeight);
            sectionCtx.fillStyle = '#000000';
            sectionCtx.font = '24px Arial';
            sectionCtx.fillText('단면도', 10, 30);
            drawGrid(sectionCtx, canvasWidth, canvasHeight, 20);
        }

        function drawGrid(ctx, width, height, step) {
            ctx.beginPath();
            for (let x = step; x <= width; x += step) {
                ctx.moveTo(x, 0);
                ctx.lineTo(x, height);
            }
            for (let y = step; y <= height; y += step) {
                ctx.moveTo(0, y);
                ctx.lineTo(width, y);
            }
            ctx.strokeStyle = '#cccccc';
            ctx.stroke();
        }

        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        const renderer = new THREE.WebGLRenderer({ canvas: modelCanvas, antialias: true });
        const controls = new THREE.OrbitControls(camera, renderer.domElement);

        camera.position.z = 5;

        function createTowerGeometry() {
            const geometry = new THREE.Geometry();
            const height = 555;
            const baseWidth = 50;
            const topWidth = 10;
            const steps = 123;
            const stepHeight = height / steps;

            for (let i = 0; i <= steps; i++) {
                const y = i * stepHeight;
                const width = baseWidth - (baseWidth - topWidth) * (i / steps);
                const vertices = [
                    new THREE.Vector3(-width / 2, y, -width / 2),
                    new THREE.Vector3(width / 2, y, -width / 2),
                    new THREE.Vector3(width / 2, y + stepHeight, -width / 2),
                    new THREE.Vector3(-width / 2, y + stepHeight, -width / 2),
                    new THREE.Vector3(-width / 2, y, width / 2),
                    new THREE.Vector3(width / 2, y, width / 2),
                    new THREE.Vector3(width / 2, y + stepHeight, width / 2),
                    new THREE.Vector3(-width / 2, y + stepHeight, width / 2)
                ];

                const faces = [
                    new THREE.Face3(0, 1, 2),
                    new THREE.Face3(0, 2, 3),
                    new THREE.Face3(4, 5, 6),
                    new THREE.Face3(4, 6, 7),
                    new THREE.Face3(0, 1, 5),
                    new THREE.Face3(1, 2, 6),
                    new THREE.Face3(2, 3, 7),
                    new THREE.Face3(3, 0, 4),
                    new THREE.Face3(4, 5, 1),
                    new THREE.Face3(5, 6, 2),
                    new THREE.Face3(6, 7, 3),
                    new THREE.Face3(7, 4, 0)
                ];

                geometry.vertices = geometry.vertices.concat(vertices);
                geometry.faces = geometry.faces.concat(faces);
            }

            const material = new THREE.MeshBasicMaterial({ color: 0x00ff00, wireframe: true });
            const tower = new THREE.Mesh(geometry, material);
            scene.add(tower);
            return tower;
        }

        const tower = createTowerGeometry();
        scene.add(tower);

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }

        animate();

        document.getElementById('rotate').addEventListener('click', () => {
            tower.rotation.x += 0.1;
            tower.rotation.y += 0.1;
        });

        document.getElementById('zoom-in').addEventListener('click', () => {
            camera.position.z -= 1;
        });

        document.getElementById('zoom-out').addEventListener('click', () => {
            camera.position.z += 1;
        });

        document.getElementById('reset-view').addEventListener('click', () => {
            camera.position.set(0, 0, 5);
            tower.rotation.set(0, 0, 0);
        });

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });
    </script>
