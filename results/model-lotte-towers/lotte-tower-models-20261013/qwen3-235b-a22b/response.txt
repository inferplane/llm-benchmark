          function renderMesh(mesh, viewProjMat, gl) {
            const vertices = mesh.geometry.vertices;
            const indices = mesh.geometry.indices;
            const color = mesh.material.color;
            const pos = mesh.position;
            const projVertices = [];
            for (let i = 0; i < vertices.length; i += 3) {
              const x = vertices[i] + pos.x;
              const y = vertices[i+1] + pos.y;
              const z = vertices[i+2] + pos.z;
              const w = viewProjMat.getElement(3,0)*x + viewProjMat.getElement(3,1)*y + viewProjMat.getElement(3,2)*z + viewProjMat.getElement(3,3);
              if (w <= 0) continue;
              const px = (viewProjMat.getElement(0,0)*x + viewProjMat.getElement(0,1)*y + viewProjMat.getElement(0,2)*z + viewProjMat.getElement(0,3)) / w;
              const py = (viewProjMat.getElement(1,0)*x + viewProjMat.getElement(1,1)*y + viewProjMat.getElement(1,2)*z + viewProjMat.getElement(1,3)) / w;
              const pz = (viewProjMat.getElement(2,0)*x + viewProjMat.getElement(2,1)*y + viewProjMat.getElement(2,2)*z + viewProjMat.getElement(2,3)) / w;
              projVertices.push({ x: px, y: py, z: pz, r: color.r, g: color.g, b: color.b });
            }
            gl.beginPath();
            for (let i = 0; i < indices.length; i += 3) {
              const i1 = indices[i], i2 = indices[i+1], i3 = indices[i+2];
              if (i1 >= projVertices.length || i2 >= projVertices.length || i3 >= projVertices.length) continue;
              const v1 = projVertices[i1], v2 = projVertices[i2], v3 = projVertices[i3];
              if (!v1 || !v2 || !v3) continue;
              const cx = gl.canvas.width / 2;
              const cy = gl.canvas.height / 2;
              const scale = Math.min(cx, cy);
              const x1 = cx + v1.x * scale;
              const y1 = cy - v1.y * scale;
              const x2 = cx + v2.x * scale;
              const y2 = cy - v2.y * scale;
              const x3 = cx + v3.x * scale;
              const y3 = cy - v3.y * scale;
              if (v1.z < 0 && v2.z < 0 && v3.z < 0) continue;
              gl.moveTo(x1, y1);
              gl.lineTo(x2, y2);
              gl.lineTo(x3, y3);
              gl.closePath();
            }
            gl.strokeStyle = 'rgba(0,51,102,0.8)';
            gl.lineWidth = 1;
            gl.stroke();
            gl.fillStyle = 'rgba(0,51,102,0.1)';
            gl.fill();

            Scene, PerspectiveCamera, WebGLRenderer, BoxGeometry,

    // 전역 상태
    let currentStage = 'plan';
    let autoRotate = false;
    const stageGeometries = {
      initial: { base: 80, top: 60, height: 400, floors: 30 },
      mid: { base: 70, top: 40, height: 480, floors: 80 },
      final: { base: 60, top: 20, height: 555, floors: 123 }
    };

    // DOM 요소
    const canvas = document.getElementById('diagramCanvas');
    const ctx = canvas.getContext('2d');
    const modelContainer = document.getElementById('modelContainer');
    const modelCanvas = document.getElementById('modelCanvas');
    const legend = document.getElementById('diagramLegend');
    const stageItems = document.querySelectorAll('.stage-item');
    const rotateBtn = document.getElementById('rotateBtn');
    const resetBtn = document.getElementById('resetBtn');
    const zoomInBtn = document.getElementById('zoomInBtn');
    const zoomOutBtn = document.getElementById('zoomOutBtn');

    // 리사이즈 핸들러
    function resize() {
      const rect = canvas.getBoundingClientRect();
      canvas.width = rect.width * window.devicePixelRatio;
      canvas.height = rect.height * window.devicePixelRatio;
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);

      const modelRect = modelContainer.getBoundingClientRect();
      if (modelRenderer && modelCamera) {
        modelRenderer.setSize(modelRect.width, modelRect.height, false);
        modelCamera.aspect = modelRect.width / modelRect.height;
        modelCamera.updateProjectionMatrix();
      }
    }
    window.addEventListener('resize', resize);

    // 도면 그리기
    function drawDiagram(stage) {
      const rect = canvas.getBoundingClientRect();
      const width = rect.width;
      const height = rect.height;
      ctx.clearRect(0, 0, width, height);
      ctx.save();
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);

      const centerX = width / 2;
      const centerY = height / 2;
      const scale = Math.min(width, height) * 0.35 / 100;

      ctx.strokeStyle = '#003366';
      ctx.lineWidth = 2;
      ctx.fillStyle = '#003366';
      ctx.font = '19px sans-serif';

      if (stage === 'plan') {
        const w = 60 * scale;
        const h = 40 * scale;
        ctx.strokeRect(centerX - w/2, centerY - h/2, w, h);
        ctx.fillText('기준층 평면도', centerX - 60, centerY - h/2 - 10);
        ctx.font = '17px sans-serif';
        ctx.fillText('단위: m (개념 추정)', centerX - 70, height - 20);
      } else if (stage === 'elevation') {
        const path = new Path2D();
        path.moveTo(centerX - 30*scale, centerY + 200*scale);
        path.lineTo(centerX - 10*scale, centerY - 200*scale);
        path.lineTo(centerX + 10*scale, centerY - 200*scale);
        path.lineTo(centerX + 30*scale, centerY + 200*scale);
        ctx.stroke(path);
        ctx.fillText('입면도', centerX - 30, centerY - 210*scale);
        ctx.font = '17px sans-serif';
        ctx.fillText('높이 555m, 123층', centerX - 50, centerY - 190*scale);
        ctx.fillText('단위: m (개념 추정)', centerX - 70, height - 20);
      } else if (stage === 'section') {
        const coreW = 15 * scale;
        const coreH = 400 * scale;
        const shellW = 30 * scale;
        ctx.strokeRect(centerX - shellW/2, centerY - coreH, shellW, coreH);
        ctx.strokeRect(centerX - coreW/2, centerY - coreH + 20*scale, coreW, coreH - 20*scale);
        ctx.fillText('단면도', centerX - 40, centerY - coreH - 10);
        ctx.font = '17px sans-serif';
        ctx.fillText('중심핵 + 외곽골조', centerX - 80, centerY - coreH + 15*scale);
        ctx.fillText('단위: m (개념 추정)', centerX - 70, height - 20);
      } else if (stage === 'site') {
        ctx.fillStyle = '#888';
        ctx.fillRect(centerX - 5*scale, centerY + 10*scale, 10*scale, 10*scale);
        ctx.fillStyle = '#4CAF50';
        ctx.fillRect(centerX - 2*scale, centerY + 5*scale, 4*scale, 4*scale);
        ctx.fillStyle = '#003366';
        ctx.fillRect(centerX - 3*scale, centerY - 3*scale, 6*scale, 6*scale);
        ctx.strokeRect(centerX - 50*scale, centerY - 50*scale, 100*scale, 100*scale);
        ctx.fillText('배치도', centerX - 35, centerY - 55*scale);
        ctx.font = '17px sans-serif';
        ctx.fillText('서울 송파구 롯데월드타워', centerX - 90, centerY - 40*scale);
        ctx.fillText('단위: m (개념 재구성)', centerX - 80, height - 20);
      }

      ctx.restore();
    }

    // 3D 모델 설정
    let scene, modelCamera, modelRenderer, controls;
    function init3D() {
      scene = new THREE.Scene();
      modelCamera = new THREE.PerspectiveCamera(50, 1, 0.1, 2000);
      modelRenderer = new THREE.WebGLRenderer({ canvas: modelCanvas, antialias: true });
      modelRenderer.setClearColor('white');
      controls = new THREE.OrbitControls(modelCamera, modelCanvas);
      controls.enableDamping = true;
      controls.dampingFactor = 0.05;
      controls.screenSpacePanning = false;
      controls.minDistance = 100;
      controls.maxDistance = 1000;
      controls.update();

      updateModel('final');
    }

    function updateModel(stageKey) {
      while(scene.children.length) {
        const obj = scene.children[0];
        if (obj.geometry) obj.geometry.dispose();
        if (obj.material) obj.material.dispose();
        scene.remove(obj);
      }
      const geo = stageGeometries[stageKey] || stageGeometries.final;
      const group = new THREE.Group();
      const segments = 20;
      for (let i = 0; i < segments; i++) {
        const segmentHeight = geo.height / segments;
        const t = i / segments;
        const baseSize = geo.base * (1 - t * 0.8);
        const h = segmentHeight * 0.9;
        const y = (geo.height / 2) - (i + 0.5) * segmentHeight;
        const mesh = new THREE.Mesh(
          new THREE.BoxGeometry(baseSize, h, baseSize),
          new THREE.MeshBasicMaterial({ color: '#003366', wireframe: true })
        );
        mesh.position.y = y;
        group.add(mesh);
      }
      scene.add(group);
    }

    // 초기화
    function init() {
      resize();
      drawDiagram(currentStage);
      init3D();

      // 이벤트 리스너
      stageItems.forEach(item => {
        item.addEventListener('click', () => {
          stageItems.forEach(i => i.classList.remove('active'));
          item.classList.add('active');
          currentStage = item.dataset.stage;
          const text = item.textContent;
          legend.textContent = '도면: ' + text + ' (개념 재구성, 공개 자료 기반)';
          drawDiagram(currentStage);
        });
      });

      rotateBtn.addEventListener('click', () => {
        autoRotate = !autoRotate;
        controls.autoRotate = autoRotate;
        rotateBtn.textContent = autoRotate ? '자동 회전 중지' : '자동 회전 토글';
      });

      resetBtn.addEventListener('click', () => {
        controls.reset();
      });

      zoomInBtn.addEventListener('click', () => {
        modelCamera.position.multiplyScalar(0.9);
        controls.update();
      });

      zoomOutBtn.addEventListener('click', () => {
        modelCamera.position.multiplyScalar(1.1);
        controls.update();
      });

      // 키보드 조작
      window.addEventListener('keydown', (e) => {
        const moveStep = 10;
        const zoomStep = 1.1;
        switch(e.key) {
          case 'ArrowLeft':
            controls.target.x -= moveStep;
            controls.update();
            e.preventDefault();
            break;
          case 'ArrowRight':
            controls.target.x += moveStep;
            controls.update();
            e.preventDefault();
            break;
          case 'ArrowUp':
            controls.target.y += moveStep;
            controls.update();
            e.preventDefault();
            break;
          case 'ArrowDown':
            controls.target.y -= moveStep;
            controls.update();
            e.preventDefault();
            break;
          case '+':
          case '=':
            modelCamera.position.multiplyScalar(1/zoomStep);
            controls.update();
            e.preventDefault();
            break;
          case '-':
            modelCamera.position.multiplyScalar(zoomStep);
            controls.update();
            e.preventDefault();
            break;
          case 'r':
          case 'R':
            controls.reset();
            e.preventDefault();
            break;
        }
      });

      // 애니메이션 루프
      function animate() {
        requestAnimationFrame(animate);
        controls.update();
        modelRenderer.render(scene, modelCamera);
      }
      animate();
    }

    // 초기 실행
    window.addEventListener('load', init);
