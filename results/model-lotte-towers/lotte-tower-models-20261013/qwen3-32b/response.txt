  <title>Lotte Tower Conceptual Model</title>
  <style>
    html, body {
      margin: 0;
      padding: 0;
      overflow: hidden;
      font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;

    #container {

<body>
  <div id="container">
    <div id="ui">
      <label for="stage">Model Stage:</label>
      <select id="stage" style="font-size: 19px;">
        <option value="initial">Initial Concept</option>
        <option value="structural">Structural Frame</option>
        <option value="envelope">Envelope</option>
        <option value="final">Final Form</option>
      </select>
      <button id="resetCamera" style="font-size: 19px;">Reset Camera</button>
    </div>
    <div id="drawings">
      <label for="drawingType">Drawing Type:</label>
      <select id="drawingType" style="font-size: 19px;">
        <option value="site">Site Plan</option>
        <option value="floor">Floor Plan</option>
        <option value="elevation">Elevation</option>
        <option value="section">Section</option>
      </select>
    </div>
    <div class="drawing" id="drawingPanel">
      <h3>Selected Drawing</h3>
      <p class="label">Description will appear here.</p>

  <script>
    // Ensure THREE is available
    if (typeof THREE === 'undefined') {
      console.error('THREE.js runtime not loaded.');
    }

    const container = document.getElementById('container');
    const canvas = document.getElementById('three-canvas');
    const stageSelect = document.getElementById('stage');
    const drawingSelect = document.getElementById('drawingType');
    const resetButton = document.getElementById('resetCamera');
    const drawingPanel = document.getElementById('drawingPanel');

    let camera, scene, renderer, modelGroup, controls, raycaster, mouse;
    let modelInitialized = false;

    // Initialize Three.js
    function init() {
      scene = new THREE.Scene();
      scene.background = new THREE.Color(0x0f0f0f);

      camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
      camera.position.set(0, 200, 300);

      renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.setPixelRatio(window.devicePixelRatio);

      controls = new THREE.OrbitControls(camera, renderer.domElement);
      controls.enableDamping = true;
      controls.dampingFactor = 0.05;
      controls.enablePan = false;

      raycaster = new THREE.Raycaster();
      mouse = new THREE.Vector2();

      window.addEventListener('resize', onWindowResize, false);
      window.addEventListener('keydown', handleKeyControls, false);
      resetButton.addEventListener('click', resetCamera);
      stageSelect.addEventListener('change', updateStage);
      drawingSelect.addEventListener('change', updateDrawing);

      createModel();
      updateDrawing(); // Show initial drawing
      animate();
    }

    function onWindowResize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    }

    function handleKeyControls(event) {
      const key = event.key.toLowerCase();
      if (key === 'a') {
        camera.position.x -= 10;
      } else if (key === 'd') {
        camera.position.x += 10;
      } else if (key === 'w') {
        camera.position.z -= 10;
      } else if (key === 's') {
        camera.position.z += 10;
      } else if (key === 'q') {
        camera.position.y += 10;
      } else if (key === 'e') {
        camera.position.y -= 10;
      }
    }

    function resetCamera() {
      camera.position.set(0, 200, 300);
      controls.update();
    }

    function createModel() {
      modelGroup = new THREE.Group();

      // Base Cylinder (narrowing upward)
      const baseHeight = 100;
      const baseRadiusTop = 20;
      const baseRadiusBottom = 40;
      const baseCylinder = new THREE.Mesh(
        new THREE.CylinderGeometry(baseRadiusBottom, baseRadiusTop, baseHeight, 32),
        new THREE.MeshStandardMaterial({ color: 0x333333, roughness: 0.7, metalness: 0.2, transparent: true, opacity: 0.5 })
      );
      baseCylinder.rotation.x = 0;
      modelGroup.add(baseCylinder);

      // Structural Frame (optional)
      const frame = new THREE.Group();
      const frameMaterial = new THREE.MeshStandardMaterial({ color: 0xaaaaaa, wireframe: true });
      const frameHeight = 100;
      const frameRadius = 20;
      const frameCylinder = new THREE.Mesh(
        new THREE.CylinderGeometry(frameRadius, frameRadius, frameHeight, 32),
        frameMaterial
      );
      frameCylinder.rotation.x = 0;
      frame.add(frameCylinder);
      modelGroup.add(frame);

      // Envelope (optional)
      const envelope = new THREE.Group();
      const envelopeMaterial = new THREE.MeshStandardMaterial({ color: 0x222222, roughness: 0.5 });
      const envelopeRadiusTop = 15;
      const envelopeRadiusBottom = 30;
      const envelopeCylinder = new THREE.Mesh(
        new THREE.CylinderGeometry(envelopeRadiusBottom, envelopeRadiusTop, baseHeight, 32),
        envelopeMaterial
      );
      envelopeCylinder.rotation.x = 0;
      envelope.add(envelopeCylinder);
      modelGroup.add(envelope);

      // Final Form (optional)
      const finalForm = new THREE.Group();
      const finalMaterial = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.3 });
      const finalRadiusTop = 10;
      const finalRadiusBottom = 25;
      const finalCylinder = new THREE.Mesh(
        new THREE.CylinderGeometry(finalRadiusBottom, finalRadiusTop, baseHeight, 32),
        finalMaterial
      );
      finalCylinder.rotation.x = 0;
      finalForm.add(finalCylinder);
      modelGroup.add(finalForm);

      // Lighting
      const ambientLight = new THREE.AmbientLight(0x404040, 0.5);
      scene.add(ambientLight);

      const directionalLight = new THREE.DirectionalLight(0xffffff, 1);
      directionalLight.position.set(100, 100, 100);
      scene.add(directionalLight);

      scene.add(modelGroup);
      modelInitialized = true;
    }

    function updateStage() {
      if (!modelInitialized) return;
      const stage = stageSelect.value;
      const base = modelGroup.children[0];
      const frame = modelGroup.children[1];
      const envelope = modelGroup.children[2];
      const finalForm = modelGroup.children[3];

      base.visible = (stage === 'initial');
      frame.visible = (stage === 'structural');
      envelope.visible = (stage === 'envelope');
      finalForm.visible = (stage === 'final');
    }

    function updateDrawing() {
      const drawing = drawingSelect.value;
      let content = '';
      if (drawing === 'site') {
        content = `
          <h3>Site Plan</h3>
          <p><strong>Lot Size:</strong> Approx. 60,000 m² (estimated)</p>
          <p><strong>Building Footprint:</strong> Circular base with radius 40m</p>
          <p><strong>Height:</strong> 555m (123 floors)</p>
          <p><strong>Design Concept:</strong> Central tower with surrounding amenities</p>
          <p class="label">* Approximate values based on public design references</p>
        `;
      } else if (drawing === 'floor') {
        content = `
          <h3>Floor Plan</h3>
          <p><strong>Typical Floor:</strong> Circular, narrowing upward</p>
          <p><strong>Ground Floor Radius:</strong> 40m</p>
          <p><strong>Top Floor Radius:</strong> 10m</p>
          <p><strong>Core:</strong> Central elevator and service shafts</p>
          <p class="label">* Dimensions are conceptual and not from official construction drawings</p>
        `;
      } else if (drawing === 'elevation') {
        content = `
          <h3>Elevation</h3>
          <p><strong>Height:</strong> 555m</p>
          <p><strong>Form:</strong> Tapering upward, inspired by Korean ceramics</p>
          <p><strong>Key Features:</strong> Skybridge at 330m, spire at 555m</p>
          <p class="label">* Design references from KPF and LERA</p>
        `;
      } else if (drawing === 'section') {
        content = `
          <h3>Section</h3>
          <p><strong>Vertical Taper:</strong> Radius reduces from 40m to 10m</p>
          <p><strong>Structural System:</strong> Central core + outrigger system</p>
          <p><strong>Key Levels:</strong> Skybridge at 330m, observation deck at 440m</p>
          <p class="label">* Sectional details are conceptual and not from official construction drawings</p>
        `;
      }
      drawingPanel.innerHTML = content;
    }

    function animate() {
      requestAnimationFrame(animate);
      controls.update();
      renderer.render(scene, camera);
    }

    init();
  </script>
</body>
</html>
