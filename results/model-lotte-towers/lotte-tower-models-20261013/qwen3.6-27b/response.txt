  function buildTower(progress) {
    while(towerGroup.children.length) {
      const c = towerGroup.children[0];
      towerGroup.remove(c);
      if(c.geometry) c.geometry.dispose();
      if(c.material) {
        if(Array.isArray(c.material)) c.material.forEach(m=>m.dispose());
        else c.material.dispose();
      }
    }

    const stage = STAGES[Math.round(progress * (STAGES.length - 1))];
    const idx = Math.round(progress * (STAGES.length - 1));
    const nextIdx = Math.min(idx + 1, STAGES.length - 1);
    const t = (progress * (STAGES.length - 1)) - idx;

    const baseW = 20;
    const totalH = 100;
    const h = totalH * stage.height;
    const taper = stage.taper;
    const segs = stage.segments;

    const shape = new THREE.Shape();
    const r = baseW / 2;
    const topR = r * (1 - taper);
    const steps = Math.max(segs, 4);

    for(let i = 0; i <= steps; i++) {
      const a = (i / steps) * Math.PI * 2;
      const cosA = Math.cos(a);
      const sinA = Math.sin(a);
      const px = cosA * r;
      const py = sinA * r;
      if(i === 0) shape.moveTo(px, py);
      else shape.lineTo(px, py);
    }
    shape.closePath();

      steps: 1,
      depth: h,
      bevelEnabled: true,
      bevelThickness: taper * 3,
      bevelSize: taper * 2,

    };
