    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const addWaveBtn = document.getElementById('addWave');
        const clearWavesBtn = document.getElementById('clearWaves');
        const toggleAnimationBtn = document.getElementById('toggleAnimation');
        const waveCountEl = document.getElementById('waveCount');

        let waves = [];
        let animationId;
        let isPaused = false;
        let time = 0;

        // 캔버스 크기 설정
        function resizeCanvas() {
            const rect = canvas.getBoundingClientRect();
            canvas.width = rect.width * window.devicePixelRatio;
            canvas.height = rect.height * window.devicePixelRatio;
            ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
        }

        resizeCanvas();
        window.addEventListener('resize', resizeCanvas);

        // 파동 클래스
        class Wave {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = 0;
                this.maxRadius = Math.max(canvas.width, canvas.height) / window.devicePixelRatio;
                this.speed = 2;
                this.opacity = 1;
                this.hue = Math.random() * 60 + 180; // 청록색 계열
                this.frequency = Math.random() * 0.02 + 0.01;
            }

            update() {
                this.radius += this.speed;
                this.opacity = Math.max(0, 1 - (this.radius / this.maxRadius));
            }

            draw() {
                if (this.opacity <= 0) return;

                const rect = canvas.getBoundingClientRect();
                
                // 메인 원
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.strokeStyle = `hsla(${this.hue}, 70%, 60%, ${this.opacity * 0.6})`;
                ctx.lineWidth = 2;
                ctx.stroke();

                // 내부 글로우
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius * 0.95, 0, Math.PI * 2);
                ctx.strokeStyle = `hsla(${this.hue}, 80%, 70%, ${this.opacity * 0.3})`;
                ctx.lineWidth = 1;
                ctx.stroke();

                // 파동 패턴
                const segments = 60;
                for (let i = 0; i < segments; i++) {
                    const angle = (Math.PI * 2 / segments) * i;
                    const wave = Math.sin(angle * 3 + time * this.frequency) * 10;
                    const x1 = this.x + Math.cos(angle) * (this.radius + wave);
                    const y1 = this.y + Math.sin(angle) * (this.radius + wave);
                    
                    if (i === 0) {
                        ctx.beginPath();
                        ctx.moveTo(x1, y1);
                    } else {
                        ctx.lineTo(x1, y1);
                    }
                }
                ctx.closePath();
                ctx.strokeStyle = `hsla(${this.hue}, 60%, 50%, ${this.opacity * 0.4})`;
                ctx.lineWidth = 1;
                ctx.stroke();
            }

            isDead() {
                return this.opacity <= 0;
            }
        }

        // 파동 추가
        function addWave(x, y) {
            const rect = canvas.getBoundingClientRect();
            const scaleX = rect.width / window.devicePixelRatio;
            const scaleY = rect.height / window.devicePixelRatio;
            waves.push(new Wave(x || scaleX / 2, y || scaleY / 2));
            updateWaveCount();
        }

        // 파동 개수 업데이트
        function updateWaveCount() {
            waveCountEl.textContent = `파동: ${waves.length}`;
        }

        // 애니메이션 루프
        function animate() {
            if (!isPaused) {
                const rect = canvas.getBoundingClientRect();
                ctx.clearRect(0, 0, rect.width, rect.height);

                // 배경 그라디언트
                const gradient = ctx.createRadialGradient(
                    rect.width / 2, rect.height / 2, 0,
                    rect.width / 2, rect.height / 2, rect.width / 2
                );
                gradient.addColorStop(0, 'rgba(26, 26, 46, 0.1)');
                gradient.addColorStop(1, 'rgba(10, 10, 15, 0.1)');
                ctx.fillStyle = gradient;
                ctx.fillRect(0, 0, rect.width, rect.height);

                // 파동 간 상호작용 시각화
                for (let i = 0; i < waves.length; i++) {
                    for (let j = i + 1; j < waves.length; j++) {
                        const dx = waves[i].x - waves[j].x;
                        const dy = waves[i].y - waves[j].y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        const radiusDiff = Math.abs(waves[i].radius - waves[j].radius);
                        
                        if (radiusDiff < 20 && distance < 300) {
                            ctx.beginPath();
                            ctx.moveTo(waves[i].x, waves[i].y);
                            ctx.lineTo(waves[j].x, waves[j].y);
                            const opacity = (1 - distance / 300) * waves[i].opacity * waves[j].opacity;
                            ctx.strokeStyle = `rgba(100, 200, 255, ${opacity * 0.3})`;
                            ctx.lineWidth = 1;
                            ctx.stroke();
                        }
                    }
                }

                // 파동 업데이트 및 그리기
                waves.forEach(wave => {
                    wave.update();
                    wave.draw();
                });

                // 죽은 파동 제거
                waves = waves.filter(wave => !wave.isDead());
                updateWaveCount();

                time += 1;
            }

            animationId = requestAnimationFrame(animate);
        }

        // 이벤트 리스너
        canvas.addEventListener('click', (e) => {
            const rect = canvas.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            addWave(x, y);
        });

        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            const rect = canvas.getBoundingClientRect();
            const touch = e.touches[0];
            const x = touch.clientX - rect.left;
            const y = touch.clientY - rect.top;
            addWave(x, y);
        });

        addWaveBtn.addEventListener('click', () => addWave());

        clearWavesBtn.addEventListener('click', () => {
            waves = [];
            updateWaveCount();
        });

        toggleAnimationBtn.addEventListener('click', () => {
            isPaused = !isPaused;
            toggleAnimationBtn.textContent = isPaused ? '재생 (P)' : '일시정지 (P)';
        });

        // 키보드 컨트롤
        document.addEventListener('keydown', (e) => {
            if (e.code === 'Space') {
                e.preventDefault();
                addWave();
            } else if (e.code === 'KeyC') {
                waves = [];
                updateWaveCount();
            } else if (e.code === 'KeyP') {
                isPaused = !isPaused;
                toggleAnimationBtn.textContent = isPaused ? '재생 (P)' : '일시정지 (P)';
            }
        });

        // 초기 파동
        addWave();
        setTimeout(() => addWave(), 500);

        // 애니메이션 시작
        animate();

        // prefers-reduced-motion 감지
        if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
            isPaused = true;
            toggleAnimationBtn.textContent = '재생 (P)';
        }
    </script>
