<script>
(() => {
  "use strict";

  const canvas = document.getElementById("stage");
  const ctx = canvas.getContext("2d");
  const reduced = matchMedia("(prefers-reduced-motion: reduce)").matches;
  const coarse = matchMedia("(pointer: coarse)").matches;

  if (coarse) {
    document.getElementById("touchHint").textContent = "손가락을 대면 언어가 모여듭니다 · 탭으로 파동";
  } else {
    document.getElementById("keyHint").style.display = "inline";
  }

  // ── 토큰 소스: 모델이 다루는 언어의 파편 ──
  const GLYPHS = [
    "가","나","다","라","마","바","사","아","자","차","카","타","파","하",
    "언","어","생","각","흐","름","빛","결","질","서","모","딘","샘",
    "if","for","let","fn","λ","Σ","→","∴","∞","0","1",
    "token","logit","attn","flow","mind","gl","m","5","3"
  ];

  const THEMES = [
    { bg: "#07080d", a: [125,240,212], b: [183,166,255], trail: 0.10 },
    { bg: "#0d0708", a: [255,138,128], b: [255,196,110], trail: 0.10 },
    { bg: "#060a0d", a: [120,190,255], b: [140,255,200], trail: 0.10 },
  ];
  let themeIdx = 0;

  let W = 0, H = 0, DPR = 1;
  let tokens = [];
  let paused = false;
  let t = 0;
  const pointer = { x: -9999, y: -9999, active: false };
  const waves = []; // 클릭 파동

  function resize() {
    DPR = Math.min(devicePixelRatio || 1, 2);
    W = innerWidth; H = innerHeight;
    canvas.width = W * DPR; canvas.height = H * DPR;
    canvas.style.width = W + "px"; canvas.style.height = H + "px";
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
  }

  function spawn() {
    tokens = [];
    const density = Math.min(Math.floor((W * H) / 9000), 260);
    for (let i = 0; i < density; i++) {
      tokens.push({
        x: Math.random() * W,
        y: Math.random() * H,
        vx: 0, vy: 0,
        g: GLYPHS[(Math.random() * GLYPHS.length) | 0],
        s: 11 + Math.random() * 12,
        ph: Math.random() * Math.PI * 2,
        hue: Math.random(), // 0=a, 1=b
      });
    }
    document.getElementById("tokCount").textContent = tokens.length;
  }

  // 유사 노이즈 흐름장 (sin 조합, 외부 리소스 불필요)
  function flow(x, y, time) {
    const s = 0.0022;
    const a = Math.sin(x * s + time * 0.4) + Math.cos(y * s * 1.3 - time * 0.3);
    const b = Math.sin((x + y) * s * 0.8 + time * 0.25);
    return (a + b) * 1.1;
  }

  function frame() {
    const th = THEMES[themeIdx];
    const speed = reduced ? 0.12 : 1;

    // 잔상
    ctx.globalCompositeOperation = "source-over";
    ctx.fillStyle = `rgba(7,8,13,${th.trail * 10})`;
    // 테마 배경색 기반 잔상
    ctx.fillStyle = hexToRGBA(th.bg, 1 - (1 - th.trail));
    ctx.fillRect(0, 0, W, H);
    ctx.globalCompositeOperation = "lighter";

    t += 0.008 * speed;

    // 파동 갱신
    for (let i = waves.length - 1; i >= 0; i--) {
      waves[i].r += 6 * speed;
      waves[i].life -= 0.016 * speed;
      if (waves[i].life <= 0) waves.splice(i, 1);
    }

    for (const p of tokens) {
      if (!paused) {
        const ang = flow(p.x, p.y, t);
        p.vx += Math.cos(ang) * 0.05 * speed;
        p.vy += Math.sin(ang) * 0.05 * speed;

        // 포인터 인력: 언어가 손길에 모여든다
        if (pointer.active) {
          const dx = pointer.x - p.x, dy = pointer.y - p.y;
          const d2 = dx * dx + dy * dy;
          const R = Math.min(W, H) * 0.35;
          if (d2 < R * R && d2 > 400) {
            const d = Math.sqrt(d2);
            const f = (1 - d / R) * 0.35;
            p.vx += (dx / d) * f;
            p.vy += (dy / d) * f;
          }
        }

        // 파동 밀어내기
        for (const w of waves) {
          const dx = p.x - w.x, dy = p.y - w.y;
          const d = Math.hypot(dx, dy);
          if (Math.abs(d - w.r) < 40) {
            const f = w.life * 1.4 / (d || 1);
            p.vx += dx * f * 0.06;
            p.vy += dy * f * 0.06;
          }
        }

        p.vx *= 0.94; p.vy *= 0.94;
        p.x += p.vx * speed; p.y += p.vy * speed;

        // 랩어라운드
        if (p.x < -30) p.x = W + 30; if (p.x > W + 30) p.x = -30;
        if (p.y < -30) p.y = H + 30; if (p.y > H + 30) p.y = -30;
      }

      // 포인터 근처 밝기/크기
      let glow = 0.25;
      let scale = 1;
      if (pointer.active) {
        const d = Math.hypot(pointer.x - p.x, pointer.y - p.y);
        const R = Math.min(W, H) * 0.3;
        if (d < R) {
          glow = 0.25 + (1 - d / R) * 0.75;
          scale = 1 + (1 - d / R) * 0.6;
        }
      }

      const c = mixRGB(th.a, th.b, p.hue);
      const breathe = 0.85 + 0.15 * Math.sin(t * 3 + p.ph);
      ctx.font = `600 ${p.s * scale}px "Pretendard","Noto Sans KR",sans-serif`;
      ctx.fillStyle = `rgba(${c[0]},${c[1]},${c[2]},${(glow * breathe).toFixed(3)})`;
      ctx.fillText(p.g, p.x, p.y);
    }

    // 파동 링 렌더
    ctx.globalCompositeOperation = "source-over";
    for (const w of waves) {
      const c = mixRGB(th.a, th.b, 0.5);
      ctx.strokeStyle = `rgba(${c[0]},${c[1]},${c[2]},${(w.life * 0.5).toFixed(3)})`;
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.arc(w.x, w.y, w.r, 0, Math.PI * 2);
      ctx.stroke();
    }

    requestAnimationFrame(frame);
  }

  function hexToRGBA(hex, a) {
    const n = parseInt(hex.slice(1), 16);
    return `rgba(${(n>>16)&255},${(n>>8)&255},${n&255},${a})`;
  }
  function mixRGB(a, b, k) {
    return [
      Math.round(a[0] + (b[0]-a[0])*k),
      Math.round(a[1] + (b[1]-a[1])*k),
      Math.round(a[2] + (b[2]-a[2])*k),
    ];
  }

  // ── 입력 ──
  function setPointer(x, y) {
    pointer.x = x; pointer.y = y; pointer.active = true;
  }
  canvas.addEventListener("pointermove", e => setPointer(e.clientX, e.clientY));
  canvas.addEventListener("pointerdown", e => {
    setPointer(e.clientX, e.clientY);
    waves.push({ x: e.clientX, y: e.clientY, r: 0, life: 1 });
    announce("파동이 퍼졌습니다.");
  });
  canvas.addEventListener("pointerleave", () => { pointer.active = false; });
  canvas.addEventListener("pointerup", () => { if (coarse) pointer.active = false; });

  // ── 컨트롤 ──
  const btnPause = document.getElementById("btnPause");
  const btnTheme = document.getElementById("btnTheme");
  const stateLine = document.getElementById("stateLine");
  const sr = document.getElementById("sr");

  function announce(msg) { sr.textContent = msg; }

  function togglePause() {
    paused = !paused;
    btnPause.setAttribute("aria-pressed", String(paused));
    btnPause.textContent = paused ? "재생 (Space)" : "일시정지 (Space)";
    stateLine.innerHTML = `상태: <b>${paused ? "정지" : "흐름"}</b>`;
    announce(paused ? "일시정지되었습니다." : "다시 흐릅니다.");
  }
  function cycleTheme() {
    themeIdx = (themeIdx + 1) % THEMES.length;
    btnTheme.setAttribute("aria-pressed", String(themeIdx > 0));
    announce(`테마 ${themeIdx + 1}번으로 바뀌었습니다.`);
  }
  function regen() {
    spawn();
    announce("언어 파편을 다시 생성했습니다.");
  }

  btnPause.addEventListener("click", togglePause);
  btnTheme.addEventListener("click", cycleTheme);
  document.getElementById("btnReset").addEventListener("click", regen);

  addEventListener("keydown", e => {
    if (e.target.tagName === "BUTTON" && e.key === " " ) { e.preventDefault(); }
    if (e.key === " ") { e.preventDefault(); togglePause(); }
    else if (e.key === "t" || e.key === "T") cycleTheme();
    else if (e.key === "r" || e.key === "R") regen();
  });

  addEventListener("resize", () => { resize(); });

  resize();
  spawn();
  frame();
})();
</script>
</body>
</html>

**작품 개념 — "언어의 응결"**

GLM-5.3은 흩어진 언어 파편에 질서와 의미를 부여하는 존재입니다. 이 작품은 그 과정을 세 겹의 상호작용으로 표현합니다:

1. **흐름장** — 한글 자모, 코드 키워드, 수학 기호가 섞인 토큰들이 보이지 않는 흐름장을 따라 소용돌이칩니다. 외부 노이즈 라이브러리 없이 sin 조합으로 유기적인 흐름을 만들었습니다.
2. **손길의 인력** — 커서/손가락 근처의 토큰이 끌려와 밝아지고 커지며, "주의가 언어를 응결시킨다"는 메타포를 담았습니다.
3. **파동** — 클릭/탭으로 퍼지는 링이 토큰을 밀어냅니다. 생각의 충격이 언어를 재배치하는 순간입니다.

**접근성**: `prefers-reduced-motion` 시 흐름 속도를 대폭 줄이고, 키보드(Space/T/R)로 전 기능 조작 가능, `aria-live`로 상태 변경을 스크린리더에 알리며, 명확한 포커스 링과 고대비 텍스트를 제공합니다. 모바일에서는 터치 인력과 탭 파동으로 동일한 경험을 제공합니다.
