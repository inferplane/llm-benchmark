<script>
(() => {
  const canvas=document.getElementById('sky'), ctx=canvas.getContext('2d');
  const density=document.getElementById('density'), drift=document.getElementById('drift');
  const dOut=document.getElementById('densityOut'), mOut=document.getElementById('driftOut');
  const status=document.getElementById('status');
  let w,h,dpr,paused=false,stars=[],ripples=[],t=0;

  function resize(){
    dpr=Math.min(devicePixelRatio||1,2); w=innerWidth; h=innerHeight;
    canvas.width=w*dpr; canvas.height=h*dpr; ctx.setTransform(dpr,0,0,dpr,0,0);
    makeStars();
  }
  function makeStars(){
    const n=+density.value; stars=Array.from({length:n},()=>({
      x:Math.random()*w,y:Math.random()*h,r:.35+Math.random()*1.8,
      p:Math.random()*Math.PI*2, a:.25+Math.random()*.7,
      hue:Math.random()<.5?190:275
    }));
  }
  function updateLabels(){
    dOut.value=density.value;mOut.value=drift.value;
    status.textContent=`별 ${density.value}개 · 움직임 ${drift.value}%`;
    makeStars();
  }
  function draw(){
    if(!paused)t+=.008*(+drift.value/42);
    ctx.clearRect(0,0,w,h);
    const cx=w*.73+Math.sin(t*.42)*w*.035, cy=h*.38+Math.cos(t*.3)*h*.025;
    const g=ctx.createRadialGradient(cx,cy,5,cx,cy,Math.max(w,h)*.34);
    g.addColorStop(0,'rgba(238,220,255,.95)');g.addColorStop(.08,'rgba(166,202,255,.34)');
    g.addColorStop(1,'rgba(100,120,255,0)');
    ctx.fillStyle=g;ctx.fillRect(0,0,w,h);
    for(const s of stars){
      const x=s.x+Math.sin(t*.45+s.p)*(+drift.value/100)*12;
      const y=s.y+Math.cos(t*.34+s.p)*(+drift.value/100)*8;
      ctx.globalAlpha=s.a*(.65+.35*Math.sin(t*1.7+s.p));
      ctx.fillStyle=`hsl(${s.hue} 90% 82%)`;ctx.beginPath();ctx.arc(x,y,s.r,0,Math.PI*2);ctx.fill();
    }
    ctx.globalAlpha=1;
    ctx.strokeStyle='rgba(176,220,255,.4)';ctx.lineWidth=1;
    for(const r of ripples){r.age+=.012;ctx.globalAlpha=1-r.age;
      ctx.beginPath();ctx.arc(r.x,r.y,r.age*130,0,Math.PI*2);ctx.stroke()}
    ripples=ripples.filter(r=>r.age<1);
    requestAnimationFrame(draw);
  }
  function plant(x=w*.5,y=h*.5){
    ripples.push({x,y,age:0});
    for(let i=0;i<8;i++)stars.push({x:x+(Math.random()-.5)*35,y:y+(Math.random()-.5)*35,r:.5+Math.random()*2,p:Math.random()*6,a:.8,hue:200+Math.random()*80});
  }
  addEventListener('resize',resize);
  density.addEventListener('input',updateLabels);drift.addEventListener('input',()=>{mOut.value=drift.value;status.textContent=`별 ${density.value}개 · 움직임 ${drift.value}%`});
  document.getElementById('seed').onclick=()=>plant();
  document.getElementById('clear').onclick=()=>{ripples=[];stars=[];status.textContent='고요한 빈 공간 · 다시 심을 수 있습니다'};
  canvas.addEventListener('pointerdown',e=>plant(e.clientX,e.clientY));
  addEventListener('keydown',e=>{
    if(e.key==='Enter')plant();
    if(e.key===' '){e.preventDefault();paused=!paused;status.textContent=paused?'일시정지됨 · Space로 재생':'재생 중';}
    if(e.key==='Escape'){ripples=[];stars=[];status.textContent='고요한 빈 공간 · 다시 심을 수 있습니다'}
  });
  resize();draw();
})();
