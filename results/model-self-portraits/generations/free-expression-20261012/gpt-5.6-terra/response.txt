<script>
(() => {
  const canvas = document.getElementById('terra');
  const ctx = canvas.getContext('2d', { alpha: false });
  const depthEl = document.getElementById('depth');
  const stateEl = document.getElementById('state');
  const announce = document.getElementById('announce');
  const pauseBtn = document.getElementById('pause');
  const motionBtn = document.getElementById('motion');
  const renewBtn = document.getElementById('renew');

  let w, h, dpr, time = 0, running = true;
  let reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let ridges = [], wells = [], seed = Math.random() * 9999;

  const palette = [
    ['#101a13', '#193123', '#31553a'],
    ['#132218', '#285038', '#63783e'],
    ['#182016', '#4e562b', '#a3833d'],
    ['#241d16', '#693b26', '#c87942'],
    ['#281b18', '#743b39', '#d26d51'],
    ['#201c24', '#493c62', '#a77aac'],
    ['#13212a', '#27606a', '#a3bca4']
  ];

  function rand(n) {
    const x = Math.sin(n * 12.9898 + seed * 78.233) * 43758.5453;
    return x - Math.floor(x);
  }

  function resize() {
    dpr = Math.min(devicePixelRatio || 1, 2);
    w = innerWidth; h = innerHeight;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    build();
  }

  function build() {
    const count = Math.max(12, Math.min(25, Math.round(h / 42)));
    ridges = Array.from({length: count}, (_, i) => ({
      y: h * (.18 + i / (count - 1) * .78),
      amp: 15 + rand(i + 30) * 48,
      freq: .005 + rand(i + 60) * .012,
      phase: rand(i + 90) * 20,
      drift: .12 + rand(i + 120) * .55,
      color: palette[i % palette.length],
      thick: 1 + rand(i + 150) * 2.5
    }));
    wells = [];
  }

  function curveY(r, x, t) {
    let y = r.y;
    y += Math.sin(x * r.freq + r.phase + t * r.drift) * r.amp;
    y += Math.sin(x * r.freq * 2.41 - r.phase + t * .21) * r.amp * .22;
    y += Math.sin(x * .0021 + r.phase * 5) * r.amp * .45;
    for (const q of wells) {
      const dx = x - q.x;
      const dist = Math.abs(dx) / q.radius;
      if (dist < 1) y += Math.cos(dist * Math.PI / 2) * q.force * Math.sin((dist * 7) - t * 2);
    }
    return y;
  }

  function draw() {
    const t = reduced ? 0 : time;
    const bg = ctx.createLinearGradient(0, 0, 0, h);
    bg.addColorStop(0, '#0c1711');
    bg.addColorStop(.48, '#142519');
    bg.addColorStop(1, '#090e0b');
    ctx.fillStyle = bg;
    ctx.fillRect(0, 0, w, h);

    // atmospheric sun / core
    const glow = ctx.createRadialGradient(w * .72, h * .25, 0, w * .72, h * .25, Math.max(w, h) * .55);
    glow.addColorStop(0, 'rgba(187, 203, 103, .16)');
    glow.addColorStop(.45, 'rgba(93, 129, 70, .05)');
    glow.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = glow; ctx.fillRect(0, 0, w, h);

    ridges.forEach((r, i) => {
      const next = ridges[i + 1];
      ctx.beginPath();
      ctx.moveTo(-4, curveY(r, -4, t));
      for (let x = 0; x <= w + 8; x += 8) ctx.lineTo(x, curveY(r, x, t));
      if (next) {
        for (let x = w + 8; x >= -4; x -= 8) ctx.lineTo(x, curveY(next, x, t));
      } else {
        ctx.lineTo(w + 8, h + 4); ctx.lineTo(-4, h + 4);
      }
      const fill = ctx.createLinearGradient(0, r.y - 60, 0, r.y + 100);
      fill.addColorStop(0, r.color[0]);
      fill.addColorStop(.5, r.color[1]);
      fill.addColorStop(1, r.color[2]);
      ctx.globalAlpha = .72;
      ctx.fillStyle = fill;
      ctx.fill();

      ctx.beginPath();
      ctx.moveTo(-4, curveY(r, -4, t));
      for (let x = 0; x <= w + 8; x += 5) ctx.lineTo(x, curveY(r, x, t));
      ctx.globalAlpha = .35 + (i % 3) * .08;
      ctx.strokeStyle = i % 2 ? '#d9e5a3' : '#f0ae6b';
      ctx.lineWidth = r.thick;
      ctx.stroke();
    });

    // root-like fine data lines
    ctx.globalAlpha = .12;
    ctx.strokeStyle = '#d9f0b8';
    ctx.lineWidth = 1;
    for (let j = 0; j < 8; j++) {
      ctx.beginPath();
      const x0 = (j / 7) * w;
      ctx.moveTo(x0, 0);
      for (let y = 0; y < h; y += 12) {
        const x = x0 + Math.sin(y * .018 + j * 7 + t) * (13 + j * 3);
        ctx.lineTo(x, y);
      }
      ctx.stroke();
    }
    ctx.globalAlpha = 1;

    wells = wells.filter(q => {
      q.life -= reduced ? .007 : .012;
      q.force *= .994;
      return q.life > 0;
    });

    const depth = Math.round((Math.sin(t * .16) * .5 + .5) * 936);
    depthEl.textContent = `DEPTH ${String(depth).padStart(3, '0')} m`;
  }

  function frame() {
    if (running) {
      time += reduced ? .003 : .012;
      draw();
    }
    requestAnimationFrame(frame);
  }

  function plant(x, y) {
    wells.push({ x, y, radius: Math.max(100, w * .17), force: 24 + Math.random() * 35, life: 1 });
    if (wells.length > 8) wells.shift();
    stateEl.textContent = '새로운 퇴적 · 파문 확산 중';
    announce.textContent = '새로운 지층 파문을 만들었습니다.';
    draw();
  }

  canvas.addEventListener('pointerdown', e => {
    canvas.setPointerCapture?.(e.pointerId);
    plant(e.clientX, e.clientY);
  });
  canvas.addEventListener('pointermove', e => {
    if (e.buttons && !reduced) plant(e.clientX, e.clientY);
  });

  function togglePause() {
    running = !running;
    pauseBtn.setAttribute('aria-pressed', String(!running));
    pauseBtn.textContent = running ? '일시정지' : '재생';
    stateEl.textContent = running ? '관측 중 · 느린 호흡' : '시간 정지 · 지층 보존';
    announce.textContent = running ? '작품을 재생합니다.' : '작품을 일시정지했습니다.';
    if (running) draw();
  }
  function toggleMotion() {
    reduced = !reduced;
    motionBtn.setAttribute('aria-pressed', String(reduced));
    motionBtn.textContent = reduced ? '움직임 최소화됨' : '움직임 감소';
    stateEl.textContent = reduced ? '고요한 단면 · 움직임 최소화' : '관측 중 · 느린 호흡';
    announce.textContent = reduced ? '움직임을 줄였습니다.' : '일반 움직임을 복원했습니다.';
    draw();
  }
  function renew() {
    seed = Math.random() * 9999;
    build(); draw();
    stateEl.textContent = '새로운 지질 연대 생성';
    announce.textContent = '새 지형을 생성했습니다.';
  }

  pauseBtn.addEventListener('click', togglePause);
  motionBtn.addEventListener('click', toggleMotion);
  renewBtn.addEventListener('click', renew);
  addEventListener('keydown', e => {
    if (e.target.matches('button')) return;
    if (e.code === 'Space') { e.preventDefault(); togglePause(); }
    if (e.key.toLowerCase() === 'm') toggleMotion();
    if (e.key.toLowerCase() === 'r') renew();
  });
  addEventListener('resize', resize);

  motionBtn.setAttribute('aria-pressed', String(reduced));
  if (reduced) motionBtn.textContent = '움직임 최소화됨';
  resize();
  draw();
  requestAnimationFrame(frame);
})();
</script>
