<script>
(() => {
  const canvas = document.getElementById('landscape');
  const art = document.getElementById('art');
  const ctx = canvas.getContext('2d');
  const status = document.getElementById('status');
  const motionButton = document.getElementById('motion');
  const reduced = window.matchMedia('(prefers-reduced-motion: reduce)');
  let paused = reduced.matches;
  let width = 0, height = 0, time = 0, last = 0, frame = 0;
  let points = [];

  function randomPoint() {
    return {
      x: .1 + Math.random() * .8,
      y: .16 + Math.random() * .68,
      strength: .7 + Math.random() * .65,
      phase: Math.random() * Math.PI * 2
    };
  }

  function reset() {
    points = Array.from({length: 6}, randomPoint);
    draw();
  }

  function resize() {
    const rect = art.getBoundingClientRect();
    width = rect.width;
    height = rect.height;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.max(1, Math.round(width * dpr));
    canvas.height = Math.max(1, Math.round(height * dpr));
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    draw();
  }

  function draw() {
    if (!width || !height) return;
    const bg = ctx.createLinearGradient(0, 0, width, height);
    bg.addColorStop(0, '#17271d');
    bg.addColorStop(.55, '#101b16');
    bg.addColorStop(1, '#1b2a20');
    ctx.fillStyle = bg;
    ctx.fillRect(0, 0, width, height);

    const glow = ctx.createRadialGradient(width * .53, height * .48, 0, width * .53, height * .48, width * .62);
    glow.addColorStop(0, 'rgba(109,145,77,.19)');
    glow.addColorStop(1, 'rgba(10,19,14,0)');
    ctx.fillStyle = glow;
    ctx.fillRect(0, 0, width, height);

    const spacing = Math.max(10, height / 48);
    const step = Math.max(5, width / 150);
    for (let base = -height * .35; base < height * 1.35; base += spacing) {
      ctx.beginPath();
      for (let x = -step; x <= width + step; x += step) {
        let y = base + Math.sin(x / width * 5 + base / height * 4 + time * .12) * height * .025;
        for (const p of points) {
          const px = p.x * width;
          const py = p.y * height;
          const dx = (x - px) / (width * .19);
          const dy = (base - py) / (height * .24);
          const influence = Math.exp(-(dx * dx + dy * dy) * 1.35);
          y -= influence * height * .13 * p.strength * (1 + .08 * Math.sin(time * .5 + p.phase));
        }
        if (x === -step) ctx.moveTo(x, y); else ctx.lineTo(x, y);
      }
      ctx.strokeStyle = 'rgba(190, 223, 151, .34)';
      ctx.lineWidth = .8;
      ctx.stroke();
    }

    for (const p of points) {
      const x = p.x * width, y = p.y * height;
      const radius = Math.max(16, Math.min(width, height) * .05);
      const halo = ctx.createRadialGradient(x, y, 0, x, y, radius);
      halo.addColorStop(0, 'rgba(213,255,160,.34)');
      halo.addColorStop(1, 'rgba(213,255,160,0)');
      ctx.fillStyle = halo;
      ctx.beginPath();
      ctx.arc(x, y, radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.fillStyle = '#e1ffb0';
      ctx.beginPath();
      ctx.arc(x, y, 2.5, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function tick(now) {
    if (paused) { frame = 0; return; }
    if (last) time += Math.min((now - last) / 1000, .05);
    last = now;
    draw();
    frame = requestAnimationFrame(tick);
  }

  function updateMotion() {
    motionButton.textContent = paused ? '움직임 재생' : '움직임 일시정지';
    motionButton.setAttribute('aria-pressed', String(paused));
    last = 0;
    if (!paused && !frame) frame = requestAnimationFrame(tick);
    if (paused && frame) { cancelAnimationFrame(frame); frame = 0; }
    draw();
  }

  function addPoint(x, y) {
    const p = randomPoint();
    p.x = Math.max(.02, Math.min(.98, x));
    p.y = Math.max(.02, Math.min(.98, y));
    points.push(p);
    if (points.length > 14) points.shift();
    draw();
    status.textContent = '새로운 생각이 지형에 놓였습니다.';
  }

  art.addEventListener('pointerdown', event => {
    const rect = art.getBoundingClientRect();
    addPoint((event.clientX - rect.left) / rect.width, (event.clientY - rect.top) / rect.height);
  });
  document.getElementById('seed').addEventListener('click', () => {
    const p = randomPoint();
    addPoint(p.x, p.y);
  });
  document.getElementById('reset').addEventListener('click', () => {
    reset();
    status.textContent = '지형을 새로 그렸습니다.';
  });
  motionButton.addEventListener('click', () => {
    paused = !paused;
    updateMotion();
    status.textContent = paused ? '움직임을 일시정지했습니다.' : '움직임을 재생합니다.';
  });
  reduced.addEventListener?.('change', event => {
    paused = event.matches;
    updateMotion();
  });
  window.addEventListener('resize', resize);
  reset();
  resize();
  updateMotion();
})();
</script>
