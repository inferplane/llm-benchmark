<script>
/* ---- Canvas & particle system ---- */
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
let width, height;
function resize(){
  width = canvas.width = window.innerWidth;
  height = canvas.height = window.innerHeight;
}
window.addEventListener('resize', resize);
resize();

/* ---- Particle definition ---- */
class Particle{
  constructor(){
    this.reset();
  }
  reset(){
    this.x = Math.random()*width;
    this.y = Math.random()*height;
    this.vx = (Math.random()-0.5)*1.5;
    this.vy = (Math.random()-0.5)*1.5;
    this.size = Math.random()*3+1;
    this.baseSize = this.size;
  }
  update(){
    this.x += this.vx;
    this.y += this.vy;
    // bounce
    if(this.x<0||this.x>width) this.vx*=-1;
    if(this.y<0||this.y>height) this.vy*=-1;
  }
  draw(){
    ctx.beginPath();
    ctx.arc(this.x,this.y,this.size,0,Math.PI*2);
    ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--particle-color') || '#0066ff';
    ctx.fill();
  }
}

/* ---- Create particles ---- */
let particles = [];
const PARTICLE_COUNT = Math.min(200, Math.max(80, Math.floor((width*height)/15000)));
for(let i=0;i<PARTICLE_COUNT;i++) particles.push(new Particle());

/* ---- Interaction ---- */
let pointer = {x: null, y: null, active:false};

function setPointer(e){
  const rect = canvas.getBoundingClientRect();
  const x = (e.touches?e.touches[0].clientX:e.clientX)-rect.left;
  const y = (e.touches?e.touches[0].clientY:e.clientY)-rect.top;
  pointer.x = x; pointer.y = y; pointer.active = true;
}
function clearPointer(){ pointer.active = false; }

canvas.addEventListener('mousemove', setPointer);
canvas.addEventListener('touchmove', setPointer, {passive:true});
canvas.addEventListener('mouseleave', clearPointer);
canvas.addEventListener('touchend', clearPointer);
canvas.addEventListener('touchcancel', clearPointer);

/* ---- Animation loop ---- */
function animate(){
  ctx.clearRect(0,0,width,height);
  for(let p of particles){
    // attract/repel based on pointer
    if(pointer.active){
      const dx = p.x - pointer.x;
      const dy = p.y - pointer.y;
      const dist = Math.hypot(dx,dy);
      if(dist<80){
        const force = (80-dist)/80; // 0..1
        p.vx += (dx/dist)*force*0.5;
        p.vy += (dy/dist)*force*0.5;
        p.size = p.baseSize + force*2; // grow near pointer
      }else{
        p.size = p.baseSize;
      }
    }else{
      p.size = p.baseSize;
    }
    p.update();
    p.draw();
  }
  requestAnimationFrame(animate);
}
animate();

/* ---- Theme toggle (keyboard accessible) ---- */
const themeToggle = document.getElementById('themeToggle');
function updateTheme(){
  const isDark = document.documentElement.dataset.theme === 'dark';
  document.documentElement.dataset.theme = isDark?'light':'dark';
  themeToggle.textContent = isDark?'다크 모드':'라이트 모드';
  themeToggle.setAttribute('aria-pressed', !isDark);
}
themeToggle.addEventListener('click', updateTheme);
themeToggle.addEventListener('keydown', e=>{
  if(e.key==='Enter' || e.key===' ') { e.preventDefault(); updateTheme();}
});

/* ---- Reset button ---- */
document.getElementById('resetBtn').addEventListener('click',()=>{
  particles.forEach(p=>p.reset());
});

/* ---- Accessibility: prefers-reduced-motion ---- */
if(window.matchMedia('(prefers-reduced-motion: reduce)').matches){
  // stop animation, show static image
  ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--particle-color') || '#0066ff';
  ctx.font = '48px system-ui';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText('gpt-oss-120b', width/2, height/2);
}

/* ---- Keyboard navigation hint (visually hidden) ---- */
const hint = document.createElement('div');
hint.className = 'visually-hidden';
hint.innerHTML = '화면을 클릭하거나 터치하면 입자가 반응합니다. 다크 모드 전환은 버튼을 클릭하거나 Tab 키로 이동 후 Enter 또는 Space 키를 누르세요.';
document.body.appendChild(hint);
</script>
