  <script>
    (function () {
      const canvas = document.getElementById("c");
      const ctx = canvas.getContext("2d");
      const live = document.getElementById("live");
      const help = document.getElementById("help");
      const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      let w = 0, h = 0, t = 0;
      const nodes = [];
      const att = { x: 0.5, y: 0.42 };
      let pulse = 0;

      function resize() {
        w = canvas.clientWidth;
        h = canvas.clientHeight;
        canvas.width = Math.floor(w * dpr);
        canvas.height = Math.floor(h * dpr);
        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      }
      function seed() {
        nodes.length = 0;
        const n = Math.max(18, Math.floor((w * h) / 28000));
        for (let i = 0; i < n; i++) {
          nodes.push({
            x: Math.random(),
            y: 0.12 + Math.random() * 0.76,
            r: 1.2 + Math.random() * 2.2,
            p: Math.random() * Math.PI * 2,
            kind: Math.random() < 0.12 ? "gold" : "pale"
          });
        }
        announce("격자가 다시 펼쳐졌습니다. 점 " + nodes.length + "개.");
      }
      function announce(msg) { live.textContent = msg; }
      function addNode(nx, ny) {
        nodes.push({ x: nx, y: ny, r: 2.4, p: t, kind: "gold" });
        if (nodes.length > 90) nodes.shift();
        announce("생각이 하나 더해졌습니다.");
      }
      function draw() {
        ctx.clearRect(0, 0, w, h);
        const ax = att.x * w, ay = att.y * h;
        const motion = reduce ? 0 : 1;
        // faint grid
        ctx.strokeStyle = "rgba(244,239,230,0.045)";
        ctx.lineWidth = 1;
        const step = 48;
        for (let x = 0; x < w; x += step) {
          ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, h); ctx.stroke();
        }
        for (let y = 0; y < h; y += step) {
          ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke();
        }
        // links
        for (let i = 0; i < nodes.length; i++) {
          const a = nodes[i];
          const px = a.x * w + Math.sin(t * 0.4 + a.p) * 6 * motion;
          const py = a.y * h + Math.cos(t * 0.35 + a.p) * 5 * motion;
          for (let j = i + 1; j < nodes.length; j++) {
            const b = nodes[j];
            const qx = b.x * w + Math.sin(t * 0.4 + b.p) * 6 * motion;
            const qy = b.y * h + Math.cos(t * 0.35 + b.p) * 5 * motion;
            const dx = px - qx, dy = py - qy;
            const dist = Math.hypot(dx, dy);
            const maxd = Math.min(w, h) * 0.22;
            if (dist < maxd) {
              const mid = Math.hypot((px + qx) / 2 - ax, (py + qy) / 2 - ay);
              const attN = Math.max(0, 1 - mid / (Math.min(w, h) * 0.45));
              const alpha = (1 - dist / maxd) * (0.12 + attN * 0.35 + pulse * 0.2);
              ctx.strokeStyle = a.kind === "gold" || b.kind === "gold"
                ? "rgba(232,195,106," + alpha + ")"
                : "rgba(127,212,208," + (alpha * 0.85) + ")";
              ctx.lineWidth = 1 + attN;
              ctx.beginPath();
              ctx.moveTo(px, py);
              ctx.lineTo(qx, qy);
              ctx.stroke();
            }
          }
        }
        // nodes
        for (const a of nodes) {
          const px = a.x * w + Math.sin(t * 0.4 + a.p) * 6 * motion;
          const py = a.y * h + Math.cos(t * 0.35 + a.p) * 5 * motion;
          const near = Math.hypot(px - ax, py - ay);
          const glow = Math.max(0, 1 - near / 180) + pulse * 0.4;
          ctx.beginPath();
          ctx.fillStyle = a.kind === "gold"
            ? "rgba(232,195,106," + (0.75 + glow * 0.25) + ")"
            : "rgba(244,239,230," + (0.55 + glow * 0.35) + ")";
          ctx.arc(px, py, a.r + glow * 2.2, 0, Math.PI * 2);
          ctx.fill();
        }
        // attention
        ctx.beginPath();
        ctx.strokeStyle = "rgba(232,195,106," + (0.35 + pulse * 0.4) + ")";
        ctx.lineWidth = 1.5;
        ctx.arc(ax, ay, 18 + pulse * 40, 0, Math.PI * 2);
        ctx.stroke();
        pulse *= reduce ? 0 : 0.94;
      }
      function loop() {
        t += reduce ? 0 : 0.016;
        draw();
        requestAnimationFrame(loop);
      }
      function pointer(e) {
        const r = canvas.getBoundingClientRect();
        const x = (e.clientX ?? e.touches?.[0]?.clientX) - r.left;
        const y = (e.clientY ?? e.touches?.[0]?.clientY) - r.top;
        att.x = x / r.width;
        att.y = y / r.height;
      }
      window.addEventListener("resize", resize);
      canvas.addEventListener("pointermove", pointer);
      canvas.addEventListener("pointerdown", (e) => {
        pointer(e);
        const r = canvas.getBoundingClientRect();
        addNode((e.clientX - r.left) / r.width, (e.clientY - r.top) / r.height);
        pulse = 1;
      });
      document.getElementById("pulse").onclick = () => { pulse = 1; announce("파동."); };
      document.getElementById("reset").onclick = seed;
      document.getElementById("helpBtn").onclick = () => {
        const open = help.hasAttribute("hidden");
        help.toggleAttribute("hidden", !open);
        document.getElementById("helpBtn").setAttribute("aria-expanded", String(open));
      };
      window.addEventListener("keydown", (e) => {
        const step = 0.04;
        if (e.key === " ") { e.preventDefault(); pulse = 1; }
        if (e.key === "r" || e.key === "R") seed();
        if (e.key === "?" || e.key === "/") document.getElementById("helpBtn").click();
        if (e.key === "ArrowLeft") att.x = Math.max(0.04, att.x - step);
        if (e.key === "ArrowRight") att.x = Math.min(0.96, att.x + step);
        if (e.key === "ArrowUp") att.y = Math.max(0.08, att.y - step);
        if (e.key === "ArrowDown") att.y = Math.min(0.92, att.y + step);
      });
      resize();
      seed();
      requestAnimationFrame(loop);
    })();
  </script>
