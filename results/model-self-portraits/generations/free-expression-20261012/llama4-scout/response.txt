        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');

        // 미디어 아트: 움직이는 원
        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.beginPath();
            ctx.arc(canvas.width / 2 + Math.sin(Date.now() * 0.001) * 100, canvas.height / 2 + Math.cos(Date.now() * 0.001) * 100, 50, 0, 2 * Math.PI);
            ctx.fillStyle = '#007bff';
            ctx.fill();
            requestAnimationFrame(draw);
        }
        draw();
    </script>
