    <script>
        // Neural network data structure
        const neuralData = {
            nodes: [],
            connections: [],
            centerX: 0,
            centerY: 0,
            scale: 1.0,
            maxNodes: 100,
            maxConnections: 200,
            colors: [
                '#8a2be2', '#ff6b6b', '#4ecdc4', '#45b7d1',
                '#96ceb4', '#feca57', '#ff9ff3', '#54a0ff'
            ]
        };

        // Canvas setup
        const canvas = document.getElementById('neuralCanvas');
        const ctx = canvas.getContext('2d');
        let canvasWidth, canvasHeight;

        // Initialize with empty state
        function initializeNeuralData() {
            neuralData.nodes = [];
            neuralData.connections = [];
            neuralData.centerX = 0;
            neuralData.centerY = 0;
        }

        // Resize canvas to fit container
        function resizeCanvas() {
            canvasWidth = canvas.parentElement.clientWidth;
            canvasHeight = canvas.parentElement.clientHeight;
            canvas.width = canvasWidth;
            canvas.height = canvasHeight;

            // Update center coordinates
            neuralData.centerX = canvasWidth / 2;
            neuralData.centerY = canvasHeight / 2;

            // Initialize neural network after canvas is properly sized
            initNeuralNetwork();
        }

        // Initialize with random nodes
        function initNeuralNetwork() {
            neuralData.nodes = [];
            neuralData.connections = [];

            const nodeCount = Math.floor(Math.random() * 20) + 10;
            for (let i = 0; i < nodeCount; i++) {
                const angle = Math.random() * Math.PI * 2;
                const distance = (Math.random() * 0.4 + 0.3) * canvasWidth;
                const x = neuralData.centerX + Math.cos(angle) * distance;
                const y = neuralData.centerY + Math.sin(angle) * distance;

                neuralData.nodes.push({
                    x, y,
                    size: 5 + Math.random() * 10,
                    color: neuralData.colors[Math.floor(Math.random() * neuralData.colors.length)],
                    value: Math.random() * 2 - 1,
                    active: false
                });
            }

            // Create connections
            for (let i = 0; i < neuralData.nodes.length; i++) {
                for (let j = i + 1; j < neuralData.nodes.length; j++) {
                    if (Math.random() > 0.7) {
                        neuralData.connections.push({
                            from: i,
                            to: j,
                            strength: Math.random() * 0.5 + 0.2,
                            color: neuralData.colors[Math.floor(Math.random() * neuralData.colors.length)]
                        });
                    }
                }
            }

            drawNeuralNetwork();
        }

        // Draw the neural network
        function drawNeuralNetwork() {
            if (!canvasWidth || !canvasHeight) return;

            // Clear canvas
            ctx.fillStyle = 'rgba(26, 26, 46, 0.8)';
            ctx.fillRect(0, 0, canvasWidth, canvasHeight);

            // Draw connections
            ctx.lineWidth = 0.5;
            neuralData.connections.forEach(conn => {
                const from = neuralData.nodes[conn.from];
                const to = neuralData.nodes[conn.to];

                if (!from || !to) return;

                const midX = (from.x + to.x) / 2;
                const midY = (from.y + to.y) / 2;

                ctx.strokeStyle = conn.color;
                ctx.lineWidth = conn.strength * 2;
                ctx.beginPath();
                ctx.moveTo(from.x, from.y);
                ctx.quadraticCurveTo(midX, midY, to.x, to.y);
                ctx.stroke();

                ctx.shadowColor = conn.color;
                ctx.shadowBlur = conn.strength * 5;
                ctx.beginPath();
                ctx.arc(midX, midY, conn.strength * 3, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
            });

            // Draw nodes
            neuralData.nodes.forEach(node => {
                if (!node) return;

                const gradient = ctx.createRadialGradient(
                    node.x, node.y, node.size * 0.5,
                    node.x, node.y, node.size * 1.5
                );
                gradient.addColorStop(0, node.color);
                gradient.addColorStop(1, node.active ? '#ffffff' : 'rgba(255, 255, 255, 0.2)');

                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(node.x, node.y, node.size, 0, Math.PI * 2);
                ctx.fill();

                ctx.strokeStyle = node.active ? '#ffffff' : 'rgba(255, 255, 255, 0.3)';
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.arc(node.x, node.y, node.size, 0, Math.PI * 2);
                ctx.stroke();

                if (node.value > 0) {
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
                    ctx.beginPath();
                    ctx.arc(node.x, node.y - node.size * 1.5, node.size * 0.5, 0, Math.PI * 2);
                    ctx.fill();
                } else if (node.value < 0) {
                    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
                    ctx.beginPath();
                    ctx.arc(node.x, node.y + node.size * 1.5, node.size * 0.5, 0, Math.PI * 2);
                    ctx.fill();
                }
            });

            // Draw center point (model core)
            ctx.fillStyle = '#8a2be2';
            ctx.beginPath();
            ctx.arc(neuralData.centerX, neuralData.centerY, 15, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillStyle = '#ffffff';
            ctx.font = '12px Arial';
            ctx.textAlign = 'center';
            ctx.fillText('ministral-14b', neuralData.centerX, neuralData.centerY + 5);
        }

        // Interaction handling
        let isDragging = false;
        let draggedNode = null;
        let offsetX, offsetY;

        function handleCanvasInteraction(e) {
            const rect = canvas.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;

            // Check if we clicked on a node
            for (let i = neuralData.nodes.length - 1; i >= 0; i--) {
                const node = neuralData.nodes[i];
                if (!node) continue;

                const distance = Math.sqrt(
                    Math.pow(node.x - mouseX, 2) +
                    Math.pow(node.y - mouseY, 2)
                );

                if (distance <= node.size) {
                    isDragging = true;
                    draggedNode = i;
                    offsetX = node.x - mouseX;
                    offsetY = node.y - mouseY;
                    node.active = true;
                    drawNeuralNetwork();
                    playNeuralSound();
                    return;
                }
            }

            // If not on a node, create new connection
            if (draggedNode === null) {
                let closestNode = null;
                let minDistance = Infinity;

                neuralData.nodes.forEach((node, index) => {
                    if (!node) return;

                    const distance = Math.sqrt(
                        Math.pow(node.x - mouseX, 2) +
                        Math.pow(node.y - mouseY, 2)
                    );

                    if (distance < minDistance) {
                        minDistance = distance;
                        closestNode = index;
                    }
                });

                if (closestNode !== null && minDistance < 100) {
                    const tempConn = {
                        from: closestNode,
                        to: -1,
                        strength: 0.3,
                        color: neuralData.colors[Math.floor(Math.random() * neuralData.colors.length)]
                    };

                    const fromNode = neuralData.nodes[tempConn.from];
                    if (fromNode) {
                        ctx.strokeStyle = tempConn.color;
                        ctx.lineWidth = tempConn.strength * 2;
                        ctx.beginPath();
                        ctx.moveTo(fromNode.x, fromNode.y);
                        ctx.lineTo(mouseX, mouseY);
                        ctx.stroke();
                    }
                }
            }
        }

        function handleCanvasMove(e) {
            if (!isDragging) return;

            const rect = canvas.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;

            if (draggedNode !== null) {
                const node = neuralData.nodes[draggedNode];
                if (node) {
                    node.x = mouseX + offsetX;
                    node.y = mouseY + offsetY;
                    drawNeuralNetwork();
                }
            } else {
                const lastConn = neuralData.connections[neuralData.connections.length - 1];
                if (lastConn && lastConn.to === -1) {
                    const fromNode = neuralData.nodes[lastConn.from];
                    if (fromNode) {
                        ctx.strokeStyle = lastConn.color;
                        ctx.lineWidth = lastConn.strength * 2;
                        ctx.beginPath();
                        ctx.moveTo(fromNode.x, fromNode.y);
                        ctx.lineTo(mouseX, mouseY);
                        ctx.stroke();
                    }
                }
            }
        }

        function handleCanvasUp() {
            if (!isDragging) return;

            isDragging = false;

            if (draggedNode !== null) {
                const node = neuralData.nodes[draggedNode];
                if (node) {
                    node.active = false;
                    drawNeuralNetwork();
                }
                draggedNode = null;
            } else {
                const rect = canvas.getBoundingClientRect();
                const mouseX = (rect.left + rect.right) / 2;
                const mouseY = (rect.top + rect.bottom) / 2;

                for (let i = 0; i < neuralData.nodes.length; i++) {
                    const node = neuralData.nodes[i];
                    if (!node) continue;

                    const distance = Math.sqrt(
                        Math.pow(node.x - mouseX, 2) +
                        Math.pow(node.y - mouseY, 2)
                    );

                    if (distance < 50) {
                        const conn = {
                            from: neuralData.connections[neuralData.connections.length - 1].from,
                            to: i,
                            strength: 0.3,
                            color: neuralData.colors[Math.floor(Math.random() * neuralData.colors.length)]
                        };
                        neuralData.connections.push(conn);
                        break;
                    }
                }

                drawNeuralNetwork();
            }
        }

        // Event listeners
        canvas.addEventListener('mousedown', handleCanvasInteraction);
        canvas.addEventListener('mousemove', handleCanvasMove);
        canvas.addEventListener('mouseup', handleCanvasUp);

        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            handleCanvasInteraction(e.touches[0]);
        });

        canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            handleCanvasMove(e.touches[0]);
        });

        canvas.addEventListener('touchend', handleCanvasUp);

        // Button controls
        document.getElementById('resetBtn').addEventListener('click', initNeuralNetwork);
        document.getElementById('randomizeBtn').addEventListener('click', () => {
            neuralData.nodes.forEach(node => {
                if (node) node.value = Math.random() * 2 - 1;
            });
            drawNeuralNetwork();
        });

        document.getElementById('expandBtn').addEventListener('click', () => {
            neuralData.scale *= 1.2;
            neuralData.maxNodes = Math.min(neuralData.maxNodes * 1.5, 200);
            neuralData.maxConnections = Math.min(neuralData.maxConnections * 1.5, 500);

            const newNodeCount = Math.floor(Math.random() * 10) + 5;
            for (let i = 0; i < newNodeCount; i++) {
                const angle = Math.random() * Math.PI * 2;
                const distance = (Math.random() * 0.6 + 0.4) * canvasWidth;
                const x = neuralData.centerX + Math.cos(angle) * distance;
                const y = neuralData.centerY + Math.sin(angle) * distance;

                neuralData.nodes.push({
                    x, y,
                    size: 5 + Math.random() * 10,
                    color: neuralData.colors[Math.floor(Math.random() * neuralData.colors.length)],
                    value: Math.random() * 2 - 1,
                    active: false
                });
            }

            drawNeuralNetwork();
        });

        document.getElementById('contractBtn').addEventListener('click', () => {
            if (neuralData.nodes.length <= 10) return;

            neuralData.scale = Math.max(0.5, neuralData.scale * 0.8);
            neuralData.maxNodes = Math.max(10, Math.floor(neuralData.maxNodes * 0.7));
            neuralData.maxConnections = Math.max(50, Math.floor(neuralData.maxConnections * 0.7));

            const nodesToRemove = Math.floor(Math.random() * 5) + 2;
            for (let i = 0; i < nodesToRemove; i++) {
                if (neuralData.nodes.length > 10) {
                    const index = Math.floor(Math.random() * neuralData.nodes.length);
                    neuralData.nodes.splice(index, 1);

                    neuralData.connections = neuralData.connections.filter(conn => {
                        return conn.from !== index && conn.to !== index;
                    });
                }
            }

            drawNeuralNetwork();
        });

        // Audio synthesis
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        let oscillator = null;
        let gainNode = null;
        let volumeSlider = document.getElementById('volume');

        function playNeuralSound() {
            if (oscillator) {
                oscillator.stop();
            }

            const frequency = 200 + (neuralData.nodes.length * 10);
            const detune = (Math.random() - 0.5) * 1000;

            oscillator = audioCtx.createOscillator();
            oscillator.type = 'sine';
            oscillator.frequency.value = frequency;
            oscillator.detune.value = detune;

            gainNode = audioCtx.createGain();
            gainNode.gain.value = 0.1 * parseFloat(volumeSlider.value);

            oscillator.connect(gainNode);
            gainNode.connect(audioCtx.destination);

            oscillator.start();
            oscillator.stop(audioCtx.currentTime + 0.5);

            const lfo = audioCtx.createOscillator();
            lfo.type = 'sine';
            lfo.frequency.value = 5;

            const lfoGain = audioCtx.createGain();
            lfoGain.gain.value = 500;

            lfo.connect(lfoGain);
            lfoGain.connect(oscillator.frequency);

            lfo.start();
            lfo.stop(audioCtx.currentTime + 0.5);
        }

        volumeSlider.addEventListener('input', () => {
            if (gainNode) {
                gainNode.gain.value = 0.1 * parseFloat(volumeSlider.value);
            }
        });

        // Keyboard accessibility
        document.addEventListener('keydown', (e) => {
            if (e.key === ' ' || e.key === 'Enter') {
                e.preventDefault();
                initNeuralNetwork();
            } else if (e.key === 'r') {
                document.getElementById('randomizeBtn').click();
            } else if (e.key === 'e') {
                document.getElementById('expandBtn').click();
            } else if (e.key === 'c') {
                document.getElementById('contractBtn').click();
            }
        });

        // Initialize everything
        window.addEventListener('resize', () => {
            resizeCanvas();
        });

        // Initial setup after DOM is loaded
        window.addEventListener('load', () => {
            resizeCanvas();
        });
    </script>
