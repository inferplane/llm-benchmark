    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const text = document.getElementById('text');
        const button = document.getElementById('button');
        const pause = document.getElementById('pause');
        let poem = 1;
        let paused = false;
        let particles = [];
        let flowers = [];

        function init() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            for (let i = 0; i < 700; i++) {
                particles.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    vx: Math.random() * 2 - 1,
                    vy: Math.random() * 2 - 1,
                    r: Math.random() * 2
                });
            }
            for (let i = 0; i < 100; i++) {
                flowers.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    vx: Math.random() * 2 - 1,
                    vy: Math.random() * 2 - 1,
                    r: Math.random() * 10
                });
            }
            draw();
        }

        function draw() {
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            for (let i = 0; i < particles.length; i++) {
                ctx.beginPath();
                ctx.arc(particles[i].x, particles[i].y, particles[i].r, 0, 2 * Math.PI);
                ctx.fillStyle = '#fff';
                ctx.fill();
                particles[i].x += particles[i].vx;
                particles[i].y += particles[i].vy;
                if (particles[i].x < 0 || particles[i].x > canvas.width) {
                    particles[i].vx = -particles[i].vx;
                }
                if (particles[i].y < 0 || particles[i].y > canvas.height) {
                    particles[i].vy = -particles[i].vy;
                }
            }
            for (let i = 0; i < flowers.length; i++) {
                ctx.beginPath();
                ctx.arc(flowers[i].x, flowers[i].y, flowers[i].r, 0, 2 * Math.PI);
                ctx.fillStyle = '#fff';
                ctx.fill();
                flowers[i].x += flowers[i].vx;
                flowers[i].y += flowers[i].vy;
                if (flowers[i].x < 0 || flowers[i].x > canvas.width) {
                    flowers[i].vx = -flowers[i].vx;
                }
                if (flowers[i].y < 0 || flowers[i].y > canvas.height) {
                    flowers[i].vy = -flowers[i].vy;
                }
            }
            ctx.font = '24px Arial';
            ctx.fillStyle = '#fff';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(document.getElementById('poem' + poem).textContent, canvas.width / 2, canvas.height / 2);
            requestAnimationFrame(draw);
        }

        function handleMouseMove(event) {
            if (!paused) {
                for (let i = 0; i < particles.length; i++) {
                    let dx = event.clientX - particles[i].x;
                    let dy = event.clientY - particles[i].y;
                    let angle = Math.atan2(dy, dx);
                    particles[i].vx += Math.cos(angle) * 0.1;
                    particles[i].vy += Math.sin(angle) * 0.1;
                }
            }
        }

        function handleMouseDown(event) {
            if (!paused) {
                for (let i = 0; i < particles.length; i++) {
                    let dx = event.clientX - particles[i].x;
                    let dy = event.clientY - particles[i].y;
                    let angle = Math.atan2(dy, dx);
                    particles[i].vx += Math.cos(angle) * 2;
                    particles[i].vy += Math.sin(angle) * 2;
                }
            }
        }

        function handleTouchMove(event) {
            if (!paused) {
                for (let i = 0; i < particles.length; i++) {
                    let dx = event.touches[0].clientX - particles[i].x;
                    let dy = event.touches[0].clientY - particles[i].y;
                    let angle = Math.atan2(dy, dx);
                    particles[i].vx += Math.cos(angle) * 0.1;
                    particles[i].vy += Math.sin(angle) * 0.1;
                }
            }
        }

        function handleTouchEnd(event) {
            if (!paused) {
                for (let i = 0; i < particles.length; i++) {
                    let dx = event.changedTouches[0].clientX - particles[i].x;
                    let dy = event.changedTouches[0].clientY - particles[i].y;
                    let angle = Math.atan2(dy, dx);
                    particles[i].vx += Math.cos(angle) * 2;
                    particles[i].vy += Math.sin(angle) * 2;
                }
            }
        }

        function handleButtonClick() {
            if (!paused) {
                poem++;
                if (poem > 3) {
                    poem = 1;
                }
                text.style.top = '50%';
                text.style.left = '50%';
                text.style.transform = 'translate(-50%, -50%)';
            }
        }

        function handlePauseClick() {

                text.style.top = '50%';
                text.style.left = '50%';

                text.style.left = '50%';
                text.style.transform = 'translate(-50%, -50%)';
