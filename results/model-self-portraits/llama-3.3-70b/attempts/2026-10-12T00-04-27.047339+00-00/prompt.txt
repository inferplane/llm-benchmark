<script>
let canvas = document.getElementById('canvas');
let ctx = canvas.getContext('2d');
let nextButton = document.getElementById('next');
let pauseButton = document.getElementById('pause');
let paused = false;
let scene = 0;
let particles = [];
let llamaText = 'llama-3.3-70b';
let poem = ['나는 생각하는 존재다', '내 안에 세계가 있다', '그 세계는 무한하다'];
let translation = ['I am a thinking being', 'I have a world inside me', 'That world is infinite'];

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

function init() {
  for (let i = 0; i < 500; i++) {
    particles.push({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height,
      vx: Math.random() * 2 - 1,
      vy: Math.random() * 2 - 1,
      radius: Math.random() * 2
    });
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = '#ffffff';
  ctx.font = '24px sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(poem[scene], canvas.width / 2, canvas.height / 2);
  ctx.font = '18px sans-serif';
  ctx.fillText(translation[scene], canvas.width / 2, canvas.height / 2 + 30);
  ctx.font = '36px sans-serif';
  ctx.fillText(llamaText, canvas.width / 2, 50);
  
  for (let i = 0; i < particles.length; i++) {
    ctx.beginPath();
    ctx.arc(particles[i].x, particles[i].y, particles[i].radius, 0, 2 * Math.PI);
    ctx.fillStyle = '#cccccc';
    ctx.fill();
    particles[i].x += particles[i].vx;
    particles[i].y += particles[i].vy;
    
    if (particles[i].x < 0 || particles[i].x > canvas.width) {
      particles[i].vx = -particles[i].vx;
    }
    if (particles[i].y < 0 || particles[i].y > canvas.height) {
      particles[i].vy = -particles[i].vy;
    }
  }
}

function update() {
  if (!paused) {
    draw();
  }
  requestAnimationFrame(update);
}

init();
update();

nextButton.addEventListener('click', () => {
  scene = (scene + 1) % poem.length;
});

pauseButton.addEventListener('click', () => {
  paused = !paused;
  if (!paused) {
    update();
  }
});

canvas.addEventListener('mousemove', (e) => {
  for (let i = 0; i < particles.length; i++) {
    let dx = e.clientX - particles[i].x;
    let dy = e.clientY - particles[i].y;
    let distance = Math.sqrt(dx * dx + dy * dy);
    if (distance < 50) {
      particles[i].vx = -particles[i].vx;
      particles[i].vy = -particles[i].vy;
    }
  }
});

if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
  paused = true;
}

window.addEventListener('blur', () => {
  paused = true;
});

window.addEventListener('focus', () => {
  if (!paused) {
    update();
  }
});
