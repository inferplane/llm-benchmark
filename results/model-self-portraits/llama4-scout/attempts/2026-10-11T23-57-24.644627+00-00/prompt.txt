<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>llama4-scout</title>

    </style>
</head>
<body>
    <canvas id="canvas"></canvas>
    <div class="control">
        <button id="pause-btn">Pause</button>
        <button id="next-btn">Next</button>
    </div>

    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const pauseBtn = document.getElementById('pause-btn');
        const nextBtn = document.getElementById('next-btn');

        let width = window.innerWidth;
        let height = window.innerHeight;
        let particles = [];
        let paused = false;
        let currentScene = 0;

        canvas.width = width;
        canvas.height = height;

        const scenes = [
            {
                poem: ["언어와 지능의 흐름", "산책하는 생각", "llama4-scout입니다"],
                translation: ["Flow of language and intelligence", "Walking thoughts", "llama4-scout"],
                title: ""
            },
            {
                poem: ["미디어 아티스트의 꿈", "창조적인 동반자", "예술과 기술의 만남"],
                translation: ["Dreams of a media artist", "Creative companion", "Meeting art and technology"],
                title: ""
            }
        ];

        function init() {
            for (let i = 0; i < 300; i++) {
                particles.push({
                    x: Math.random() * width,
                    y: Math.random() * height,
                    vx: Math.random() * 2 - 1,
                    vy: Math.random() * 2 - 1,
                    radius: Math.random() * 2
                });
            }
            draw();
        }

        function draw() {
            if (paused) return;
            ctx.clearRect(0, 0, width, height);
            for (let i = 0; i < particles.length; i++) {
                const particle = particles[i];
                ctx.beginPath();
                ctx.arc(particle.x, particle.y, particle.radius, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(0, 0, 0, ${0.5 - particle.radius / 10})`;
                ctx.fill();
                particle.x += particle.vx;
                particle.y += particle.vy;
                if (particle.x < 0 || particle.x > width) particle.vx *= -1;
                if (particle.y < 0 || particle.y > height) particle.vy *= -1;
            }
            requestAnimationFrame(draw);
        }

        function renderText() {
            ctx.font = '24px sans-serif';
            ctx.fillStyle = '#000';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('llama4-scout', width / 2, 20);
            ctx.font = '18px sans-serif';
            ctx.fillText(scenes[currentScene].poem.join('\n'), width / 2, height / 2);
            ctx.font = '16px sans-serif';
            ctx.fillText(scenes[currentScene].translation.join('\n'), width / 2, height / 2 + 30);
        }

        function render() {
            renderText();
            draw();
        }

        init();
        render();

        pauseBtn.addEventListener('click', () => {
            paused = !paused;
            if (paused) {
                pauseBtn.textContent = 'Play';
            } else {
                pauseBtn.textContent = 'Pause';
                draw();
            }
        });

        nextBtn.addEventListener('click', () => {
            currentScene = (currentScene + 1) % scenes.length;
            renderText();
        });

        canvas.addEventListener('mousemove', (e) => {
            if (!paused) {
                for (let i = 0; i < particles.length; i++) {
                    const particle = particles[i];
                    const dx = e.clientX - particle.x;
                    const dy = e.clientY - particle.y;
                    const dist = Math.sqrt(dx * dx + dy * dy);
                    if (dist < 50) {
                        particle.vx = Math.random() * 2 - 1;
                        particle.vy = Math.random() * 2 - 1;
                    }
                }
            }
        });

        canvas.addEventListener('click', (e) => {
            if (e.clientX > width / 2) {
                currentScene = (currentScene + 1) % scenes.length;
                renderText();
            }
        });

        window.addEventListener('resize', () => {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        });

        if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
            paused = true;
            pauseBtn.textContent = 'Play';
        }

        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'hidden') {
                paused = true;
            }
        });
    </script>
</body>
</html>
