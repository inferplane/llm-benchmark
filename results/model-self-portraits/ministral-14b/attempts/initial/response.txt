        // State
        const state = {
            canvas: null,
            ctx: null,
            particles: [],
            ripples: [],
            mouse: { x: 0, y: 0 },
            touch: { x: 0, y: 0 },
            isPaused: false,
            isReducedMotion: false,
            currentScene: 0,
            lastTime: 0,
            animationId: null,
            devicePixelRatio: 1,
            canvasWidth: 0,
            canvasHeight: 0,
            centerX: 0,
            centerY: 0
        };

        // Initialize
        function init() {
            state.canvas = document.getElementById('canvas');
            state.ctx = state.canvas.getContext('2d');

            // Check for reduced motion
            state.isReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

            // Set up event listeners
            setupEventListeners();

            // Initialize canvas
            resizeCanvas();
            window.addEventListener('resize', resizeCanvas);

            // Create particles
            createParticles();

            // Set up UI
            setupUI();

            // Start animation
            if (!state.isReducedMotion) {
                state.lastTime = performance.now();
                state.animationId = requestAnimationFrame(animate);
            }
        }

        // Setup event listeners
        function setupEventListeners() {
            // Mouse events
            state.canvas.addEventListener('mousemove', (e) => {
                if (state.isPaused) return;

                const rect = state.canvas.getBoundingClientRect();
                state.mouse.x = e.clientX - rect.left;
                state.mouse.y = e.clientY - rect.top;
            });

            state.canvas.addEventListener('mousedown', (e) => {
                if (state.isPaused) return;
                createRipple(e.clientX, e.clientY);
            });

            state.canvas.addEventListener('mouseleave', () => {
                state.mouse.x = state.centerX;
                state.mouse.y = state.centerY;
            });

            // Touch events
            state.canvas.addEventListener('touchmove', (e) => {
                if (state.isPaused) return;
                e.preventDefault();

                const rect = state.canvas.getBoundingClientRect();
                const touch = e.touches[0];
                state.touch.x = touch.clientX - rect.left;
                state.touch.y = touch.clientY - rect.top;
            });

            state.canvas.addEventListener('touchstart', (e) => {
                if (state.isPaused) return;
                e.preventDefault();
                const rect = state.canvas.getBoundingClientRect();
                const touch = e.touches[0];
                createRipple(touch.clientX, touch.clientY);
            });

            // Button controls
            document.getElementById('next-btn').addEventListener('click', () => {
                state.currentScene = (state.currentScene + 1) % config.scenes.length;
                updateUI();
            });

            document.getElementById('pause-btn').addEventListener('click', () => {
                state.isPaused = !state.isPaused;
                if (state.isPaused) {
                    cancelAnimationFrame(state.animationId);
                } else {
                    state.lastTime = performance.now();
                    state.animationId = requestAnimationFrame(animate);
                }
            });
        }

        // Resize canvas
        function resizeCanvas() {
            state.devicePixelRatio = Math.min(2, window.devicePixelRatio || 1);
            state.canvasWidth = window.innerWidth * state.devicePixelRatio;
            state.canvasHeight = window.innerHeight * state.devicePixelRatio;

            state.canvas.style.width = `${window.innerWidth}px`;
            state.canvas.style.height = `${window.innerHeight}px`;
            state.canvas.width = state.canvasWidth;
            state.canvas.height = state.canvasHeight;

            state.centerX = state.canvasWidth / 2;
            state.centerY = state.canvasHeight / 2;
        }

        // Create particles
        function createParticles() {
            state.particles = [];

            for (let i = 0; i < config.particles; i++) {
                const angle = Math.random() * Math.PI * 2;
                const distance = Math.random() * Math.min(state.canvasWidth, state.canvasHeight) * 0.4;
                const speed = config.maxSpeed * Math.random();
                const size = config.baseSize + Math.random() * config.maxSize;

                state.particles.push({
                    x: state.centerX + Math.cos(angle) * distance,
                    y: state.centerY + Math.sin(angle) * distance,
                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    size: size,
                    targetX: state.centerX,
                    targetY: state.centerY,
                    color: config.colors[Math.floor(Math.random() * config.colors.length)],
                    originalColor: config.colors[Math.floor(Math.random() * config.colors.length)],
                    lifetime: Math.random() * 10000,
                    age: 0,
                    isRipple: false,
                    rippleProgress: 0
                });
            }
        }

        // Create ripple effect
        function createRipple(clientX, clientY) {
            const rect = state.canvas.getBoundingClientRect();
            const x = clientX - rect.left;
            const y = clientY - rect.top;

            // Find nearest particle to create ripple
            let closestParticle = null;
            let minDistance = Infinity;

            state.particles.forEach(particle => {
                const dx = particle.x - x;
                const dy = particle.y - y;
                const distance = Math.sqrt(dx * dx + dy * dy);

                if (distance < minDistance && !particle.isRipple) {
                    minDistance = distance;
                    closestParticle = particle;
                }
            });

            if (closestParticle) {
                closestParticle.isRipple = true;
                closestParticle.rippleProgress = 0;
                closestParticle.targetX = x;
                closestParticle.targetY = y;
                closestParticle.vx = (x - closestParticle.x) * 0.1;
                closestParticle.vy = (y - closestParticle.y) * 0.1;
            }
        }

        // Update UI
        function updateUI() {
            const scene = config.scenes[state.currentScene];
            const poemText = document.querySelector('.poem-text');
            const poemTitle = document.querySelector('.poem-title');
            const poemAuthor = document.querySelector('.poem-author');

            poemTitle.textContent = scene.title;
            poemText.innerHTML = scene.text.join('');
            poemAuthor.textContent = scene.author;
        }

        // Animation loop
        function animate(timestamp) {
            if (state.isPaused) {
                state.animationId = requestAnimationFrame(animate);
                return;
            }

            const deltaTime = timestamp - state.lastTime;
            state.lastTime = timestamp;

            // Clear canvas
            state.ctx.clearRect(0, 0, state.canvasWidth, state.canvasHeight);

            // Draw background gradient
            const gradient = state.ctx.createRadialGradient(
                state.centerX, state.centerY, 0,
                state.centerX, state.centerY, Math.max(state.canvasWidth, state.canvasHeight) * 0.6
            );
            gradient.addColorStop(0, 'rgba(10, 14, 35, 0.8)');
            gradient.addColorStop(1, 'rgba(26, 16, 48, 0.9)');
            state.ctx.fillStyle = gradient;
            state.ctx.fillRect(0, 0, state.canvasWidth, state.canvasHeight);

            // Update and draw particles
            state.particles.forEach(particle => {
                // Handle ripple effect
                if (particle.isRipple) {
                    particle.rippleProgress += deltaTime / config.rippleDuration;

                    if (particle.rippleProgress >= 1) {
                        particle.isRipple = false;
                        particle.rippleProgress = 0;
                        particle.vx = (particle.targetX - particle.x) * 0.05;
                        particle.vy = (particle.targetY - particle.y) * 0.05;
                    } else {
                        // Ripple movement
                        const progress = particle.rippleProgress;
                        const targetX = particle.targetX;
                        const targetY = particle.targetY;

                        particle.x = particle.x + (targetX - particle.x) * progress * 0.5;
                        particle.y = particle.y + (targetY - particle.y) * progress * 0.5;

                        // Change color during ripple
                        const hue = 240 + Math.sin(progress * Math.PI * 2) * 30;
                        particle.color = `hsla(${hue}, 100%, 50%, ${0.7 + Math.sin(progress * Math.PI) * 0.3})`;
                    }
                } else {
                    // Normal particle movement
                    particle.age += deltaTime;

                    // Apply mouse/touch attraction
                    const mouseDistance = Math.sqrt(
                        Math.pow(particle.x - state.mouse.x, 2) +
                        Math.pow(particle.y - state.mouse.y, 2)
                    );

                    const touchDistance = Math.sqrt(
                        Math.pow(particle.x - state.touch.x, 2) +
                        Math.pow(particle.y - state.touch.y, 2)
                    );

                    const minDistance = Math.min(mouseDistance, touchDistance);
                    const attractionForce = Math.max(0, 1 - minDistance / (Math.min(state.canvasWidth, state.canvasHeight) * 0.3));

                    if (attractionForce > 0.1) {
                        const angle = Math.atan2(
                            (state.mouse.x + state.touch.x) / 2 - particle.x,
                            (state.mouse.y + state.touch.y) / 2 - particle.y
                        );

                        particle.vx += Math.cos(angle) * attractionForce * config.mouseSensitivity;
                        particle.vy += Math.sin(angle) * attractionForce * config.mouseSensitivity;
                    }

                    // Apply gravity and friction
                    particle.vy += config.gravity;
                    particle.vx *= config.friction;
                    particle.vy *= config.friction;

                    // Update position
                    particle.x += particle.vx;
                    particle.y += particle.vy;

                    // Boundary check
                    if (particle.x < 0) {
                        particle.x = 0;
                        particle.vx *= -0.8;
