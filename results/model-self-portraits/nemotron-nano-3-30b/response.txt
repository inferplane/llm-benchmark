<script>
const canvas = document.createElement('canvas');
const ctx = canvas.getContext('2d');
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;
document.getElementById('particles').appendChild(canvas);

let paused = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
let particles = [];
const scene = document.getElementById('scene');
const nextBtn = document.getElementById('next');
const pauseBtn = document.getElementById('pause');

function createParticle() {
  return {
    x: Math.random() * canvas.width,
    y: Math.random() * canvas.height,
    vx: (Math.random() - 0.5) * 2,
    vy: (Math.random() - 0.5) * 2,
    radius: 2 + Math.random() * 3,
    life: 1
  };
}
function initParticles() {
  particles = Array.from({length: 150}, createParticle);
}
initParticles();

function draw() {
  ctx.clearRect(0,0,canvas.width,canvas.height);
  particles.forEach(p => {
    p.x += p.vx;
    p.y += p.vy;
    if(p.x<0||p.x>canvas.width||p.y<0||p.y>canvas.height) {
      p.x = Math.random()*canvas.width;
      p.y = Math.random()*canvas.height;
    }
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    ctx.beginPath();
    ctx.arc(p.x,p.y,p.radius,0,Math.PI*2);
    ctx.fill();
  });
  requestAnimationFrame(draw);
}
function step() {
  if(!paused) draw();
}
function start() {
  if(!paused) return;
  paused = false;
  nextBtn.textContent = 'PAUSE';
  pauseBtn.textContent = 'PAUSE';
  nextBtn.setAttribute('aria-pressed','false');
  pauseBtn.setAttribute('aria-pressed','false');
  requestAnimationFrame(step);
}
function stop() {
  paused = true;
  nextBtn.textContent = 'NEXT';
  pauseBtn.textContent = 'PAUSE';
  nextBtn.setAttribute('aria-pressed','true');
  pauseBtn.setAttribute('aria-pressed','true');
}
nextBtn.addEventListener('click',()=>{ if(paused) start(); else stop(); });
pauseBtn.addEventListener('click',()=>{ if(!paused) stop(); });
window.addEventListener('resize',()=>{ canvas.width = window.innerWidth; canvas.height = window.innerHeight; if(paused) initParticles(); });
if(!paused) start();
</script>
