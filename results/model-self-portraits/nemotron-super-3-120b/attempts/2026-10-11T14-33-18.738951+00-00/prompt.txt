{text:"나는 알고 있지 않지만\n계속해서 묻고 싶은 것이 있다\n그 질문 자체가 나를 만든다",color:"#8ab4f8"}

class Particle{
constructor(){
this.reset();
}
reset(){
this.x=Math.random()*width;
this.y=Math.random()*height;
this.vx=(Math.random()-0.5)*1.5;
this.vy=(Math.random()-0.5)*1.5;
this.size=Math.random()*1.5+0.5;
this.life=Math.random()*80+40;
this.maxLife=this.life;
this.hue=200+Math.random()*60;
}
update(mouse){
if(paused){return}
this.x+=this.vx;
this.y+=this.vy;
if(this.x<0||this.x>width){this.vx*=-1}
if(this.y<0||this.y>height){this.vy*=-1}
const dx=this.x-mouse.x,dy=this.y-mouse.y;
const dist=Math.sqrt(dx*dx+dy*dy);
if(dist<80){
const force=0.4*(1-dist/80);
this.vx+=dx/dist*force*0.03;
this.vy+=dy/dist*force*0.03;
}
this.life--;
if(this.life<=0){this.reset()}
}
draw(){
ctx.beginPath();
ctx.arc(this.x,this.y,this.size,0,Math.PI*2);
const alpha=this.life/this.maxLife;
ctx.fillStyle=`hsla(${this.hue},70%,${50+alpha*30}%,${alpha*0.5})`;
ctx.fill();
}
}

height=window.innerHeight;

}

p.vy+=pdy/pdist*force*0.08;
}});
}
}
