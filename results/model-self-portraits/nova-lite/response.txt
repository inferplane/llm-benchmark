```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <style>
        body, html {
            margin: 0;
            padding: 0;
            overflow: hidden;
            width: 100%;
            height: 100%;
            display: flex;
            justify-content: center;
            align-items: center;
            background: #000;
            color: #fff;
            font-family: 'Dotum', sans-serif;
        }
        canvas {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
        }
        .text-container {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            text-align: center;
            z-index: 10;
        }
        .title {
            font-size: 24px;
            margin-bottom: 20px;
        }
        .poetry {
            font-size: 18px;
            margin-bottom: 10px;
        }
        .translation {
            font-size: 14px;
        }
        .controls {
            position: absolute;
            bottom: 50px;
            left: 50%;
            transform: translateX(-50%);
            display: flex;
            justify-content: center;
            gap: 20px;
        }
       .control-button {
            padding: 12px 24px;
            background: rgba(255, 255, 255, 0.1);
            border: none;
            border-radius: 5px;
            cursor: pointer;
            color: #fff;
            font-size: 16px;
            outline: none;
        }
        .control-button:focus {
            outline: 2px solid #fff;
        }
    </style>
</head>
<body>
    <canvas id="canvas"></canvas>
    <div class="text-container">
        <div class="title">nova-lite</div>
        <div class="poetry" id="poetry">빛의 입자가 날아오르며, 우주의 숨결을 느낀다.</div>
        <div class="translation" id="translation">Particles of light soar, feeling the breath of the universe.</div>
    </div>
    <div class="controls">
        <button class="control-button" id="next" aria-pressed="false">NEXT</button>
        <button class="control-button" id="pause" aria-pressed="false">PAUSE</button>
    </div>
    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const poetryElement = document.getElementById('poetry');
        const translationElement = document.getElementById('translation');
        const nextButton = document.getElementById('next');
        const pauseButton = document.getElementById('pause');

        let isPaused = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
        let currentSceneIndex = 0;
        const scenes = [
            { korean: "빛의 입자가 날아오르며, 우주의 숨결을 느낀다.", english: "Particles of light soar, feeling the breath of the universe." },
            { korean: "깊은 우주의 검은 밤, 별들이 빛나는 순간.", english: "In the deep black night of space, the stars shine." }
        ];
        const particleCount = 700;
        const maxParticleSize = 2;
        const maxSpeed = 2;
        const minSpeed = 1;
        const maxOpacity = 0.5;
        const minOpacity = 0.1;
        let particles = [];

        function init() {
            canvas.width = window.innerWidth * window.devicePixelRatio;
            canvas.height = window.innerHeight * window.devicePixelRatio;
            ctx.scale(window.devicePixelRatio, window.devicePixelRatio);

            for (let i = 0; i < particleCount; i++) {
                particles.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    vx: (Math.random() - 0.5) * (maxSpeed - minSpeed) + minSpeed,
                    vy: (Math.random() - 0.5) * (maxSpeed - minSpeed) + minSpeed,
                    size: Math.random() * maxParticleSize + 1,
                    opacity: Math.random() * (maxOpacity - minOpacity) + minOpacity
                });
            }

            if (!isPaused) {
                requestAnimationFrame(animate);
            }
            drawStaticFrame();
        }

        function animate() {
            if (isPaused) return;

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            for (let i = 0; i < particles.length; i++) {
                const p = particles[i];
                p.x += p.vx;
                p.y += p.vy;

                if (p.x > canvas.width || p.x < 0) p.vx = -p.vx;
                if (p.y > canvas.height || p.y < 0) p.vy = -p.vy;

                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${p.opacity})`;
                ctx.fill();
            }

            requestAnimationFrame(animate);
        }

        function drawStaticFrame() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            for (let i = 0; i < particles.length; i++) {
                const p = particles[i];
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${p.opacity})`;
                ctx.fill();
            }
        }

        function changeScene() {
            currentSceneIndex = (currentSceneIndex + 1) % scenes.length;
            const nextScene = scenes[currentSceneIndex];

            poetryElement.textContent = `nova-lite\n\n${nextScene.korean}`;
            translationElement.textContent = nextScene.english;
            drawStaticFrame();
        }

        function togglePause() {
            isPaused = !isPaused;
            if (isPaused) {
                ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
            } else {
                requestAnimationFrame(animate);
            }
            pauseButton.textContent = isPaused ? 'PLAY' : 'PAUSE';
            pauseButton.setAttribute('aria-pressed', isPaused);
        }

        canvas.addEventListener('mousemove', (e) => {
            if (isPaused) return;
            for (let i = 0; i < particles.length; i++) {
                const p = particles[i];
                const dx = e.clientX - p.x;
                const dy = e.clientY - p.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                if (distance < 100) {
                    p.vx = (dx / distance) * 2;
                    p.vy = (dy / distance) * 2;
                }
            }
        });

        nextButton.addEventListener('click', changeScene);
        pauseButton.addEventListener('click', togglePause);

        init();
    </script>
</body>
</html>
```
