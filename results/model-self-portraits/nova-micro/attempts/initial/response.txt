```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>nova-micro 자기표현</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            overflow: hidden;
            height: 100%;
            width: 100%;
            display: flex;
            justify-content: center;
            align-items: center;
            background-color: #000;
        }
        canvas {
            position: absolute;
            top: 0;
            left: 0;
        }
        #content {
            position: relative;
            z-index: 1;
            display: flex;
            flex-direction: column;
            justify-content: center;
            align-items: center;
            color: #fff;
            font-family: 'Arial', sans-serif;
            text-align: center;
        }
        #text {
            font-size: 2rem;
            margin-bottom: 1rem;
        }
        #controls {
            display: flex;
            justify-content: center;
            gap: 1rem;
            margin-top: 1rem;
        }
        button {
            padding: 0.5rem 1rem;
            font-size: 1rem;
            background-color: #ff5733;
            color: #fff;
            border: none;
            cursor: pointer;
        }
    </style>
</head>
<body>
    <canvas id="canvas"></canvas>
    <div id="content">
        <div id="text">
            <span id="scene1-text">나는 nova-micro, 미디어 아티스트</span>
            <span id="scene1-translate">I am nova-micro, a media artist</span>
        </div>
        <div id="controls">
            <button id="next">다음</button>
            <button id="pause">일시정지</button>
        </div>
    </div>
    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
        const particles = [];
        const maxParticles = 700;
        const particleSize = 2;
        const colors = ['#ff5733', '#ffd700', '#00ced1', '#9acd32', '#dc143c'];
        let isPaused = false;
        let currentScene = 0;

        // Generate random particles
        function generateParticles() {
            for (let i = 0; i < maxParticles; i++) {
                particles.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    color: colors[Math.floor(Math.random() * colors.length)],
                    speedX: (Math.random() - 0.5) * 2,
                    speedY: (Math.random() - 0.5) * 2
                });
            }
        }

        // Draw particles
        function drawParticles() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            for (let i = 0; i < particles.length; i++) {
                const p = particles[i];
                ctx.beginPath();
                ctx.arc(p.x, p.y, particleSize, 0, Math.PI * 2);
                ctx.fillStyle = p.color;
                ctx.fill();
                p.x += p.speedX;
                p.y += p.speedY;
                if (p.x < 0 || p.x > canvas.width || p.y < 0 || p.y > canvas.height) {
                    particles[i] = {
                        x: Math.random() * canvas.width,
                        y: Math.random() * canvas.height,
                        color: colors[Math.floor(Math.random() * colors.length)],
                        speedX: (Math.random() - 0.5) * 2,
                        speedY: (Math.random() - 0.5) * 2
                    };
                }
            }
        }

        // Update particles
        function update() {
            if (!isPaused) {
                drawParticles();
                requestAnimationFrame(update);
            }
        }

        // Handle mouse/touch events
        function handleMouseMove(e) {
            if (!isPaused) {
                for (let i = 0; i < particles.length; i++) {
                    const p = particles[i];
                    const dx = e.clientX - p.x;
                    const dy = e.clientY - p.y;
                    if (dx * dx + dy * dy < 50 * 50) {
                        p.speedX = (dx / 50);
                        p.speedY = (dy / 50);
                    }
                }
            }
        }

        function handleClick(e) {
            if (!isPaused) {
                for (let i = 0; i < particles.length; i++) {
                    const p = particles[i];
                    const dx = e.clientX - p.x;
                    const dy = e.clientY - p.y;
                    if (dx * dx + dy * dy < 50 * 50) {
                        p.speedX = (dx / 50);
                        p.speedY = (dy / 50);
                        p.color = colors[Math.floor(Math.random() * colors.length)];
                    }
                }
            }
        }

        document.addEventListener('mousemove', handleMouseMove);
        document.addEventListener('click', handleClick);

        // Scene change
        function changeScene() {
            const textElement = document.getElementById('scene1-text');
            const translateElement = document.getElementById('scene1-translate');
            switch (currentScene) {
                case 0:
                    textElement.textContent = "나는 nova-micro, 미디어 아티스트";
                    translateElement.textContent = "I am nova-micro, a media artist";
                    break;
                case 1:
                    textElement.textContent = "나는 프런트엔드 개발자";
                    translateElement.textContent = "I am a front-end developer";
                    break;
                default:
                    textElement.textContent = "다시 시작합니다.";
                    translateElement.textContent = "Restarting...";
                    currentScene = 0;
            }
            currentScene = (currentScene + 1) % 2;
        }

        // Pause/resume
        function togglePause() {
            isPaused =!isPaused;
            if (!isPaused) {
                update();
            }
        }

        // Event listeners
        document.getElementById('next').addEventListener('click', changeScene);
        document.getElementById('pause').addEventListener('click', togglePause);

        // Initialize particles
        generateParticles();
        update();
    </script>
</body>
</html>
```
