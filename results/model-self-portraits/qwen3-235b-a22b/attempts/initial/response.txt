  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const dpr = Math.min(window.devicePixelRatio, 2);
    const maxParticles = 700;
    let particles = [];
    let isPaused = false;
    let currentScene = 1;
    const text1 = document.getElementById('text1');
    const text2 = document.getElementById('text2');
    const toggleBtn = document.getElementById('toggle');
    const nextBtn = document.getElementById('next');

    // Responsive setup
    function resize() {
      const rect = document.body.getBoundingClientRect();
      canvas.width = rect.width * dpr;
      canvas.height = rect.height * dpr;
      canvas.style.width = rect.width + 'px';
      canvas.style.height = rect.height + 'px';
      ctx.scale(dpr, dpr);
      initParticles();
    }
    window.addEventListener('resize', () => { if (!isPaused) resize(); });
    resize();

    // Check reduced motion preference
    const motionQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    if (motionQuery.matches) isPaused = true;

    // Particle system
    function initParticles() {
      particles = [];
      const count = Math.min(maxParticles, (canvas.width * canvas.height) / (dpr*10000) * 500);
      for (let i = 0; i < count; i++) {
        particles.push({
          x: Math.random() * canvas.clientWidth,
          y: Math.random() * canvas.clientHeight,
          vx: (Math.random() - 0.5) * 1.2,
          vy: (Math.random() - 0.5) * 1.2,
          radius: Math.random() * 1.5 + 0.5,
          alpha: Math.random() * 0.5 + 0.3,
          color: Math.random() > 0.5 ? '#6ee7b7' : '#a78bfa',
          trail: []
        });
      }
    }

    function drawParticles() {
      if (isPaused) return;
      ctx.clearRect(0, 0, canvas.clientWidth, canvas.clientHeight);
      
      particles.forEach(p => {
        p.trail.push({x: p.x, y: p.y});
        if (p.trail.length > 8) p.trail.shift();

        // Draw trail
        p.trail.forEach((point, i) => {
          const ratio = i / p.trail.length;
          ctx.beginPath();
          ctx.arc(point.x, point.y, p.radius * ratio, 0, Math.PI * 2);
          ctx.fillStyle = `${p.color}${Math.floor(ratio * p.alpha * 255).toString(16).padStart(2, '0')}`;
          ctx.fill();
        });

        // Move
        p.x += p.vx;
        p.y += p.vy;

        // Bounce
        if (p.x < 0 || p.x > canvas.clientWidth) p.vx *= -1;
        if (p.y < 0 || p.y > canvas.clientHeight) p.vy *= -1;

        // Attract to center slowly
        const dx = (canvas.clientWidth / 2 - p.x) * 0.005;
        const dy = (canvas.clientHeight / 2 - p.y) * 0.005;
        p.vx += dx;
        p.vy += dy;
      });

      requestAnimationFrame(drawParticles);
    }

    // Interaction
    let pointer = { x: canvas.clientWidth / 2, y: canvas.clientHeight / 2 };
    canvas.addEventListener('pointermove', e => {
      pointer = { x: e.clientX, y: e.clientY };
      if (isPaused) return;
      particles.forEach(p => {
        const dx = p.x - pointer.x;
        const dy = p.y - pointer.y;
        const dist = Math.hypot(dx, dy);
        if (dist < 100) {
          p.vx += dx / dist * 0.2;
          p.vy += dy / dist * 0.2;
        }
      });
    });

    canvas.addEventListener('pointerdown', e => {
      const cx = e.clientX;
      const cy = e.clientY;
      particles.forEach(p => {
        const dx = p.x - cx;
        const dy = p.y - cy;
        const dist = Math.hypot(dx, dy);
        if (dist < 150) {
          p.vx += (dx / dist) * 1.2;
          p.vy += (dy / dist) * 1.2;
        }
      });
    });

    // Controls
    toggleBtn.addEventListener('click', () => {
      isPaused = !isPaused;
      toggleBtn.textContent = isPaused ? 'PLAY' : 'PAUSE';
      if (!isPaused) drawParticles();
    });

    nextBtn.addEventListener('click', () => {
      text1.classList.toggle('hidden');
      text2.classList.toggle('hidden');
      currentScene = currentScene === 1 ? 2 : 1;
    });

    // Start animation unless reduced motion
    if (!motionQuery.matches) {
      drawParticles();
    } else {
      toggleBtn.textContent = 'PLAY';
    }

    // Reduce animation in background
    let hidden = false;
    document.addEventListener('visibilitychange', () => {
      hidden = document.hidden;
      if (hidden) {
        // Reduce CPU usage
        particles.forEach(p => {
          p.vx *= 0.1;
          p.vy *= 0.1;
        });
      }
    });
  </script>
