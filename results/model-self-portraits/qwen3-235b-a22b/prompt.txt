    let isPaused = false;

      canvas.height = rect.height * dpr;

      canvas.style.height = rect.height + 'px';

    // Check reduced motion preference

    // Particle system
    function initParticles() {
      particles = [];
      const count = Math.min(maxParticles, (canvas.width * canvas.height) / (dpr*10000) * 500);
      for (let i = 0; i < count; i++) {
        particles.push({
          x: Math.random() * canvas.clientWidth,
          y: Math.random() * canvas.clientHeight,
          vx: (Math.random() - 0.5) * 1.2,
          vy: (Math.random() - 0.5) * 1.2,
          radius: Math.random() * 1.5 + 0.5,
          alpha: Math.random() * 0.5 + 0.3,
          color: Math.random() > 0.5 ? '#6ee7b7' : '#a78bfa',
          trail: []
        });
      }
    }

    function drawParticles() {
      if (isPaused) return;
      ctx.clearRect(0, 0, canvas.clientWidth, canvas.clientHeight);
      
      particles.forEach(p => {
        p.trail.push({x: p.x, y: p.y});
        if (p.trail.length > 8) p.trail.shift();

        // Draw trail
        p.trail.forEach((point, i) => {
          const ratio = i / p.trail.length;
          ctx.beginPath();
          ctx.arc(point.x, point.y, p.radius * ratio, 0, Math.PI * 2);
          ctx.fillStyle = `${p.color}${Math.floor(ratio * p.alpha * 255).toString(16).padStart(2, '0')}`;
          ctx.fill();

        p.x += p.vx;
        p.y += p.vy;

        // Bounce

        p.vy += dy;
