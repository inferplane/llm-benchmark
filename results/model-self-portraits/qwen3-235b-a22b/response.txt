  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const maxParticles = 700;
    let particles = [];
    let isPaused = false;
    let currentScene = 1;
    const text1 = document.getElementById('text1');
    const text2 = document.getElementById('text2');
    const toggleBtn = document.getElementById('toggle');
    const nextBtn = document.getElementById('next');

    function resize() {
      const w = window.innerWidth;
      const h = window.innerHeight;
      canvas.width = w * dpr;
      canvas.height = h * dpr;
      canvas.style.width = w + 'px';
      canvas.style.height = h + 'px';
      ctx.scale(dpr, dpr);
      initParticles();
    }
    window.addEventListener('resize', resize);
    resize();

    const motionQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    if (motionQuery.matches) isPaused = true;

    function initParticles() {
      particles = [];
      const count = Math.min(maxParticles, (canvas.width * canvas.height) / (dpr * 12000) * 400);
      for (let i = 0; i < count; i++) {
        particles.push({
          x: Math.random() * window.innerWidth,
          y: Math.random() * window.innerHeight,
          vx: (Math.random() - 0.5) * 1.4,
          vy: (Math.random() - 0.5) * 1.4,
          radius: Math.random() * 1.8 + 0.4,
          alpha: Math.random() * 0.6 + 0.2,
          color: Math.random() > 0.6 ? '#4ade80' : Math.random() > 0.3 ? '#60a5fa' : '#f87171',
          trail: []
        });
      }
    }

    function drawParticles() {
      if (isPaused) return;
      ctx.clearRect(0, 0, window.innerWidth, window.innerHeight);
      
      particles.forEach(p => {
        p.trail.push({x: p.x, y: p.y});
        if (p.trail.length > 7) p.trail.shift();

        p.trail.forEach((point, i) => {
          const ratio = i / p.trail.length;
          ctx.beginPath();
          ctx.arc(point.x, point.y, p.radius * ratio * 1.2, 0, Math.PI * 2);
          const alpha = ratio * p.alpha;
          ctx.fillStyle = `${p.color}${Math.floor(alpha * 255).toString(16).padStart(2, '0')}`;
          ctx.fill();
        });

        p.x += p.vx;
        p.y += p.vy;

        if (p.x < 0 || p.x > window.innerWidth) p.vx *= -1;
        if (p.y < 0 || p.y > window.innerHeight) p.vy *= -1;

        const dx = (window.innerWidth / 2 - p.x) * 0.0008;
        const dy = (window.innerHeight / 2 - p.y) * 0.0008;
        p.vx += dx;
        p.vy += dy;
      });

      requestAnimationFrame(drawParticles);
    }

    let pointer = { x: window.innerWidth / 2, y: window.innerHeight / 2 };
    ['pointermove', 'touchmove'].forEach(evt => {
      canvas.addEventListener(evt, e => {
        e.preventDefault();
        const x = e.clientX || (e.touches && e.touches[0].clientX);
        const y = e.clientY || (e.touches && e.touches[0].clientY);
        pointer = { x, y };
        if (isPaused) return;
        particles.forEach(p => {
          const dx = p.x - pointer.x;
          const dy = p.y - pointer.y;
          const dist = Math.hypot(dx, dy);
          if (dist < 120) {
            p.vx += dx / dist * 0.3;
            p.vy += dy / dist * 0.3;
          }
        });
      });
    });

    ['pointerdown', 'touchstart'].forEach(evt => {
      canvas.addEventListener(evt, e => {
        const x = e.clientX || (e.touches && e.touches[0].clientX);
        const y = e.clientY || (e.touches && e.touches[0].clientY);
        particles.forEach(p => {
          const dx = p.x - x;
          const dy = p.y - y;
          const dist = Math.hypot(dx, dy);
          if (dist < 160) {
            p.vx += (dx / dist) * 1.4;
            p.vy += (dy / dist) * 1.4;
          }
        });
      });
    });

    toggleBtn.addEventListener('click', () => {
      isPaused = !isPaused;
      toggleBtn.textContent = isPaused ? 'PLAY' : 'PAUSE';
      if (!isPaused) drawParticles();
    });

    nextBtn.addEventListener('click', () => {
      text1.classList.toggle('hidden');
      text2.classList.toggle('hidden');
      currentScene = currentScene === 1 ? 2 : 1;
    });

    if (!motionQuery.matches) {
      drawParticles();
    } else {
      toggleBtn.textContent = 'PLAY';
    }

    let hidden = false;
    document.addEventListener('visibilitychange', () => {
      hidden = document.hidden;
      if (hidden && !isPaused) {
        particles.forEach(p => {
          p.vx *= 0.1;
          p.vy *= 0.1;
        });
      }
    });
  </script>
